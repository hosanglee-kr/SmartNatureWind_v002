1.2 N-16 — CL_A40_MutexGuard_Semaphore Lazy-init heap alloc 회피

대상: src/v0203/A40_Com_Utils_0203.h
위치: class CL_A40_MutexGuard_Semaphore 내 _internalInitAndTake()

Before

```cpp
    void _internalInitAndTake(TickType_t p_timeout) {
        if (!_mutexPtr) return;
        if (*_mutexPtr == nullptr) {
            static portMUX_TYPE s_initMux = portMUX_INITIALIZER_UNLOCKED;
            portENTER_CRITICAL(&s_initMux);
            if (*_mutexPtr == nullptr) {
                *_mutexPtr = xSemaphoreCreateRecursiveMutex();
                if (*_mutexPtr == nullptr)
                    CL_D10_Logger::log(EN_L10_LOG_ERROR, "[A40][%s] CreateRecursiveMutex failed", _caller);
            }
            portEXIT_CRITICAL(&s_initMux);
        }
        if (*_mutexPtr && xSemaphoreTakeRecursive(*_mutexPtr, p_timeout) == pdTRUE) _acquired = true;
    }
```

After

```cpp
    // ======================================================
    // [N-16] Lazy-init 재설계: heap alloc을 critical section 밖으로 이관
    //
    //  [배경]
    //   - ESP-IDF에서 portENTER_CRITICAL 내부의 heap alloc(xSemaphoreCreate*,
    //     malloc, new 등)은 인터럽트 금지 상태를 위반하여 assert/hang 유발.
    //   - 기존 코드는 xSemaphoreCreateRecursiveMutex()를 critical section 내부에서
    //     호출 → 콜드 부팅 시 여러 task가 동시에 초기 접근하면 데드락 위험.
    //
    //  [해결]
    //   ① Fast path: 이미 생성됨 → 바로 획득
    //   ② Slow path: critical 밖에서 생성 → critical 안에서 CAS (double-check)
    //   ③ 경합 패배한 task가 만든 mutex는 vSemaphoreDelete로 즉시 폐기
    // ======================================================
    void _internalInitAndTake(TickType_t p_timeout) {
        if (!_mutexPtr) return;

        // ① Fast path (이미 초기화 완료)
        if (*_mutexPtr != nullptr) {
            if (xSemaphoreTakeRecursive(*_mutexPtr, p_timeout) == pdTRUE) {
                _acquired = true;
            }
            return;
        }

        // ② Slow path: critical section "밖"에서 mutex 생성 (heap alloc 안전)
        SemaphoreHandle_t v_newMutex = xSemaphoreCreateRecursiveMutex();
        if (v_newMutex == nullptr) {
            CL_D10_Logger::log(EN_L10_LOG_ERROR,
                               "[A40][%s] CreateRecursiveMutex failed (heap exhausted?)",
                               _caller);
            return;
        }

        // ③ Critical section 안에서는 포인터 비교/대입만 수행
        static portMUX_TYPE s_initMux = portMUX_INITIALIZER_UNLOCKED;
        bool v_owned = false;

        portENTER_CRITICAL(&s_initMux);
        if (*_mutexPtr == nullptr) {
            *_mutexPtr = v_newMutex;   // 우리 것이 채택됨
            v_owned    = true;
        }
        portEXIT_CRITICAL(&s_initMux);

        // 경합 패배: 다른 task가 먼저 생성 → 우리 mutex 폐기
        if (!v_owned) {
            vSemaphoreDelete(v_newMutex);
        }

        // 최종 mutex 획득 (fast path 실패 후 재시도 포함)
        if (*_mutexPtr && xSemaphoreTakeRecursive(*_mutexPtr, p_timeout) == pdTRUE) {
            _acquired = true;
        }
    }
```

근거:

· xSemaphoreCreateRecursiveMutex()는 내부적으로 pvPortMalloc() 호출 → critical section 밖에서만 허용
· portENTER_CRITICAL(&s_initMux)는 포인터 비교 + 대입 만 수행 (수 μs)
· 경합 시 vSemaphoreDelete로 리소스 회수 → leak 없음
· 부팅 시 s_stateMutex, s_recursiveMutex, s_wifiMutex 등 10여 개 mutex의 lazy-init이 모두 이 클래스 경유 → 전역적 효과

회귀 리스크: 최초 부팅 시 heap 할당 1회 실패 시 vSemaphoreDelete 대상이 없어 안전. 다만 _acquired = false 상태로 반환되어 caller가 null-guard 필수 (기존과 동일).

검증: 콜드 부팅 10회 반복, esp_task_wdt 리셋 없이 정상 부팅 확인.

1.3 N-18 — Modal 리스너 누수 방지

대상: src/v0203/data_v0203_src/html_v0203/js/999.app.js
위치: SNW.UI.modal 함수 내 close 및 overlay.addEventListener

Before

```js
        let closed = false;
        const close = (confirmed) => {
            if (closed) return;
            closed = true;
            document.removeEventListener('keydown', _onEsc);
            overlay.classList.remove('is-open');
            setTimeout(() => {
                overlay.innerHTML = '';
                overlay.style.display = 'none';
                if (!confirmed && typeof onCancel === 'function') {
                    try { onCancel(); } catch (e) { }
                }
            }, 150);
        };

        const _onEsc = (e) => { if (e.key === 'Escape') close(false); };
        document.addEventListener('keydown', _onEsc);
        overlay.addEventListener('click', (e) => { if (e.target === overlay) close(false); });
```

After

```js
        // ======================================================
        // [N-18] 리스너 참조 보관 → close()에서 반드시 제거
        //  - 기존: overlay.addEventListener('click', 익명 함수) → 제거 불가
        //  - 문제: overlay 재사용 시 모달 열 때마다 리스너 누적
        //  - 해결: 이름 있는 핸들러 참조 → close()에서 removeEventListener
        // ======================================================
        let closed = false;
        let _onEsc = null;
        let _onOverlayClick = null;

        const close = (confirmed) => {
            if (closed) return;
            closed = true;

            // [N-18] 모든 리스너 해제 (중복 등록 방지)
            if (_onEsc)          document.removeEventListener('keydown', _onEsc);
            if (_onOverlayClick) overlay.removeEventListener('click', _onOverlayClick);
            _onEsc = null;
            _onOverlayClick = null;

            overlay.classList.remove('is-open');
            setTimeout(() => {
                overlay.innerHTML = '';
                overlay.style.display = 'none';
                if (!confirmed && typeof onCancel === 'function') {
                    try { onCancel(); } catch (e) { }
                }
            }, 150);
        };

        _onEsc = (e) => { if (e.key === 'Escape') close(false); };
        document.addEventListener('keydown', _onEsc);

        _onOverlayClick = (e) => { if (e.target === overlay) close(false); };
        overlay.addEventListener('click', _onOverlayClick);
```

근거:

· 누수 시나리오: 모달 N회 열기 → overlay에 N개의 클릭 리스너 누적. 각 리스너는 내부적으로 close(false)를 호출하나 closed 플래그로 중복 실행 방지됨. 다만 클릭 1회당 N개의 콜백 실행 → 미미한 성능 저하 + DevTools에서 리스너 카운트 증가 (디버깅 혼란)
· N-19 동시 해결: 이번 수정으로 onCancel이 ESC/외부클릭에서 정상 호출되므로 Phase 5.4의 simulationModal Promise hang도 부분 해소 (단, onCancel 콜백을 140.ui.simulation.js에서 제공하는 Phase 5.4 작업이 별도 필요)

회귀 리스크: close가 _onEsc/_onOverlayClick 참조 시점에 null 체크 → 최초 close 호출 전 리스너 등록 순서만 보장되면 안전. _onEsc는 close 정의 직후 등록되므로 문제 없음.

```cpp
bool CL_W10_WebAPI::checkApiKey(AsyncWebServerRequest* p_request) {

```

After

```cpp
bool CL_W10_WebAPI::checkApiKey(AsyncWebServerRequest* p_request) {
    // [A-3] null 방어 (Null Deref → Core Dump 방지)
    if (!p_request) return false;

    // ======================================================
    // [N-10] B-2 규약 준수: async_tcp 컨텍스트에서 g_A20_config_root
    //         직접 참조 금지 → getRootSnapshot 원자 스냅샷 사용
    //
    //  [문제]
    //   - 이전: g_A20_config_root.system 포인터 직접 역참조
    //   - reloadAll swap 중이면 dangling 또는 torn read 위험
    //     (E-1 3초 grace로 UAF는 방지되나 논리적 torn read 가능)
    //
    //  [해결]
    //   - s_rootSwapMux 보호 하에 8개 포인터 원자 캡처
    //   - 보안 키는 즉시 스택 버퍼로 복사 → 이후 swap과 무관
```

```cpp
// ======================================================
// [Phase3 N-1] Session 엔진 (ADR-1 위임 모델)
//
//  [설계 원칙]
//   - Session은 "표시 계층(_session mirror)" 이며, 실제 실행은
//     기존 검증 경로(startUserProfileByNo / startOverridePreset / startOverrideFixed)
//     에 위임한다.
//   - _session은 exportStateJson_v02에서만 read → FSM 결정에 절대 미참여.
//   - pause는 실행 상태 백업 + 물리 정지, resume은 백업 복원 + Smart Resume.
//
//  [파급]
//   - decideRunSource()가 overrideState/runSource 검사만으로
//     Session 실행 여부를 자동 판정 (별도 분기 추가 불필요)
//   - startSession 후 runSource/overrideState가 활성이므로
//     tickLoop → tickOverride / tickUserProfile 정상 진입
// ======================================================
bool CL_CT10_ControlManager::startSession(EN_CT10_session_type_t p_type,
                                          EN_CT10_trigger_t      p_trigger,
                                          const JsonDocument&    p_payloadDoc) {
    auto& self = instance();
    CL_A40_MutexGuard_Semaphore v_guard(s_stateMutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_guard.isAcquired()) return false;

    // ─────────────────────────────────────────────
    // ① 기존 세션/실행 상태 정리 (멱등)
    // ─────────────────────────────────────────────
    if (self._session.type != EN_CT10_SESSION_NONE) {
        CL_D10_Logger::log(EN_L10_LOG_INFO,
                           "[CT10] startSession: replacing existing (type=%u)",
                           (unsigned)self._session.type);
        self.stopSession();
    } else if (self.overrideState.active || self.runSource != EN_CT10_RUN_NONE) {
        // 세션 없이 override/profile 실행 중인 경우도 정리 (routeControl 경로)
        self.sim.stop();
        if (self.pwm) self.pwm->setDutyPercent(0.0f);
        if (self.overrideState.active) self.stopOverride();
        if (self.runSource == EN_CT10_RUN_USER_PROFILE) self.stopUserProfile();
    }

    // ─────────────────────────────────────────────
    // ② 타입별 위임 실행
    // ─────────────────────────────────────────────
    bool v_ok = false;

    switch (p_type) {
        // =====================================================
        // PLAYLIST → startUserProfileByNo 위임
        // =====================================================
        case EN_CT10_SESSION_PLAYLIST: {
            uint16_t v_profileId = A40_ComFunc::Json_getNum<uint16_t>(
                p_payloadDoc.as<JsonObjectConst>(), "profileId", 0);
            if (v_profileId == 0 || v_profileId > 255) {
                CL_D10_Logger::log(EN_L10_LOG_WARN,
                                   "[CT10] startSession(PLAYLIST): invalid profileId=%u",
                                   (unsigned)v_profileId);
                return false;
            }

            if (!g_A20_config_root.userProfiles) return false;
            auto& v_profiles = *g_A20_config_root.userProfiles;

            const ST_A20_UserProfileItem_t* v_profPtr = nullptr;
            for (uint8_t v_i = 0; v_i < v_profiles.count; v_i++) {
                if (v_profiles.items[v_i].profileId == (uint8_t)v_profileId) {
                    v_profPtr = &v_profiles.items[v_i];
                    break;
                }
            }
            if (!v_profPtr || !v_profPtr->enabled || v_profPtr->segCount == 0) {
                CL_D10_Logger::log(EN_L10_LOG_WARN,
                                   "[CT10] startSession(PLAYLIST): profile unavailable (id=%u)",
                                   (unsigned)v_profileId);
                return false;
            }

            // 위임 실행 (runSource=USER_PROFILE, curProfileIndex 설정, AutoOff 초기화)
            v_ok = self.startUserProfileByNo(v_profPtr->profileNo);
            if (v_ok) {
                self.setProfileMode(true);   // schedule 배제

                // payload mirror
                self._session.payload.playlist.profileId   = (uint8_t)v_profileId;
                self._session.payload.playlist.profileNo   = v_profPtr->profileNo;
                A40_ComFunc::copyStr2Buffer_safe(
                    self._session.payload.playlist.profileName,
                    v_profPtr->name,
                    sizeof(self._session.payload.playlist.profileName));
                self._session.payload.playlist.totalSteps         = v_profPtr->segCount;
                self._session.payload.playlist.currentStepIdx     = 0;
                self._session.payload.playlist.currentStepStartMs = millis();
                self._session.payload.playlist.currentStepDurationMs =
                    (uint32_t)v_profPtr->segments[0].durationMin * 60000UL;
            }
            break;
        }

        // =====================================================
        // PRESET → startOverridePreset 위임 (forever)
        // =====================================================
        case EN_CT10_SESSION_PRESET: {
            JsonObjectConst p = p_payloadDoc.as<JsonObjectConst>();
            const char* pc = A40_ComFunc::Json_getStr(p, "presetCode", A20_Const::DEFAULT_PRESET_CODE);
            const char* sc = A40_ComFunc::Json_getStr(p, "styleCode",  A20_Const::DEFAULT_STYLE_CODE);
            if (!pc || !pc[0] || !sc || !sc[0]) return false;

            ST_A20_SegmentOverride_t v_ovr;
            memset(&v_ovr, 0, sizeof(v_ovr));
            v_ovr.enabled         = false;
            v_ovr.windIntensity   = -1.0f;
            v_ovr.windVariability = -1.0f;
            v_ovr.gustFrequency   = -1.0f;

            if (p["override"].is<JsonObjectConst>()) {
                JsonObjectConst ovr   = p["override"].as<JsonObjectConst>();
                v_ovr.enabled         = ovr["enabled"] | false;
                if (ovr["windIntensity"].is<float>())
                    v_ovr.windIntensity   = ovr["windIntensity"].as<float>();
                if (ovr["windVariability"].is<float>())
                    v_ovr.windVariability = ovr["windVariability"].as<float>();
                if (ovr["gustFrequency"].is<float>())
                    v_ovr.gustFrequency   = ovr["gustFrequency"].as<float>();
            }

            self.startOverridePreset(pc, sc, &v_ovr, 0, true);   // forever
            v_ok = self.overrideState.active;

            if (v_ok) {
                A40_ComFunc::copyStr2Buffer_safe(
                    self._session.payload.preset.presetCode, pc,
                    sizeof(self._session.payload.preset.presetCode));
                A40_ComFunc::copyStr2Buffer_safe(
                    self._session.payload.preset.styleCode, sc,
                    sizeof(self._session.payload.preset.styleCode));
                self._session.payload.preset.override = v_ovr;
            }
            break;
        }

        // =====================================================
        // FIXED → startOverrideFixed 위임 (forever)
        // =====================================================
        case EN_CT10_SESSION_FIXED: {
            float v_pct = A40_ComFunc::Json_getNum<float>(
                p_payloadDoc.as<JsonObjectConst>(), "percent", 50.0f);
            if (v_pct < 0.0f)   v_pct = 0.0f;
            if (v_pct > 100.0f) v_pct = 100.0f;

            self.startOverrideFixed(v_pct, 0, true);   // forever
            v_ok = self.overrideState.active;

            if (v_ok) self._session.payload.fixed.percent = v_pct;
            break;
        }

        default:
            return false;
    }

    if (!v_ok) {
        CL_D10_Logger::log(EN_L10_LOG_WARN,
                           "[CT10] startSession: delegation failed (type=%u)",
                           (unsigned)p_type);
        return false;
    }

    // ─────────────────────────────────────────────
    // ③ 세션 메타 세팅 (mirror layer)
    // ─────────────────────────────────────────────
    self._session.type          = p_type;
    self._session.trigger       = p_trigger;
    self._session.state         = EN_SESSION_ACTIVE;
    self._session.startedAt     = millis();
    self._session.pausedAccumMs = 0;
    self._session.pausedAtMs    = 0;

    // ─────────────────────────────────────────────
    // ④ NVS 세션 영속화
    // ─────────────────────────────────────────────
    if (p_type == EN_CT10_SESSION_PLAYLIST && p_trigger == EN_CT10_TRIGGER_MANUAL) {
        CL_N10_NvsManager::setManualSession(
            self._session.payload.playlist.profileNo, true);
    } else {
        CL_N10_NvsManager::setManualSession(0, false);
    }

    // ─────────────────────────────────────────────
    // ⑤ Dirty + Monitor
    // ─────────────────────────────────────────────
    self.markDirty("state");
    self.markDirty("metrics");
    self.markDirty("summary");

    const char* v_label =
        (p_type == EN_CT10_SESSION_PLAYLIST) ? "재생 시작" :
        (p_type == EN_CT10_SESSION_PRESET)   ? "프리셋 재생" :
                                               "일반 선풍기";
    CL_MON_Monitor::pushSessionEvent(
        EN_MON_EVENT_SESSION_START, v_label,
        (uint8_t)p_type, (uint8_t)p_trigger,
        0, 0, self._session.payload.playlist.profileNo, 0);

    CL_D10_Logger::log(EN_L10_LOG_INFO,
                       "[CT10] startSession: type=%u trigger=%u (delegated)",
                       (unsigned)p_type, (unsigned)p_trigger);
    return true;
}
```

```cpp
// ======================================================
// [Phase3 N-1] pauseSession — 실행 상태 백업 + 물리 정지
// ======================================================
bool CL_CT10_ControlManager::pauseSession() {
    auto& self = instance();
    CL_A40_MutexGuard_Semaphore v_guard(s_stateMutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_guard.isAcquired()) return false;

    if (self._session.type == EN_CT10_SESSION_NONE) return false;
    if (self._session.state != EN_SESSION_ACTIVE)   return false;

    // ① 실행 컨텍스트 백업 (resumeSession에서 복원)
    self._pausedExec.valid        = true;
    self._pausedExec.runSource    = self.runSource;
    self._pausedExec.profileIndex = self.curProfileIndex;
    self._pausedExec.override     = self.overrideState;
    self._pausedExec.profileSegRt = self.profileSegRt;
    self._pausedExec.autoOffRt    = self.autoOffRt;

    // ② 물리 정지
    self.sim.stop();
    if (self.pwm) self.pwm->setDutyPercent(0.0f);

    // ③ 실행 상태 해제 (decideRunSource → IDLE 자연 판정)
    memset(&self.overrideState, 0, sizeof(self.overrideState));
    self.runSource       = EN_CT10_RUN_NONE;
    self.curProfileIndex = -1;
    memset(&self.profileSegRt, 0, sizeof(self.profileSegRt));
    self.profileSegRt.index = -1;

    // ④ 세션 상태
    self._session.state      = EN_SESSION_PAUSED;
    self._session.pausedAtMs = millis();

    self.markDirty("state");
    self.markDirty("metrics");

    CL_MON_Monitor::pushSessionEvent(
        EN_MON_EVENT_PAUSE, "일시정지",
        (uint8_t)self._session.type, (uint8_t)self._session.trigger,
        self._session.payload.playlist.currentStepIdx,
        self._session.payload.playlist.totalSteps, 0, 0);

    CL_D10_Logger::log(EN_L10_LOG_INFO, "[CT10] pauseSession()");
    return true;
}

// ======================================================
// [Phase3 N-1] resumeSession — 백업 복원 + Smart Resume
//  - <5분    : 연속 재개 (phaseStartMs 시프트)
//  - 5~30분  : Step 진행 유지, phase 재시작
//  - >30분   : Step 1 리셋
// ======================================================
bool CL_CT10_ControlManager::resumeSession() {
    auto& self = instance();
    CL_A40_MutexGuard_Semaphore v_guard(s_stateMutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_guard.isAcquired()) return false;

    if (self._session.type == EN_CT10_SESSION_NONE)  return false;
    if (self._session.state != EN_SESSION_PAUSED)    return false;
    if (!self._pausedExec.valid)                     return false;

    // ① 일시정지 시간 누적
    uint32_t v_pauseDuration = millis() - self._session.pausedAtMs;
    self._session.pausedAccumMs += v_pauseDuration;
    self._session.pausedAtMs = 0;

    // ② 실행 컨텍스트 복원
    self.runSource       = self._pausedExec.runSource;
    self.curProfileIndex = self._pausedExec.profileIndex;
    self.overrideState   = self._pausedExec.override;
    self.profileSegRt    = self._pausedExec.profileSegRt;
    self.autoOffRt       = self._pausedExec.autoOffRt;

    // ③ Smart Resume 정책
    if (self._session.type == EN_CT10_SESSION_PLAYLIST) {
        if (v_pauseDuration > S_PAUSE_STEP_RESET_MS) {
            // 30분 초과: Step 1부터 재시작
            self.profileSegRt.index        = -1;
            self.profileSegRt.phaseStartMs = millis();
            self.profileSegRt.loopCount    = 0;
            self._session.payload.playlist.currentStepIdx     = 0;
            self._session.payload.playlist.currentStepStartMs = millis();
            CL_D10_Logger::log(EN_L10_LOG_INFO,
                               "[CT10] resumeSession: >30min → restart from Step 1");
        } else if (v_pauseDuration > S_PAUSE_CONTINUE_MS) {
            // 5~30분: Step 유지, phase 재시작
            self.profileSegRt.phaseStartMs = millis();
            self._session.payload.playlist.currentStepStartMs = millis();
            CL_D10_Logger::log(EN_L10_LOG_INFO,
                               "[CT10] resumeSession: 5~30min → step phase reset");
        } else {
            // <5분: 연속 재개 (일시정지 시간만큼 시프트)
            self.profileSegRt.phaseStartMs += v_pauseDuration;
            self._session.payload.playlist.currentStepStartMs += v_pauseDuration;
            CL_D10_Logger::log(EN_L10_LOG_INFO,
                               "[CT10] resumeSession: <5min → continuous resume");
        }
    }

    self._session.state    = EN_SESSION_ACTIVE;
    self._pausedExec.valid = false;

    self.markDirty("state");

    CL_MON_Monitor::pushSessionEvent(
        EN_MON_EVENT_RESUME, "재개",
        (uint8_t)self._session.type, (uint8_t)self._session.trigger,
        self._session.payload.playlist.currentStepIdx,
        self._session.payload.playlist.totalSteps, 0, 0);

    CL_D10_Logger::log(EN_L10_LOG_INFO, "[CT10] resumeSession()");
    return true;
}

// ======================================================
// [Phase3 N-1] stopSession — 전체 실행 상태 리셋 (멱등)
// ======================================================
bool CL_CT10_ControlManager::stopSession() {
    auto& self = instance();
    CL_A40_MutexGuard_Semaphore v_guard(s_stateMutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_guard.isAcquired()) return false;

    // 멱등: 이미 완전 정지 상태면 no-op
    if (self._session.type == EN_CT10_SESSION_NONE &&
        !self.overrideState.active &&
        self.runSource == EN_CT10_RUN_NONE) {
        return false;
    }

    // ① 물리 정지
    self.sim.stop();
    if (self.pwm) self.pwm->setDutyPercent(0.0f);

    // ② 위임 경로 정리 (재귀 mutex → 안전)
    if (self.overrideState.active)                 self.stopOverride();
    if (self.runSource == EN_CT10_RUN_USER_PROFILE) self.stopUserProfile();

    // ③ 실행 상태 전체 리셋
    memset(&self.overrideState, 0, sizeof(self.overrideState));
    self.runSource        = EN_CT10_RUN_NONE;
    self.curProfileIndex  = -1;
    memset(&self.profileSegRt, 0, sizeof(self.profileSegRt));
    self.profileSegRt.index = -1;

    // ④ 세션 리셋 (Monitor 로그용으로 이전 값 보존)
    EN_CT10_session_type_t v_oldType   = self._session.type;
    EN_CT10_trigger_t      v_oldTrg    = self._session.trigger;
    uint16_t               v_oldProfNo = self._session.payload.playlist.profileNo;

    memset(&self._session, 0, sizeof(self._session));
    self._session.type  = EN_CT10_SESSION_NONE;
    self._session.state = EN_SESSION_COMPLETED;

    self._pausedExec.valid = false;

    // ⑤ NVS 해제
    CL_N10_NvsManager::setManualSession(0, false);
    if (CL_N10_NvsManager::isManualOvrSchedule()) {
        CL_N10_NvsManager::setManualOvrSchedule(false);
        CL_D10_Logger::log(EN_L10_LOG_INFO, "[CT10] Manual override released on stop");
    }

    self.markDirty("state");
    self.markDirty("metrics");
    self.markDirty("summary");

    CL_MON_Monitor::pushSessionEvent(
        EN_MON_EVENT_SESSION_END, "세션 종료",
        (uint8_t)v_oldType, (uint8_t)v_oldTrg,
        0, 0, v_oldProfNo, 0);

    CL_D10_Logger::log(EN_L10_LOG_INFO, "[CT10] stopSession()");
    return true;
}
```

```cpp
// ======================================================
// [Phase3 N-1] tickSchedule (ADR-3 profileId fallback)
//
//  [배경]
//   - Stage 5 UI(SchedulePanel)는 profileId만 저장 (segments 없음)
//   - 기존 C++는 자체 segments[]만 사용 → segCount==0이면 재생 불가
//   - 해결: profileId가 있으면 참조 프로필의 steps[]로 실행
//
//  [우선순위]
//   ① profileId > 0 && 프로필 유효 → 프로필 segments 실행
//   ② profileId가 없거나 유효하지 않음 → 자체 segments 실행
// ======================================================
bool CL_CT10_ControlManager::tickSchedule() {
    if (runSource != EN_CT10_RUN_SCHEDULE) return false;
    if (!g_A20_config_root.schedules) return false;
    if (curScheduleIndex < 0) return false;

    ST_A20_SchedulesRoot_t& v_cfg = *g_A20_config_root.schedules;
    if ((uint8_t)curScheduleIndex >= v_cfg.count) return false;

    ST_A20_ScheduleItem_t& v_schedule = v_cfg.items[(uint8_t)curScheduleIndex];
    if (!v_schedule.enabled) return false;

    // AutoOff 검사 (profileId fallback 이전)
    EN_CT10_reason_t v_reason = EN_CT10_REASON_NONE;
    if (checkAutoOff(&v_reason)) {
        onAutoOffTriggered(v_reason);
        return true;
    }

    // ① profileId 참조 우선
    if (v_schedule.profileId > 0 && g_A20_config_root.userProfiles) {
        auto& v_profiles = *g_A20_config_root.userProfiles;
        for (uint8_t v_i = 0; v_i < v_profiles.count; v_i++) {
            if (v_profiles.items[v_i].profileId == v_schedule.profileId) {
                ST_A20_UserProfileItem_t& v_prof = v_profiles.items[v_i];
                if (!v_prof.enabled || v_prof.segCount == 0) {
                    CL_D10_Logger::log(EN_L10_LOG_WARN,
                                       "[CT10] tickSchedule: profile invalid (id=%u)",
                                       (unsigned)v_schedule.profileId);
                    return false;
                }
                return tickSegmentSequence(
                    v_prof.repeatSegments,
                    v_prof.repeatCount,
                    v_prof.segments,
                    v_prof.segCount,
                    scheduleSegRt);   // schedule용 runtime 사용
            }
        }
        CL_D10_Logger::log(EN_L10_LOG_WARN,
                           "[CT10] tickSchedule: profile not found (id=%u)",
                           (unsigned)v_schedule.profileId);
    }

    // ② fallback: 자체 segments[]
    if (v_schedule.segCount == 0) return false;

    return tickSegmentSequence(
        v_schedule.repeatSegments,
        v_schedule.repeatCount,
        v_schedule.segments,
        v_schedule.segCount,
        scheduleSegRt);
}
```

```js
        _animate(canvas, points) {
            if (!canvas || !points || points.length === 0) return;

            const ctx = canvas.getContext('2d');
            const w   = canvas.width;
            const h   = canvas.height;

            let minV = Infinity, maxV = -Infinity;
            for (const p of points) {
                const v = Number(p.wind) || 0;
                if (v < minV) minV = v;
                if (v > maxV) maxV = v;
            }
            if (!Number.isFinite(minV) || !Number.isFinite(maxV) || maxV <= minV) {
                minV = 0; maxV = 1;
            }

            ctx.clearRect(0, 0, w, h);

            // 그리드 (5줄)
            ctx.strokeStyle = 'rgba(139,148,158,0.1)';
            ctx.lineWidth = 1;
            for (let i = 0; i <= 4; i++) {
                const y = (h / 4) * i;
                ctx.beginPath();
                ctx.moveTo(0, y);
                ctx.lineTo(w, y);
                ctx.stroke();
            }

            const pad = 10;
            const innerH = h - pad * 2;
            const n = points.length;

            // 파형
            ctx.strokeStyle = '#4a9eff';
            ctx.lineWidth = 2;
            ctx.lineJoin = 'round';
            ctx.beginPath();
            for (let i = 0; i < n; i++) {
                const x = (i / (n - 1)) * w;
                const y = pad + (h - pad * 2) - ((points[i].wind - minV) / (maxV - minV)) * innerH;
                if (i === 0) ctx.moveTo(x, y);
                else         ctx.lineTo(x, y);
            }
            ctx.stroke();

            // 돌풍 마커
            ctx.fillStyle = 'rgba(240,136,62,0.9)';
            for (let i = 0; i < n; i++) {
                if (!points[i].gust) continue;
                const x = (i / (n - 1)) * w;
                const y = pad + (h - pad * 2) - ((points[i].wind - minV) / (maxV - minV)) * innerH;
                ctx.beginPath();
                ctx.arc(x, y, 3, 0, Math.PI * 2);
                ctx.fill();
            }
        },
```

수정 3 — _drawOverlay + _ensurePreviewBuffer 신규 메서드 (클래스 하단)

위치: _exportCsv() 다음, 클래스 }; 앞

```js
        // ======================================================
        // [Phase5 5.5] 예상 파형 오버레이 (ADR-5 이후 신규)
        //
        //  [설계]
        //   - 차트 캔버스 위에 반투명 캔버스를 절대 위치로 삽입
        //   - 실측 데이터(_buffer)의 min/max 범위에 예상 파형을 매핑
        //   - 점선 + 돌풍 마커로 시각 구분
        //   - _showOverlay=false면 즉시 제거
        // ======================================================
        async _ensurePreviewBuffer() {
            if (this._previewBuffer) return;
            if (!SNW.Preview || typeof SNW.Preview.generate !== 'function') return;

            const sim = SNW.Store.get().sim || {};
            try {
                const wave = await SNW.Preview.generate(
                    sim.presetCode || 'OCEAN',
                    sim.styleCode || 'BALANCE',
                    null
                );
                if (wave && wave.points) this._previewBuffer = wave.points;
            } catch (e) {
                if (SNW.LogUtil) SNW.LogUtil.warn('[Chart] overlay preview fail', e);
            }
        }

        _drawOverlay() {
            // 기존 오버레이 제거
            const existing = this.el.querySelector('.snw-overlay-canvas');
            if (existing) existing.remove();

            // 조건 확인
            if (!this._showOverlay)         return;
            if (!this._previewBuffer)       return;
            if (this._activeTab !== 'live') return;   // 예상 파형 탭에서는 이미 실측이 예상이므로 오버레이 불필요
            if (!this._buffer || this._buffer.length < 2) return;

            const chartSlot = this.el.querySelector('[data-slot="chart"]');
            if (!chartSlot) return;

            if (getComputedStyle(chartSlot).position === 'static') {
                chartSlot.style.position = 'relative';
            }

            const ovl = document.createElement('canvas');
            ovl.className = 'snw-overlay-canvas';
            ovl.style.cssText = 'position:absolute;inset:0;width:100%;height:100%;pointer-events:none;z-index:1;';
            chartSlot.appendChild(ovl);

            requestAnimationFrame(() => {
                const rect = chartSlot.getBoundingClientRect();
                if (rect.width < 10 || rect.height < 10) return;
                if (!ovl.isConnected) return;

                ovl.width  = Math.floor(rect.width);
                ovl.height = Math.floor(rect.height);

                // 실측 + 예상 통합 min/max
                let mn = Infinity, mx = -Infinity;
                this._buffer.forEach(p => {
                    const v = Number(p.windSpeed);
                    if (!Number.isFinite(v)) return;
                    if (v < mn) mn = v;
                    if (v > mx) mx = v;
                });
                this._previewBuffer.forEach(p => {
                    const v = Number(p.wind);
                    if (!Number.isFinite(v)) return;
                    if (v < mn) mn = v;
                    if (v > mx) mx = v;
                });
                if (!Number.isFinite(mn) || !Number.isFinite(mx) || mx <= mn) return;

                const ctx = ovl.getContext('2d');
                const pad = 6;
                const innerH = ovl.height - pad * 2;
                const n = this._previewBuffer.length;
                if (n < 2) return;

                // 점선 오버레이
                ctx.strokeStyle = 'rgba(240,136,62,0.75)';
                ctx.lineWidth = 1.5;
                ctx.setLineDash([5, 4]);
                ctx.beginPath();
                for (let i = 0; i < n; i++) {
                    const x = (i / (n - 1)) * ovl.width;
                    const y = pad + ((mx - this._previewBuffer[i].wind) / (mx - mn)) * innerH;
                    if (i === 0) ctx.moveTo(x, y);
                    else         ctx.lineTo(x, y);
                }
                ctx.stroke();
                ctx.setLineDash([]);
            });
        }
```

```python
    cur = start

RE_T_CALL     = re.compile(r"(?:SNW\.)?I18n\.t\(\s*['\"]([a-zA-Z0-9_][a-zA-Z0-9._]*)['\"]\s*(?:,|\))")

def collect_used_keys(root: Path) -> dict[str, set[str]]:
    """파일별로 사용된 키를 수집. 반환: {file: set(keys)}"""
    used: dict[str, set[str]] = {}

    for sdir in SEARCH_DIRS:
        d = root / sdir
        if not d.is_dir():
            continue
        for fp in d.rglob("*.js"):

                keys.add(m.group(1))
            for m in RE_T_CALL2.finditer(content):
```

After — 전체 교체

```js
        // ======================================================
        // [Phase6 6.1 N-12] resolvePreset 재작성
        //  - 이전: target.params.* (C++/JSON 모두 factors 사용 → 항상 undefined)
        //  - 이후: target.factors.* + 스타일 factors 곱연산 (S20 미러)
        //  - REFRESH/WAVE는 실존하지 않는 스타일 → ACTIVE/RELAX로 매핑
        // ======================================================
        function resolvePreset(presetCode, styleCode) {
            const presets = (cache.windDict && cache.windDict.windDict && cache.windDict.windDict.presets) || [];
            const styles  = (cache.windDict && cache.windDict.windDict && cache.windDict.windDict.styles)  || [];

            const target = presets.find(p => p.code === presetCode);
            const pf     = target?.factors || {};

            let intensity   = pf.windIntensity   ?? 50;
            let variability = pf.windVariability ?? 40;
            let gustFreq    = pf.gustFrequency   ?? 30;

            // [S20 미러] 스타일 factor 적용 (곱연산)
            const sf = (styles.find(s => s.code === styleCode)?.factors) || {
                intensityFactor: 1, variabilityFactor: 1, gustFactor: 1,
            };
            intensity   = Math.max(0, Math.min(100, Math.round(intensity   * (sf.intensityFactor   ?? 1))));
            variability = Math.max(0, Math.min(100, Math.round(variability * (sf.variabilityFactor ?? 1))));
            gustFreq    = Math.max(0, Math.min(100, Math.round(gustFreq    * (sf.gustFactor        ?? 1))));

            return { intensity, variability, gustFreq };
        }
```

```js
        // ======================================================
        // [Phase6 6.3 N-13] sessionStart 재작성
        //  - state: 'RUNNING' → 'ACTIVE' (C++ 스펙)
        //  - physics 반영: intensity/variability/gustFreq
        //  - PLAYLIST: profile 조회 → profileName/steps/stepDuration 설정
        //  - PRESET: resolvePreset → sim 반영
        //  - FIXED: percent → intensity 반영
        //  - sim.active = true (stopSession 이후 복귀 대응)
        // ======================================================
        A.sessionStart = async (type, payload, trigger = 'MANUAL') => {
            const isPlaylist = (type === 'PLAYLIST' || type === 1);
            const isPreset   = (type === 'PRESET'   || type === 2);
            const isFixed    = (type === 'FIXED'    || type === 3);
            const now = Date.now();

            // ─────────── ① physics 반영 ───────────
            runtime.sim.active     = true;
            runtime.control.active = true;

            if (isPreset) {
                const pc = (payload && payload.presetCode) || 'OCEAN';
                const sc = (payload && payload.styleCode)  || 'BALANCE';
                const r  = resolvePreset(pc, sc);
                runtime.sim.presetCode  = pc;
                runtime.sim.styleCode   = sc;
                runtime.sim.intensity   = r.intensity;
                runtime.sim.variability = r.variability;
                runtime.sim.gustFreq    = r.gustFreq;
            } else if (isFixed) {
                const pct = (payload && payload.percent) || 50;
                runtime.sim.intensity   = Math.max(0, Math.min(100, pct));
                runtime.sim.variability = 0;
                runtime.sim.gustFreq    = 0;
            }

            // ─────────── ② 세션 컨텍스트 ───────────
            let profileId   = 0;
            let profileNo   = 0;
            let profileName = '';
            let totalSteps  = 1;
            let stepDurationMs = 20 * 60000;   // 기본 20분

            if (isPlaylist && payload && payload.profileId) {
                profileId = payload.profileId;
                const profiles = (cache.userProfiles?.userProfiles?.profiles) || [];
                const profile  = profiles.find(p => p.profileId === profileId);
                if (profile) {
                    profileNo   = profile.profileNo || profileId;
                    profileName = profile.name || '';
                    const steps = profile.steps || profile.segments || [];
                    totalSteps  = steps.length || 1;
                    if (steps.length > 0) {
                        const s0 = steps[0];
                        const durMin = s0.durationMin
                            ?? ((s0.onMinutes || 20) + (s0.offMinutes || 0));
                        stepDurationMs = (durMin || 20) * 60000;
                    }
                }
            }

            runtime.control.session = {
                active: true,
                type:     isPlaylist ? 1 : (isPreset ? 2 : 3),
                typeName: isPlaylist ? 'PLAYLIST' : (isPreset ? 'PRESET' : 'FIXED'),
                trigger:  trigger === 'SCHEDULE' ? 2 : 1,
                state:    'ACTIVE',                                    // [6.3 N-13]
                profileId, profileNo, profileName,
                presetCode: (payload && payload.presetCode) || 'OCEAN',
                styleCode:  (payload && payload.styleCode)  || 'BALANCE',
                percent:    (payload && payload.percent)    || 50,
                currentStep:   1,
                totalSteps,
                stepRemainSec: Math.floor(stepDurationMs / 1000),
                currentStepStartMs:   now,                             // [6.3 신규]
                currentStepDurationMs: stepDurationMs,                 // [6.3 신규]
                loopCount:    0,
                pausedAccumMs: 0,
            };

            if (SNW.Store) {
                SNW.Store.set({
                    control: runtime.control,
                    sim:     runtime.sim,
                    session: runtime.control.session,
                    lastUpdateMs: now,
                });
            }

            return { ok: true, result: 'session_started' };
        };
```

```js
                // step 시계 시프트 (일시정지 시간만큼)
                const pausedMs = Date.now() - (sess.pausedAtMs || Date.now());
                if (sess.currentStepStartMs) {
                    sess.currentStepStartMs += pausedMs;
                }
                sess.pausedAccumMs = (sess.pausedAccumMs || 0) + pausedMs;
                sess.state         = 'ACTIVE';
                sess.pausedAtMs    = 0;
                runtime.sim.active = true;       // 물리 재개
            }
            if (SNW.Store) SNW.Store.set({
                control: runtime.control, sim: runtime.sim, session: sess,
                lastUpdateMs: Date.now(),
            });
            return { ok: true, result: 'session_resumed' };
        };

        A.stopSession = async () => {
            const sess = runtime.control.session;
            if (sess) {
                sess.active = false;
                sess.state  = 'COMPLETED';
            }
            runtime.sim.active     = false;
            runtime.control.active = false;
            runtime.control.state     = 'IDLE';
            runtime.control.stateCode = 0;
            runtime.control.reason     = 'NONE';
            runtime.control.reasonCode = 0;

            if (SNW.Store) SNW.Store.set({
                control: runtime.control, sim: runtime.sim, session: sess,
                fsmState: 0, fsmReason: 0,
                lastUpdateMs: Date.now(),
            });
            return { ok: true, result: 'session_stopped' };
        };
```

```python
def find_repo_root(start: Path) -> Path:
    cur = start
    while cur != cur.parent:
        if (cur / "platformio.ini").is_file():
            return cur
        cur = cur.parent
    return Path.cwd()


RE_T_CALL    = re.compile(r"(?:SNW\.)?I18n\.t\(\s*['\"]([a-zA-Z0-9_][a-zA-Z0-9._]*)['\"]")
RE_DATA_I18N = re.compile(r'data-i18n="([a-zA-Z0-9_][a-zA-Z0-9._]*)"')
RE_DICT_KEY  = re.compile(r"^\s*'([a-zA-Z0-9_][a-zA-Z0-9._]*)':", re.MULTILINE)


def collect_used_keys(root: Path) -> dict[str, set[str]]:
    used: dict[str, set[str]] = {}
    src = root / JS_SRC_DIR
    if not src.is_dir():
        return used

    for fp in src.rglob("*.js"):
        # i18n 자기 자신, 번들 산출물, 오프라인 어댑터 제외
        if fp.name == "020.i18n.js":            continue
        if fp.name.startswith("P010_bundle"):   continue
        if fp.name.startswith("998.offline"):   continue

        try:
            content = fp.read_text(encoding="utf-8", errors="replace")
        except Exception:
            continue

        keys = set(RE_T_CALL.findall(content))
        if keys:
            used[str(fp.relative_to(root)).replace("\\", "/")] = keys

    html = root / HTML_FILE
    if html.is_file():
        try:
            content = html.read_text(encoding="utf-8", errors="replace")
            keys = set(RE_DATA_I18N.findall(content))
            if keys:
                used[str(html.relative_to(root)).replace("\\", "/")] = keys
        except Exception:
            pass

    return used


def collect_defined_keys(root: Path) -> set[str]:
    fp = root / I18N_FILE
    if not fp.is_file():
        print(f"[ERROR] i18n 파일 없음: {fp}", file=sys.stderr)
        sys.exit(EXIT_IO)
    content = fp.read_text(encoding="utf-8", errors="replace")
    return set(RE_DICT_KEY.findall(content))

def main() -> int:
    parser = argparse.ArgumentParser(description="i18n DICT 완전성 검증")
    parser.add_argument("--strict",  action="store_true", help="미정의 키 발견 시 exit 1")
    parser.add_argument("--verbose", action="store_true", help="모든 사용 키 개수 출력")
    args = parser.parse_args()

    root     = find_repo_root(Path(__file__).resolve().parent)
    used_map = collect_used_keys(root)
    defined  = collect_defined_keys(root)

    unique_used = len(set().union(*used_map.values())) if used_map else 0

    print("=" * 64)
    print(f" i18n DICT 완전성 검증")
    print(f" 검사 파일: {len(used_map)}개 / 고유 키: {unique_used}개 / 정의: {len(defined)}개")
    print("=" * 64)

    missing: dict[str, list[str]] = {}
    for rel, keys in used_map.items():
        m = sorted(k for k in keys if k not in defined)
        if m:
            missing[rel] = m

    total_missing = sum(len(v) for v in missing.values())
    if total_missing == 0:
        print(f"\n[OK] 모든 {unique_used}개 키가 정의되어 있습니다.")
        return EXIT_OK

    print(f"\n[FAIL] {total_missing}개 미정의 키 ({len(missing)}개 파일):\n")
    for rel in sorted(missing.keys()):
        print(f"  {rel}")
        for k in missing[rel]:
            print(f"    - {k}")
        print()

    if args.strict:
        print("[STRICT] exit 1 (CI gate)")
        return EXIT_MISSING

    print("[INFO] --strict 옵션으로 CI gate 활성화 가능")
    return EXIT_OK


if __name__ == "__main__":
    try:
        sys.exit(main())
    except KeyboardInterrupt:
        print("\n[INT] 사용자 중단", file=sys.stderr)
```

```python
def _run_i18n_check(project_root: Path) -> int:
    script = project_root / "src" / "v0203" / "tools_0203" / "check_i18n.py"
    if not script.is_file():
        _log("[WARN] check_i18n.py 없음 → i18n gate 스킵", tag="i18n")
        return 0

    _log("[i18n] check_i18n.py --strict 실행", tag="i18n")
    try:
        rc = subprocess.run(
            [sys.executable, str(script), "--strict"],
            cwd=str(project_root),

        _log(f"[ERROR] i18n 검증 실행 실패: {e}", tag="i18n")

        _log(f"[FAIL] i18n 검증 실패 (exit={rc}) → staging 중단", tag="i18n")
    else:
        _log("[OK] i18n 검증 통과", tag="i18n")
```
