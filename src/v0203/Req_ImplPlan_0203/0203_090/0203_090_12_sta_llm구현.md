    #include <stdint.h>

    typedef enum : uint8_t {
        EN_VOICE_IDLE         = 0,   // 웨이크워드 대기
        EN_VOICE_WAKE_DETECTED,      // 감지 → LED+비프
        EN_VOICE_RECORDING,          // 5초 녹음
        EN_VOICE_UPLOADING,          // WAV+Base64+Gemini
        EN_VOICE_PROCESSING,         // FunctionCall 실행
        EN_VOICE_TTS_FETCHING,       // TTS 오디오 수신
        EN_VOICE_PLAYING,            // 스피커 재생
        EN_VOICE_ERROR,              // 오류
    } EN_VOICE_state_t;

    typedef enum : uint8_t {
        EN_VOICE_EVT_WAKE = 0,       // 웨이크워드 감지
        EN_VOICE_EVT_RECORD_DONE,    // 녹음 완료
        EN_VOICE_EVT_UPLOAD_DONE,    // Gemini 응답
        EN_VOICE_EVT_PROCESS_DONE,   // Function 실행 완료
        EN_VOICE_EVT_TTS_DONE,       // TTS 수신
        EN_VOICE_EVT_PLAY_DONE,      // 재생 완료
        EN_VOICE_EVT_ERROR,          // 오류
    } EN_VOICE_event_t;

    // 공개 API (Main에서만 노출)
    void voice_init();
    void voice_loop();                       // loop()에서 호출
    void voice_fsm_post_event(EN_VOICE_event_t evt);

    static EN_VOICE_state_t s_state = EN_VOICE_IDLE;
    static uint32_t s_stateEnteredMs = 0;

    // 이전 상태 전이
    static void _enterState(EN_VOICE_state_t next) {
        s_state = next;
        s_stateEnteredMs = millis();
        VOICE_LOG("[FSM] enter state=%u", next);
        switch (next) {
            case EN_VOICE_IDLE:          voice_led_idle_blue();   break;
            case EN_VOICE_WAKE_DETECTED: voice_led_green();
                                          voice_beep(1000, 100);   break;
            case EN_VOICE_RECORDING:     voice_led_green_blink(); break;
            case EN_VOICE_UPLOADING:     voice_led_yellow();      break;
            case EN_VOICE_PROCESSING:    voice_led_yellow_blink();break;
            case EN_VOICE_TTS_FETCHING:  voice_led_yellow_blink();break;
            case EN_VOICE_PLAYING:       voice_led_red();         break;
            case EN_VOICE_ERROR:         voice_led_purple_x3();
                                          voice_beep(500, 200);    break;
            default: break;
        }
    }

    // 초기화
    void voice_init() {
        voice_i2s_mic_init();
        voice_i2s_amp_init();
        voice_led_init();
        if (!voice_wakeword_init()) {
            VOICE_LOG("[ERR] wakeword init failed");
            _enterState(EN_VOICE_ERROR);
            return;
        }
        voice_wakeword_start_task();   // Core 0, prio 5
        _enterState(EN_VOICE_IDLE);
    }

    // 이벤트 처리
    void voice_fsm_post_event(EN_VOICE_event_t evt) {
        VOICE_LOG("[FSM] evt=%u from state=%u", evt, s_state);
        switch (s_state) {
            case EN_VOICE_IDLE:
                if (evt == EN_VOICE_EVT_WAKE)
                    _enterState(EN_VOICE_WAKE_DETECTED);
                break;
            case EN_VOICE_WAKE_DETECTED:
                // 300ms LED 지속 후 녹음
                if (millis() - s_stateEnteredMs > 300)
                    _enterState(EN_VOICE_RECORDING);
                break;
            case EN_VOICE_RECORDING:
                if (evt == EN_VOICE_EVT_RECORD_DONE)
                    _enterState(EN_VOICE_UPLOADING);
                break;
            case EN_VOICE_UPLOADING:
                if (evt == EN_VOICE_EVT_UPLOAD_DONE)
                    _enterState(EN_VOICE_PROCESSING);
                if (evt == EN_VOICE_EVT_ERROR)
                    _enterState(EN_VOICE_ERROR);
                break;
            case EN_VOICE_PROCESSING:
                if (evt == EN_VOICE_EVT_PROCESS_DONE)
                    _enterState(EN_VOICE_TTS_FETCHING);
                if (evt == EN_VOICE_EVT_ERROR)
                    _enterState(EN_VOICE_ERROR);
                break;
            case EN_VOICE_TTS_FETCHING:
                if (evt == EN_VOICE_EVT_TTS_DONE)
                    _enterState(EN_VOICE_PLAYING);
                break;
            case EN_VOICE_PLAYING:
                if (evt == EN_VOICE_EVT_PLAY_DONE)
                    _enterState(EN_VOICE_IDLE);
                break;
            case EN_VOICE_ERROR:
                // LED 3회 점멸 후 2초 뒤 IDLE
                if (millis() - s_stateEnteredMs > 2000)
                    _enterState(EN_VOICE_IDLE);
                break;
            default: break;
        }
    }

    // 메인 루프 (Arduino loop에서 호출)
    void voice_loop() {
        // WAKE_DETECTED 지연 처리
        if (s_state == EN_VOICE_WAKE_DETECTED &&
            millis() - s_stateEnteredMs > 300) {
            voice_fsm_post_event(EN_VOICE_EVT_WAKE);
        }
        // ERROR 자동 복구
        if (s_state == EN_VOICE_ERROR &&
            millis() - s_stateEnteredMs > 2000) {
            _enterState(EN_VOICE_IDLE);
        }
        // 상태별 tick
        switch (s_state) {
            case EN_VOICE_RECORDING: {
                // 5초 녹음 (blocking, RTOS 태스크)
                uint8_t* pcm = voice_record_5s();
                if (pcm) {
                    voice_set_pcm_buffer(pcm);
                    voice_fsm_post_event(EN_VOICE_EVT_RECORD_DONE);
                } else {
                    voice_fsm_post_event(EN_VOICE_EVT_ERROR);
                }
                break;
            }
            case EN_VOICE_UPLOADING: {
                uint8_t* pcm = voice_get_pcm_buffer();
                uint8_t* wav = voice_wav_encode(pcm, VOICE_PCM_BYTES);
                if (!wav) { voice_fsm_post_event(EN_VOICE_EVT_ERROR); break; }
                String resp = voice_send_to_gemini(wav, 44 + VOICE_PCM_BYTES);
                free(wav);
                if (resp.length() == 0) { voice_fsm_post_event(EN_VOICE_EVT_ERROR); break; }
                voice_set_gemini_response(resp);
                voice_fsm_post_event(EN_VOICE_EVT_UPLOAD_DONE);
                break;
            }
            case EN_VOICE_PROCESSING: {
                // FunctionCall 실행 (Stage 8.1 재사용)
                String resp = voice_get_gemini_response();
                bool ok = voice_handle_function_call(resp);
                if (ok) voice_fsm_post_event(EN_VOICE_EVT_PROCESS_DONE);
                else    voice_fsm_post_event(EN_VOICE_EVT_ERROR);
                break;
            }
            case EN_VOICE_TTS_FETCHING: {
                const char* text = voice_get_tts_text();
                uint8_t* pcm = nullptr; size_t len = 0;
                if (voice_tts_fetch(text, &pcm, &len)) {
                    voice_set_tts_buffer(pcm, len);
                    voice_fsm_post_event(EN_VOICE_EVT_TTS_DONE);
                } else {
                    voice_fsm_post_event(EN_VOICE_EVT_ERROR);
                }
                break;
            }
            case EN_VOICE_PLAYING: {
                // 반이중 재생
                voice_i2s_mute_mic();
                uint8_t* pcm = nullptr; size_t len = 0;
                voice_get_tts_buffer(&pcm, &len);
                voice_play_pcm(pcm, len);
                voice_i2s_unmute_mic();
                voice_free_tts_buffer();
                voice_fsm_post_event(EN_VOICE_EVT_PLAY_DONE);
                break;
            }
            default: break;
        }
    }

    #pragma once
    #include <stdint.h>

    #define VOICE_PCM_BYTES  (16000 * 2 * 5)   // 5초 × 16kHz × 16bit = 160000 B

    void voice_i2s_mic_init();
    void voice_i2s_amp_init();
    uint8_t* voice_record_5s();               // PSRAM 할당, 실패 시 nullptr

    // .cpp
    #include <driver/i2s.h>
    #include <esp_heap_caps.h>
    #include "P_Voice_I2S_0203.h"

    void voice_i2s_mic_init() {
        i2s_config_t cfg = {
            .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
            .sample_rate = 16000,
            .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
            .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
            .communication_format = I2S_COMM_FORMAT_STAND_I2S,
            .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
            .dma_buf_count = 8,
            .dma_buf_len = 1024,
            .use_apll = false,
        };
        i2s_driver_install(I2S_NUM_0, &cfg, 0, NULL);
        i2s_pin_config_t pins = {
            .bck_io_num = 4, .ws_io_num = 5,
            .data_out_num = I2S_PIN_NO_CHANGE, .data_in_num = 6,
        };
        i2s_set_pin(I2S_NUM_0, &pins);
    }

    void voice_i2s_amp_init() {
        i2s_config_t cfg = {
            .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_TX),
            .sample_rate = 24000,
            .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
            .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
            .communication_format = I2S_COMM_FORMAT_STAND_I2S,
            .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
            .dma_buf_count = 8, .dma_buf_len = 1024, .use_apll = false,
        };
        i2s_driver_install(I2S_NUM_1, &cfg, 0, NULL);
        i2s_pin_config_t pins = {
            .bck_io_num = 15, .ws_io_num = 16,
            .data_out_num = 17, .data_in_num = I2S_PIN_NO_CHANGE,
        };
        i2s_set_pin(I2S_NUM_1, &pins);
    }

    // ★ PSRAM 160KB 할당
    uint8_t* voice_record_5s() {
        uint8_t* buf = (uint8_t*)heap_caps_malloc(VOICE_PCM_BYTES, MALLOC_CAP_SPIRAM);
        if (!buf) return nullptr;
        size_t total = 0, read = 0;
        while (total < VOICE_PCM_BYTES) {
            i2s_read(I2S_NUM_0, buf + total, VOICE_PCM_BYTES - total, &read, 1000);
            total += read;
        }
        return buf;
    }

    void voice_i2s_mute_mic() {
        i2s_stop(I2S_NUM_0);   // 마이크 DMA 중지 (AEC 대체)
    }
    void voice_i2s_unmute_mic() {
        i2s_start(I2S_NUM_0);
    }

    static srmodel_list_t* models = nullptr;
    static esp_sr_iface_t* sr_handle = nullptr;
    static esp_sr_data_t*  sr_data = nullptr;

    bool voice_wakeword_init() {
        models = esp_srmodel_init("model");
        if (!models) return false;
        sr_handle = esp_sr_handle_from_name("wn9_hilexin");   // 프리셋
        if (!sr_handle) return false;
        sr_data = sr_handle->create(models, nullptr);
        return (sr_data != nullptr);
    }

    // 감지 태스크 (Core 0, prio 5)
    static void _wakeword_task(void* arg) {
        int16_t buf[512];
        size_t bytesRead = 0;
        while (1) {
            // 감지 대기 모드일 때만 처리
            i2s_read(I2S_NUM_0, buf, sizeof(buf), &bytesRead, portMAX_DELAY);
            int wakeIdx = sr_handle->detect(sr_data, buf);
            if (wakeIdx >= 0) {
                VOICE_LOG("[WAKE] detected (idx=%d)", wakeIdx);
                voice_fsm_post_event(EN_VOICE_EVT_WAKE);
            }
        }
    }

    void voice_wakeword_start_task() {
        xTaskCreatePinnedToCore(
            _wakeword_task, "wakeword",
            4096, nullptr, 5, nullptr, 0);   // Core 0
    }

### 2.5 P_Voice_WAV_0203.h/.cpp — WAV 인코딩

    // .h
    #pragma once
    #include <stdint.h>
    uint8_t* voice_wav_encode(const uint8_t* pcm, size_t pcmLen);

    // .cpp
    #include <esp_heap_caps.h>
    #include <string.h>
    #include "P_Voice_WAV_0203.h"

    typedef struct __attribute__((packed)) {
        char     riff[4];       // "RIFF"
        uint32_t fileSize;      // total - 8
        char     wave[4];       // "WAVE"
        char     fmt[4];        // "fmt "
        uint32_t fmtSize;       // 16
        uint16_t audioFormat;   // 1 (PCM)
        uint16_t numChannels;   // 1 (mono)
        uint32_t sampleRate;    // 16000
        uint32_t byteRate;      // 32000
        uint16_t blockAlign;    // 2
        uint16_t bitsPerSample; // 16
        char     data[4];       // "data"
        uint32_t dataSize;      // pcmLen
    } WAVHeader_t;

    uint8_t* voice_wav_encode(const uint8_t* pcm, size_t pcmLen) {
        size_t total = 44 + pcmLen;
        uint8_t* wav = (uint8_t*)heap_caps_malloc(total, MALLOC_CAP_SPIRAM);
        if (!wav) return nullptr;
        WAVHeader_t* h = (WAVHeader_t*)wav;
        memcpy(h->riff, "RIFF", 4); h->fileSize = total - 8;
        memcpy(h->wave, "WAVE", 4);
        memcpy(h->fmt,  "fmt ", 4); h->fmtSize = 16;
        h->audioFormat = 1; h->numChannels = 1;
        h->sampleRate = 16000; h->byteRate = 32000;
        h->blockAlign = 2; h->bitsPerSample = 16;
        memcpy(h->data, "data", 4); h->dataSize = pcmLen;
        memcpy(wav + 44, pcm, pcmLen);
        return wav;
    }

    // Base64 (PSRAM 버퍼 사용)
    static bool _base64_encode(const uint8_t* src, size_t srcLen, char* dst, size_t dstSize) {
        static const char TB[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
        size_t i = 0, o = 0;
        while (i < srcLen && o + 4 < dstSize) {
            uint32_t b = (i < srcLen ? src[i++] : 0) << 16;
            if (i < srcLen) b |= src[i++] << 8;
            if (i < srcLen) b |= src[i++];
            dst[o++] = TB[(b >> 18) & 63];
            dst[o++] = TB[(b >> 12) & 63];
            dst[o++] = TB[(b >>  6) & 63];
            dst[o++] = TB[b & 63];
        }
        dst[o] = 0;
        return true;
    }

    String voice_send_to_gemini(const uint8_t* wav, size_t wavLen) {
        // Base64 (PSRAM 213KB)
        size_t b64Len = 4 * ((wavLen + 2) / 3) + 1;
        char* b64 = (char*)heap_caps_malloc(b64Len, MALLOC_CAP_SPIRAM);
        if (!b64) return "";
        _base64_encode(wav, wavLen, b64, b64Len);

        // 요청 조립
        JsonDocument req;
        req["system_instruction"]["parts"][0]["text"] =
            CT10_buildAIPromptStrict("ko", true);
        req["contents"][0]["role"] = "user";
        req["contents"][0]["parts"][0]["text"] =
            "Transcribe and interpret this audio as a fan control command. "
            "Call the appropriate tool. Reply in Korean.";
        req["contents"][0]["parts"][1]["inline_data"]["mime_type"] = "audio/wav";
        req["contents"][0]["parts"][1]["inline_data"]["data"]      = b64;

        // Function Declarations 12개 재사용
        JsonArray tools = req["tools"].to<JsonArray>();
        JsonArray decls = tools.add<JsonObject>()["function_declarations"].to<JsonArray>();
        CT10_appendAIFunctionDeclarations(decls);

        String body;
        serializeJson(req, body);
        free(b64);

        // Gemini POST
        String url = "https://generativelanguage.googleapis.com/v1beta/models/"
                     "gemini-2.0-flash-exp:generateContent";
        const char* apiKey = g_A20_config_root.system->security.geminiApiKey;

        WiFiClientSecure tls; tls.setInsecure();
        HTTPClient http;
        http.begin(tls, url);
        http.addHeader("Content-Type", "application/json");
        http.addHeader("x-goog-api-key", apiKey);
        http.setTimeout(15000);
        int code = http.POST(body);
        String resp = (code == 200) ? http.getString() : "";
        http.end();
        return resp;
    }

    // Function Call 파싱 + 실행 (Stage 8.1의 CT10_executeAIAction 재사용)
    bool voice_handle_function_call(const String& resp) {
        JsonDocument doc;
        if (deserializeJson(doc, resp) != DeserializationError::Ok) return false;
        JsonArray parts = doc["candidates"][0]["content"]["parts"].as<JsonArray>();
        if (parts.isNull()) return false;

        for (JsonObject part : parts) {
            JsonObject fc = part["functionCall"].as<JsonObject>();
            if (fc.isNull()) {
                // 텍스트 응답 수집
                const char* t = part["text"] | "";
                if (t[0]) s_ttsText = t;
                continue;
            }
            const char* fnName = fc["name"] | "";
            JsonObjectConst args = fc["args"].as<JsonObjectConst>();
            String result = CT10_executeAIAction(fnName, args);
            VOICE_LOG("[AI] fn=%s result=%s", fnName, result.c_str());

            // _speechHint 폴백 (2-turn 없이 로컬 생성)
            String hint = CT10_buildSpeechHint(fnName, args, result == "ok");
            s_ttsText = hint;
            return true;
        }
        return false;
    }

    String voice_get_gemini_response() { return s_geminiResp; }
    const char* voice_get_tts_text() { return s_ttsText.c_str(); }

    bool voice_tts_fetch(const char* text, uint8_t** outPcm, size_t* outLen) {
        JsonDocument req;
        req["input"]["text"] = text;
        req["voice"]["languageCode"] = "ko-KR";
        req["voice"]["name"]         = "ko-KR-Neural2-A";
        req["audioConfig"]["audioEncoding"]   = "LINEAR16";
        req["audioConfig"]["sampleRateHertz"] = 24000;

        String body; serializeJson(req, body);
        String url = "https://texttospeech.googleapis.com/v1/text:synthesize";
        const char* key = g_A20_config_root.system->security.geminiApiKey;

        WiFiClientSecure tls; tls.setInsecure();
        HTTPClient http; http.begin(tls, url);
        http.addHeader("Content-Type", "application/json");
        http.addHeader("x-goog-api-key", key);
        http.setTimeout(10000);
        int code = http.POST(body);
        String resp = (code == 200) ? http.getString() : "";
        http.end();
        if (resp.length() == 0) return false;

        // 응답 파싱: { audioContent: "base64 PCM" }
        JsonDocument rd;
        if (deserializeJson(rd, resp) != DeserializationError::Ok) return false;
        const char* b64 = rd["audioContent"] | "";
        size_t b64Len = strlen(b64);
        if (b64Len == 0) return false;

        size_t pcmLen = 3 * b64Len / 4;
        uint8_t* pcm = (uint8_t*)heap_caps_malloc(pcmLen, MALLOC_CAP_SPIRAM);
        if (!pcm) return false;
        // base64 디코딩 (간이)
        size_t i = 0, o = 0;
        auto dec = [](char c)->uint8_t {
            if (c>='A'&&c<='Z') return c-'A';
            if (c>='a'&&c<='z') return c-'a'+26;
            if (c>='0'&&c<='9') return c-'0'+52;
            if (c=='+') return 62; if (c=='/') return 63; return 0;
        };
        while (i < b64Len && o < pcmLen) {
            uint32_t v = dec(b64[i++]) << 18;
            if (i < b64Len) v |= dec(b64[i++]) << 12;
            if (i < b64Len) v |= dec(b64[i++]) <<  6;
            if (i < b64Len) v |= dec(b64[i++]);
            pcm[o++] = (v >> 16) & 0xFF;
            if (o < pcmLen) pcm[o++] = (v >> 8) & 0xFF;
            if (o < pcmLen) pcm[o++] = v & 0xFF;
        }
        *outPcm = pcm;
        *outLen = o;
        return true;
    }

    void voice_play_pcm(const uint8_t* pcm, size_t len) {
        size_t written = 0;
        i2s_write(I2S_NUM_1, pcm, len, &written, portMAX_DELAY);
    }

    void voice_set_tts_buffer(uint8_t* pcm, size_t len) { s_ttsBuf = pcm; s_ttsLen = len; }
    void voice_get_tts_buffer(uint8_t** pcm, size_t* len) { *pcm = s_ttsBuf; *len = s_ttsLen; }
    void voice_free_tts_buffer() {
        if (s_ttsBuf) { heap_caps_free(s_ttsBuf); s_ttsBuf = nullptr; s_ttsLen = 0; }
    }

    void voice_beep(int freqHz, int durationMs) {
        int samples = 24000 * durationMs / 1000;
        int16_t* buf = (int16_t*)malloc(samples * 2);
        if (!buf) return;
        for (int i = 0; i < samples; i++) {
            buf[i] = (int16_t)(8000 * sin(2 * M_PI * freqHz * i / 24000.0));
        }
