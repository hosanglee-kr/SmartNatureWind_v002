#!/usr/bin/env python3
"""
src/v0200/contract_doc_0200 커밋 스크립트 (Python 3.7+ 크로스 플랫폼)

기능:
  1. git 저장소 루트 자동 감지 (git rev-parse --show-toplevel)
  2. src/v0200/contract_doc_0200/ 필수 문서 존재 검증
     (README_0200/SPEC_0200/CONTRACT_0200/STATE_0200/FLOW_0200/BUDGET_0200/TEST_PLAN_0200)
  3. staging 후 단일 커밋
  4. 커밋 후 요약 출력 (author/date, 다음 push 명령)

사용:
  python src/v0200/tools_0200/commit_contracts_0200.py
  python src/v0200/tools_0200/commit_contracts_0200.py -m "src/v0200/contract_doc_0200: 계약 문서 갱신 (F-01~F-06, I-1~I-4 반영)"
  python src/v0200/tools_0200/commit_contracts_0200.py --dry-run

  # Windows (py 런처)
  py -3 src/v0200/tools_0200/commit_contracts_0200.py

종료 코드:
  0 성공 (또는 커밋할 변경 없음)
  1 필수 파일 누락
  2 git 오류 (저장소 아님 / git 없음 / commit 실패)
  130 사용자 중단 (Ctrl+C)

지원 OS: Windows / macOS / Linux
인코딩 : UTF-8 강제 (Windows cp949 회피)
"""

from __future__ import annotations

import argparse
import os
import subprocess
import sys
from pathlib import Path
from typing import List, Sequence, Tuple

# ============================================================
# 상수
# ============================================================
REQUIRED_FILES: Tuple[str, ...] = (
    "src/v0200/contract_doc_0200/README_0200.md",
    "src/v0200/contract_doc_0200/SPEC_0200.md",
    "src/v0200/contract_doc_0200/CONTRACT_0200.md",
    "src/v0200/contract_doc_0200/STATE_0200.md",
    "src/v0200/contract_doc_0200/FLOW_0200.md",
    "src/v0200/contract_doc_0200/BUDGET_0200.md",
    "src/v0200/contract_doc_0200/TEST_PLAN_0200.md",
)

DEFAULT_MSG = "src/v0200/contract_doc_0200: 계약 문서 갱신 (F-01~F-06, I-1~I-4 반영)"
CONTRACTS_REL = "src/v0200/contract_doc_0200"

EXIT_OK      = 0
EXIT_MISSING = 1
EXIT_GIT     = 2
EXIT_INT     = 130


# ============================================================
# 헬퍼
# ============================================================
def _run(cmd: Sequence[str], cwd: Path, check: bool = True) -> subprocess.CompletedProcess:
    """서브프로세스 실행 (UTF-8 강제)."""
    env = os.environ.copy()
    env.setdefault("PYTHONIOENCODING", "utf-8")
    return subprocess.run(
        cmd,
        cwd=str(cwd),
        capture_output=True,
        text=True,
        encoding="utf-8",
        errors="replace",
        check=check,
        env=env,
    )


def _err(msg: str) -> None:
    print(f"[ERROR] {msg}", file=sys.stderr)


def _info(msg: str = "") -> None:
    print(msg)


def find_repo_root(start: Path) -> Path:
    """`git rev-parse --show-toplevel`로 저장소 루트 감지."""
    try:
        res = _run(["git", "rev-parse", "--show-toplevel"], cwd=start, check=True)
    except FileNotFoundError:
        _err("git 실행 파일을 찾을 수 없습니다. Git 설치 및 PATH 확인.")
        sys.exit(EXIT_GIT)
    except subprocess.CalledProcessError as e:
        _err("git 저장소 루트를 찾을 수 없습니다 (not a git repository).")
        if e.stderr:
            print(e.stderr.strip(), file=sys.stderr)
        sys.exit(EXIT_GIT)

    root = res.stdout.strip()
    if not root:
        _err("git 저장소 루트를 찾을 수 없습니다 (empty output).")
        sys.exit(EXIT_GIT)
    return Path(root)


def verify_required_files(repo_root: Path) -> None:
    """필수 계약 문서 존재 확인."""
    missing: List[str] = [rel for rel in REQUIRED_FILES if not (repo_root / rel).is_file()]
    if missing:
        _err("필수 문서 누락:")
        for m in missing:
            print(f"  - {m}", file=sys.stderr)
        sys.exit(EXIT_MISSING)


# ============================================================
# main
# ============================================================
def main() -> int:
    parser = argparse.ArgumentParser(
        description="src/v0200/contract_doc_0200 커밋 스크립트 (Python 크로스 플랫폼)",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog=(
            "예시:\n"
            "  python src/v0200/tools_0200/commit_contracts_0200.py\n"
            "  python src/v0200/tools_0200/commit_contracts_0200.py -m 'docs(contracts): F-05 반영'\n"
            "  python src/v0200/tools_0200/commit_contracts_0200.py --dry-run\n"
        ),
    )
    parser.add_argument(
        "-m", "--message",
        default=DEFAULT_MSG,
        help="커밋 메시지 (기본: 계약 문서 갱신 기본 메시지)",
    )
    parser.add_argument(
        "-n", "--dry-run",
        action="store_true",
        help="실제 커밋 없이 검증만 수행",
    )
    args = parser.parse_args()

    # ---- 저장소 루트 감지 -----------------------------------
    script_dir = Path(__file__).resolve().parent
    repo_root  = find_repo_root(script_dir)
    verify_required_files(repo_root)

    # ---- 헤더 요약 -----------------------------------------
    _info("=" * 60)
    _info(" 커밋 대상: docs/contracts/")
    _info(f" 저장소   : {repo_root}")
    _info(f" 메시지   : {args.message}")
    if args.dry_run:
        _info(" 모드     : DRY-RUN (실제 커밋 없음)")
    _info("=" * 60)

    # ---- [1/4] 변경 사항 확인 ------------------------------
    _info()
    _info("[1/4] 변경 사항 확인")
    status = _run(["git", "status", "--short", CONTRACTS_REL], cwd=repo_root, check=True)
    print(status.stdout, end="")

    if args.dry_run:
        _info()
        _info("[DRY-RUN] git add / commit 생략")
        return EXIT_OK

    # ---- [2/4] staging -------------------------------------
    _info()
    _info("[2/4] staging")
    try:
        _run(["git", "add", CONTRACTS_REL], cwd=repo_root, check=True)
    except subprocess.CalledProcessError as e:
        _err("git add 실패")
        if e.stderr:
            print(e.stderr.strip(), file=sys.stderr)
        return EXIT_GIT

    # ---- [3/4] staged 파일 목록 ----------------------------
    staged = _run(
        ["git", "diff", "--cached", "--name-only", "--", CONTRACTS_REL],
        cwd=repo_root, check=True,
    ).stdout.strip()

    if not staged:
        _info()
        _info("[INFO] staging된 변경 사항이 없습니다. 커밋을 생략합니다.")
        return EXIT_OK

    _info()
    _info("[3/4] staged 파일:")
    for line in staged.splitlines():
        _info(f"  + {line}")

    # ---- [4/4] 커밋 ----------------------------------------
    _info()
    _info("[4/4] 커밋")
    commit = _run(["git", "commit", "-m", args.message], cwd=repo_root, check=False)
    if commit.stdout:
        print(commit.stdout, end="")
    if commit.stderr:
        print(commit.stderr, end="", file=sys.stderr)

    if commit.returncode != 0:
        _err("커밋 실패 (returncode != 0)")
        return EXIT_GIT

    # ---- 요약 ----------------------------------------------
    _info()
    _info("=" * 60)
    _info(" ✅ 커밋 완료")
    _info("=" * 60)

    summary = _run(
        ["git", "log", "-1", "--pretty=format:%h %s%nauthor: %an <%ae>%ndate  : %ad",
         "--date=iso"],
        cwd=repo_root, check=True,
    ).stdout
    if summary:
        print(summary)
    _info()

    branch = _run(
        ["git", "rev-parse", "--abbrev-ref", "HEAD"],
        cwd=repo_root, check=True,
    ).stdout.strip()
    _info(" 다음 단계:")
    _info(f"   git push origin {branch}")

    return EXIT_OK


# ============================================================
# entrypoint
# ============================================================
if __name__ == "__main__":
    try:
        sys.exit(main())
    except KeyboardInterrupt:
        _err("사용자 중단")
        sys.exit(EXIT_INT)
```

## 커밋 스크립트 사용법

Python 3.7+ 크로스 플랫폼 (Windows / macOS / Linux 단일 파일).

### 기본 실행 (Linux / macOS)
```bash
cd /path/to/project
python3 src/v0200/tools_0200/commit_contracts_0200.py
```

기본 실행 (Windows)

```powershell

# 또는
python src/v0200/tools_0200/commit_contracts_0200.py
```

커스텀 메시지

```bash
python src/v0200/tools_0200/commit_contracts_0200.py -m "docs(contracts): F-05 reload 위임 반영"
```

Dry-run (검증만, 실제 커밋 없음)

```bash
python src/v0200/tools_0200/commit_contracts_0200.py --dry-run
```

종료 코드

코드 의미
0 성공 또는 커밋할 변경 없음
1 필수 문서 누락 (7개 파일 중 하나 이상 부재)
2 git 오류 (저장소 아님 / git 없음 / commit 실패)
130 사용자 중단 (Ctrl+C)

스크립트가 하는 일

1. git rev-parse --show-toplevel로 저장소 루트 감지
2. 7개 필수 문서 존재 검증
3. git add docs/contracts/ (staging)
4. staged 파일이 없으면 커밋 skip
5. git commit -m <메시지>
6. author/date + git push 안내 출력

요구사항

· Python 3.7 이상
· Git 2.0 이상 (git rev-parse, git diff --cached 사용)
· UTF-8 콘솔 권장 (Windows는 기본 UTF-8 강제 처리)

```

> 💡 **Windows 콘솔 인코딩 주의**: 스크립트가 `PYTHONIOENCODING=utf-8`을 내부적으로 강제하므로 별도 `chcp 65001` 불필요.
