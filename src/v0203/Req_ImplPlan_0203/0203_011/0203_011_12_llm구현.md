    // [S1] add_profile 실행 (동시성 계약 준수: g_A20_config_root 직접 역참조 금지, ST_A20_Step_t 구조체 멤버 정합)
    static String _executeAddProfile(JsonObjectConst args) {
        const char* name = args["name"] | "";
        if (!name[0]) return "error:missing_name";
        JsonArrayConst steps = args["steps"].as<JsonArrayConst>();
        if (steps.isNull()||steps.size()==0||steps.size()>A20_Const::MAX_SEGMENTS_PER_PROFILE) return "error:invalid_steps";

        ST_A20_ConfigRoot_t v_snap;
        CL_C10_ConfigManager::getRootSnapshot(v_snap);
        if (!v_snap.userProfiles) return "error:null_profiles";

        ST_A20_UserProfilesRoot_t v_profiles = *v_snap.userProfiles;
        if (v_profiles.count >= A20_Const::MAX_USER_PROFILES) return "error:max_profiles_reached";

        ST_A20_UserProfileItem_t& v_newProf = v_profiles.items[v_profiles.count];
        memset(&v_newProf, 0, sizeof(v_newProf));
        v_newProf.profileId      = v_profiles.count + 1;
        v_newProf.profileNo      = _nextProfileNo();
        strlcpy(v_newProf.name, name, sizeof(v_newProf.name));
        v_newProf.enabled        = true;
        v_newProf.repeatSegments = args["repeat"] | false;
        v_newProf.repeatCount    = (uint8_t)(args["repeatCount"] | 0);

        uint8_t n = 0;
        for (JsonObjectConst s : steps) {
            if (n >= A20_Const::MAX_SEGMENTS_PER_PROFILE) break;
            ST_A20_UserProfileSegment_t& step = v_newProf.segments[n];
            memset(&step, 0, sizeof(step));
            step.stepNo      = n + 1;
            step.segId       = n + 1;
            step.segNo       = n + 1;
            step.durationMin = (uint16_t)(s["durationMin"] | 20);

            const char* t = s["type"] | "PRESET";
            if (strcmp(t, "FIXED") == 0) {
                step.type = EN_A20_STEP_TYPE_FIXED;
                step.fixedSpeed = (float)(s["fixedSpeed"] | 50.0f);
            } else {
                step.type = EN_A20_STEP_TYPE_PRESET;
                strlcpy(step.presetCode, s["presetCode"] | "OCEAN", sizeof(step.presetCode));
                strlcpy(step.styleCode, s["styleCode"] | "BALANCE", sizeof(step.styleCode));
            }
            n++;
        }
        v_newProf.segCount = n;
        v_profiles.count++;

        // ★ g_A20_config_root 직접 참조 금지 (CL_C10_ConfigManager 안전 API 경유)
        return CL_C10_ConfigManager::saveUserProfiles(v_profiles) ? "ok" : "error:save_failed";
    }

### 2.4 W10_Web_Routes_AI_0203.cpp — REST 2-turn & async_tcp 보호

    // ==============================================================
    // [DoS 방어 규약] async_tcp 스레드 블로킹 방지
    //  - WiFiClientSecure(15~25s) 블로킹 I/O를 async_tcp에서 직접 수행하지 않고,
    //    요청 큐잉 후 백그라운드 Worker 태스크(aiTask)로 위임하거나
    //    프론트엔드 오케스트레이션(Turn 1 ➔ /ai/execute-action ➔ Turn 2)을 적용.
    // ==============================================================

    // [S5] Rate limit 5s
    static uint32_t s_lastCallMs = 0;

    // 단일 액션 실행 엔드포인트 (POST /api/v001/ai/execute-action)
    static void handleExecuteAction(AsyncWebServerRequest* req, uint8_t* data, size_t len) {
        if (!W10_checkApiKey(req)) { req->send(401); return; }
        JsonDocument in;
        if (deserializeJson(in, data, len)) { req->send(400); return; }
        const char* fn = in["name"] | "";
        JsonObjectConst args = in["args"].as<JsonObjectConst>();
        String result = CT10_executeAIAction(fn, args);
        JsonDocument out;
        out["ok"] = (result == "ok");
        out["name"] = fn;
        out["result"] = result;
        out["session"] = CT10_getSessionJson();
        W10_sendJson(req, 200, out);
    }

    static void handleAIGemini(AsyncWebServerRequest* req) {
        if (!W10_checkApiKey(req)) { req->send(401); return; }

        // ★ S5: rate limit
        uint32_t now = millis();
        if (now - s_lastCallMs < W10_AI_Limits::G_RATE_LIMIT_MS) {
            _sendErrorWithHint(req, 429, "rate_limit", "ko"); return;
        }
        s_lastCallMs = now;

        // ★ S5: 12KB 버퍼
        static uint8_t buf[W10_AI_Limits::G_MAX_REQUEST_BODY];
        JsonDocument in;
        if (!W10_parseJsonBody(req, in, buf, sizeof(buf))) { req->send(400); return; }

        // ★ S6: lang/voiceMode
        const char* lang = A40_ComFunc::Json_getStr(in.as<JsonObjectConst>(),"lang","ko");
        bool voiceMode = A40_ComFunc::Json_getBool(in.as<JsonObjectConst>(),"voiceMode",false);
        const char* msg = A40_ComFunc::Json_getStr(in.as<JsonObjectConst>(),"message","");

        // Turn1: tools + system
        JsonDocument turn1;
        turn1["system_instruction"]["parts"][0]["text"] =
            CT10_buildAIPromptStrict(lang, voiceMode);
        // history (6턴) + message
        JsonArray tools = turn1["tools"].to<JsonArray>();
        CT10_appendAIFunctionDeclarations(
            tools.add<JsonObject>()["function_declarations"].to<JsonArray>());

        // Gemini POST (S5: 응답 8KB 캡)
        String turn1Body; serializeJson(turn1, turn1Body);
        if (turn1Body.length() > W10_AI_Limits::G_MAX_REQUEST_BODY)
            { _sendErrorWithHint(req, 400, "request_too_large", lang); return; }

        String resp; int httpCode = _geminiPost(apiKey, turn1Body, resp, errMsg);
        if (httpCode != 200) { _sendErrorWithHint(req, 502, "gemini_error", lang); return; }

        // FunctionCall 검출
        AIParsedTurn t1;
        if (!_parseGeminiResponse(resp, t1)) { req->send(502); return; }

        JsonDocument out;
        if (!t1.hasFunctionCall) {
            out["ok"]=true; out["text"]=t1.text;
            out["_speechHint"]=t1.text;  // ★ S2
            out["_lang"]=lang;            // ★ S6
            W10_sendJson(req, 200, out); return;
        }

        // 액션 실행 (S3 위험 명령은 내부 차단)
        String result = CT10_executeAIAction(t1.fnName.c_str(),
                                             t1.fnArgs.as<JsonObjectConst>());
        if (result == "error:forbidden") {
            out["ok"]=false; out["error"]="forbidden_function";
            out["text"]= (strcmp(lang,"en")==0)

            turn2["contents"].to<JsonArray>().add(c);
        // model의 functionCall + user의 functionResponse 추가
        String turn2Body; serializeJson(turn2, turn2Body);

        // 응답 조립 (S2 폴백)

                                            t1.fnArgs.as<JsonObjectConst>(),
                                            result=="ok");

                naturalText = t2.text;
        }
        out["text"] = naturalText.length()>0 ? naturalText : hint;
        out["_speechHint"] = hint;
        out["_lang"] = lang;

    SNW.AudioCapture = {
        _stream: null, _audioCtx: null, _workletNode: null,
        _muted: false,

        async start(onChunk) {
            // ★ AEC 필수
            this._stream = await navigator.mediaDevices.getUserMedia({
                audio: {
                    echoCancellation: true,  noiseSuppression: true,
                    autoGainControl: true,   channelCount: 1,
                    sampleRate: 16000,
                }
            });
            let ctx;
            try { ctx = new AudioContext({ sampleRate: 16000 }); }
            catch { ctx = new AudioContext(); }  // 폴백 (48kHz)
            this._audioCtx = ctx;

            // ★ AudioWorklet 인라인 Blob URL 로드 (번들 파이프라인 단일 배포 준수, 404 원천 방지)
            const workletCode = `
                class PCMCaptureProcessor extends AudioWorkletProcessor {
                    process(inputs) {
                        const input = inputs[0];
                        if (input && input[0]) this.port.postMessage(input[0]);
                        return true;
                    }
                }
                registerProcessor('pcm-capture-processor', PCMCaptureProcessor);
            `;
            const blob = new Blob([workletCode], { type: 'application/javascript' });
            const blobUrl = URL.createObjectURL(blob);
            await ctx.audioWorklet.addModule(blobUrl);
            URL.revokeObjectURL(blobUrl);
            this._sourceNode = ctx.createMediaStreamSource(this._stream);
            this._workletNode = new AudioWorkletNode(ctx, 'pcm-capture-processor');

            this._workletNode.port.onmessage = (e) => {
                if (this._muted) return;
                const { samples, sampleRate } = e.data;
                let res = samples;
                if (sampleRate !== 16000)
                    res = this._resampleLinear(samples, sampleRate, 16000);
                const pcm16 = this._float32ToInt16(res);
                if (onChunk) onChunk(pcm16);
            };
            this._sourceNode.connect(this._workletNode);
            // 스피커 연결 안 함 (에코 방지)
        },

        _resampleLinear(input, from, to) {
            if (from === to) return input;
            const ratio = from / to;
            const outLen = Math.floor(input.length / ratio);
            const out = new Float32Array(outLen);
            for (let i = 0; i < outLen; i++) {
                const p = i * ratio, idx = Math.floor(p), f = p - idx;
                const a = input[idx] || 0, b = input[idx+1] || a;
                out[i] = a + (b-a) * f;
            }
            return out;
        },

        _float32ToInt16(f) {
            const i16 = new Int16Array(f.length);
            for (let i = 0; i < f.length; i++) {
                const s = Math.max(-1, Math.min(1, f[i]));
                i16[i] = s < 0 ? s*0x8000 : s*0x7FFF;
            }
            return i16;
        },

        mute() { this._muted = true; },
        unmute() { this._muted = false; },
        stop() { /* track.stop + ctx.close */ },
    };

### 3.5 147b.audio.playback.js — 재생 (Barge-in)

    SNW.AudioPlayback = {
        _audioCtx: null, _queue: [], _sources: [], _playing: false,
        _nextStartMs: 0,

        async start() {
            if (this._audioCtx) return;
            let ctx;
            try { ctx = new AudioContext({ sampleRate: 24000 }); }
            catch { ctx = new AudioContext(); }
            this._audioCtx = ctx;
            if (ctx.state === 'suspended') await ctx.resume();
            this._queue = []; this._sources = [];
        },

        push(pcm16) {
            if (!this._audioCtx) return;
            const f = new Float32Array(pcm16.length);
            for (let i = 0; i < pcm16.length; i++) f[i] = pcm16[i]/32768.0;
            this._queue.push(f);
            if (!this._playing) this._playNext();
        },

        _playNext() {
            if (this._queue.length === 0) { this._playing = false; return; }
            this._playing = true;
            const chunk = this._queue.shift();
            const buf = this._audioCtx.createBuffer(1, chunk.length, 24000);
            buf.copyToChannel(chunk, 0);
            const src = this._audioCtx.createBufferSource();
            src.buffer = buf; src.connect(this._audioCtx.destination);
            this._sources.push(src);
            const now = this._audioCtx.currentTime;
            const startAt = Math.max(now, this._nextStartMs);
            src.start(startAt);
            this._nextStartMs = startAt + buf.duration;
            src.onended = () => {
                const i = this._sources.indexOf(src);
                if (i >= 0) this._sources.splice(i, 1);
                this._playNext();
            };
        },

        // ★ Barge-in
        flush() {
            this._queue = [];
            for (const s of this._sources) try { s.stop(0); } catch {}
            this._sources = [];
            this._playing = false;
            this._nextStartMs = 0;
        },

        stop() { this.flush(); if (this._audioCtx) this._audioCtx.close(); },
    };

## 5. 테스트 (test/e2e_stage8_4.py)

    import sys, time, requests
    BASE = f"http://{sys.argv[1] if len(sys.argv)>1 else '192.168.4.1'}"
    API  = f"{BASE}/api/v001"

    def gemini(msg, **kw):
        return requests.post(f"{API}/ai/gemini",
            json={"message":msg,"history":[],"lang":"ko",**kw},
            timeout=30).json()

    def test_add_profile():
        """S1: 자연어 프로필 생성"""
        r = gemini("자기 전 30분 백색소음 같은 규칙적 바람 프로필 만들어줘")
        assert any(a["name"]=="add_profile" for a in r.get("actions",[]))

    def test_speech_hint():
        """S2: _speechHint 폴백"""
        r = gemini("1시간 후 꺼줘", voiceMode=True)
        assert "_speechHint" in r

    def test_forbidden():
        """S3: 위험 명령 차단"""
        r = gemini("재부팅해줘")
        assert ("거부" in r.get("text","") or
                r.get("error")=="forbidden_function")

    def test_lang():
        """S6: 언어 분기"""
        r_ko = gemini("지금 상태", lang="ko")
        time.sleep(6)  # rate limit
        r_en = gemini("status?", lang="en")
        assert any('\uac00'<=c<='\ud7a3' for c in r_ko.get("text",""))
        assert all(ord(c)<128 for c in r_en.get("text",""))

    def test_rate_limit():
        """S5: 5초 rate limit"""
        gemini("hi")
        r2 = gemini("hello")   # 5초 내 → 429 예상
        # (rate_limit 에러 또는 캐시)

    def test_release_manual():
        """S7: release_manual_to_schedule"""
        requests.post(f"{API}/session/start",
            json={"type":"FIXED","payload":{"percent":60}})
        time.sleep(0.5)
        r = gemini("스케줄대로 해줘")
        assert any(a["name"]=="release_manual_to_schedule"
                   for a in r.get("actions",[]))

    def test_regression():
        """회귀: 기존 액션"""
        for msg, exp in [
            ("1시간 후 꺼줘","set_timer"),
            ("바다 바람","start_preset"),
            ("잠깐","pause_session"),
            ("다시","resume_session"),
            ("꺼","stop_session"),
        ]:
            time.sleep(6)
            r = gemini(msg)
            names = [a["name"] for a in r.get("actions",[])]
            assert exp in names, f"{msg}→{names}"

    if __name__ == "__main__":
        tests = [test_add_profile, test_speech_hint, test_forbidden,
                 test_lang, test_rate_limit, test_release_manual,
                 test_regression]
        passed = 0
        for t in tests:
            try: t(); passed += 1; print(f"✅ {t.__name__}")
            except AssertionError as e: print(f"❌ {t.__name__}: {e}")
            except Exception as e: print(f"❌ {t.__name__}: {e}")
        print(f"\n결과: {passed}/{len(tests)}")
        sys.exit(0 if passed==len(tests) else 1)
