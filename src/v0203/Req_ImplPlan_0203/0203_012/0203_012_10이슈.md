T-122. [BE-P0-2] AI add_profile CRUD 검증 우회 차단

목적: AI가 생성한 프로필이 CRUD 헬퍼(C10_nextProfileId_u8/C10_isProfileNoDuplicated/C10_validateUserProfilesBeforeSave)를 경유하는지 검증.

검증 절차:

```bash
# 1. 기존 프로필 개수 + 최대 profileNo 기록
BEFORE=$(curl -s http://$IP/api/v001/user_profiles | jq '.userProfiles.profiles | length')
MAX_NO=$(curl -s http://$IP/api/v001/user_profiles | jq '.userProfiles.profiles | map(.profileNo) | max')

# 2. AI로 프로필 생성 (동일 profileNo 재사용 유도)
curl -X POST http://$IP/api/v001/ai/execute-action \
  -H "Content-Type: application/json" \
  -d '{"name":"add_profile","args":{"name":"AI Test","steps":[
    {"durationMin":20,"type":"PRESET","presetCode":"OCEAN","styleCode":"BALANCE"}
  ]}}' | jq '.result'

# 3. 신규 profileNo가 기존 max+1 이상인지 확인
AFTER_MAX=$(curl -s http://$IP/api/v001/user_profiles | jq '.userProfiles.profiles | map(.profileNo) | max')
echo "Before max=$MAX_NO / After max=$AFTER_MAX"
# 기대: AFTER_MAX > MAX_NO (자동 발급 정책 준수)

# 4. userProfiles 전체 로드 재검증 (재부팅)
curl -X POST http://$IP/api/v001/control/reboot
sleep 25
curl -s http://$IP/api/v001/user_profiles | jq '.userProfiles.profiles | length'
# 기대: BEFORE+1 (검증 실패 시 C10_validateUserProfilesLoaded가 reject → 전체 로드 실패)

# 5. 시리얼 로그 확인
grep "duplicated profileNo\|Load rejected" serial.log
# 기대: 0건
```

합격 판정:

☐ 3단계: 신규 profileNo가 기존 max+1 이상
☐ 4단계: 재부팅 후 프로필 개수 유지
☐ 5단계: 검증 실패 로그 0건

실패 시 조치: executeAddProfile을 addUserProfilesFromJson 위임으로 재작성.

    v_prof["name"]           = p_args["name"] | "AI Profile";

3.6 상세 — validate_contracts.py 신규

```python
#!/usr/bin/env python3
"""
validate_contracts.py — SSOT 자동 대조 CI
- SPEC ↔ W10_Web_Const 경로 대조
- SPEC ↔ W10_AI_Limits 상수 대조
- C++ FSM 상태명 ↔ offline 어댑터 대조
- SPEC ↔ config.js API 경로 대조
"""
import re
import sys
from pathlib import Path

def check_api_paths(repo: Path) -> list[str]:
    """SPEC §5.1.4.1 표 ↔ W10_Web_Const_0203.h 대조"""
    spec = (repo / "src/v0203/contract_doc_0203/SPEC_0203.md").read_text(encoding="utf-8")
    hdr = (repo / "src/v0203/W10_Web_Const_0203.h").read_text(encoding="utf-8")

    violations = []
    # SPEC의 `/ai/live-*` 경로들 추출
    for m in re.finditer(r"\| `/api/v001(/ai/[a-z-]+)`", spec):
        spec_path = m.group(1)
        # 헤더에 해당 경로 상수가 있는지 (문자열 매칭)
        if f'"/ai/{spec_path.split("/ai/")[-1]}"' not in hdr and f'"{spec_path}"' not in hdr:
            violations.append(f"SPEC {spec_path} ↔ 헤더 불일치")
    return violations

def check_ai_limits(repo: Path) -> list[str]:
    """SPEC §5.1.4.4 ↔ W10_Web_0203.h 대조"""
    spec = (repo / "src/v0203/contract_doc_0203/SPEC_0203.md").read_text(encoding="utf-8")
    hdr = (repo / "src/v0203/W10_Web_0203.h").read_text(encoding="utf-8")

    violations = []
    # G_MAX_REQUEST_BODY 값 추출
    m_h = re.search(r"G_MAX_REQUEST_BODY\s*=\s*(\d+)", hdr)
    m_s = re.search(r"`G_MAX_REQUEST_BODY\s*=\s*(\d+)`", spec)
    if m_h and m_s and m_h.group(1) != m_s.group(1):
        violations.append(f"G_MAX_REQUEST_BODY: SPEC={m_s.group(1)} vs 헤더={m_h.group(1)}")
    return violations

def check_fsm_names(repo: Path) -> list[str]:
    """C++ FSM 상태명 ↔ offline 어댑터 대조"""
    ctl = (repo / "src/v0203/CT10_Ctl_StMG_0203.cpp").read_text(encoding="utf-8")
    offline = (repo / "src/v0203/data_v0203_src/html_v0203/js/998.offline_0203.js").read_text(encoding="utf-8")

    valid_states = set(re.findall(r'return "([A-Z_]+)";', ctl))
    used_states = set(re.findall(r"state\s*[:=]\s*'([A-Z_]+)'", offline))

    violations = []
    for st in used_states:
        if st and st not in valid_states and st not in ("ACTIVE", "PAUSED", "COMPLETED", "NONE"):
            violations.append(f"offline state '{st}' not in C++ FSM")
    return violations

def main() -> int:
    repo = Path(__file__).resolve().parents[3]
    all_v = []
    all_v += check_api_paths(repo)
    all_v += check_ai_limits(repo)
    all_v += check_fsm_names(repo)

    if all_v:
        print(f"[FAIL] {len(all_v)} SSOT violations:")
        for v in all_v:
            print(f"  - {v}")
        return 1
    print("[OK] All SSOT contracts consistent.")
    return 0

if __name__ == "__main__":
    sys.exit(main())
```
