4. 동기화 스크립트 (tools/sync-agent-configs.py)

Windows/macOS/Linux 크로스 플랫폼, symlink 불필요.

```python
#!/usr/bin/env python3
"""`.agents/` → `.claude/` 미러. pre-commit hook 또는 수동 실행."""
import re, shutil, sys
from pathlib import Path

ROOT = Path(__file__).parent.parent
AGENTS = ROOT / ".agents"
CLAUDE  = ROOT / ".claude"

def parse_frontmatter(text: str):
    """(frontmatter_dict_raw, body) 반환. `---` 블록 단순 파싱."""
    m = re.match(r"^---\n(.*?)\n---\n(.*)$", text, re.DOTALL)
    if not m:
        return {}, text
    fm_raw, body = m.group(1), m.group(2)
    return {"raw": fm_raw}, body

def convert_rule_frontmatter(raw: str) -> str:
    """Antigravity(trigger+globs) → Claude Code(paths)."""
    if "trigger:" not in raw or "globs:" not in raw:
        return ""  # model_decision 등은 paths 없음 → frontmatter 생략
    m = re.search(r'globs:\s*"([^"]+)"', raw)
    if not m:
        return ""
    globs = [g.strip() for g in m.group(1).split(",") if g.strip()]
    lines = ["paths:"]
    for g in globs:
        lines.append(f'  - "{g}"')
    return "\n".join(lines)

def sync_rules():
    src_dir = AGENTS / "rules"
    dst_dir = CLAUDE / "rules"
    if not src_dir.exists(): return
    dst_dir.mkdir(parents=True, exist_ok=True)
    for src in src_dir.glob("*.md"):
        fm, body = parse_frontmatter(src.read_text(encoding="utf-8"))
        new_fm = convert_rule_frontmatter(fm.get("raw", ""))
        out = f"---\n{new_fm}\n---\n{body}" if new_fm else body
        (dst_dir / src.name).write_text(out, encoding="utf-8")
        print(f"[rule] {src.name} → {dst_dir/ src.name}")

def sync_skills():
    src_dir = AGENTS / "skills"
    dst_dir = CLAUDE / "skills"
    if not src_dir.exists(): return
    for skill_dir in src_dir.iterdir():
        if not (skill_dir / "SKILL.md").exists(): continue
        target = dst_dir / skill_dir.name
        if target.exists(): shutil.rmtree(target)
        shutil.copytree(skill_dir, target)
        print(f"[skill] {skill_dir.name}")

def sync_workflows():
    src_dir = AGENTS / "workflows"
    dst_dir = CLAUDE / "commands"
    if not src_dir.exists(): return
    dst_dir.mkdir(parents=True, exist_ok=True)
    for src in src_dir.glob("*.md"):
        body = src.read_text(encoding="utf-8")
        # 첫 H1을 description으로
        m = re.search(r"^#\s+(.+)$", body, re.MULTILINE)
        desc = m.group(1).strip() if m else src.stem
        out = f"---\ndescription: {desc}\n---\n\n{body}"
        (dst_dir / src.name).write_text(out, encoding="utf-8")
        print(f"[workflow] {src.name}")

if __name__ == "__main__":
    CLAUDE.mkdir(exist_ok=True)
    sync_rules()
    sync_skills()
    sync_workflows()
    print("Sync complete.")
```

1. tools/sync-agent-configs.py

```python
#!/usr/bin/env python3
"""`.agents/` → `.claude/` 동기화 (SmartNatureWind v0203).

`.agents/` 를 canonical로 하고 `.claude/` 는 생성물로 취급한다.
CI/pre-commit 에서 `--check` 로 drift 감지.

Usage:
    python tools/sync-agent-configs.py           # sync
    python tools/sync-agent-configs.py --check   # drift 검사 (exit 1 if drift)
"""
from __future__ import annotations

import argparse
import re
import shutil
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
AGENTS = ROOT / ".agents"
CLAUDE = ROOT / ".claude"

FRONTMATTER_RE = re.compile(r"^---\r?\n(.*?)\r?\n---\r?\n(.*)$", re.DOTALL)


def split_frontmatter(text: str) -> tuple[str, str]:
    m = FRONTMATTER_RE.match(text)
    return (m.group(1), m.group(2)) if m else ("", text)


def extract_globs(fm_raw: str) -> list[str]:
    m = re.search(r'^globs:\s*"?([^"\n]+)"?', fm_raw, re.MULTILINE)
    if not m:
        return []
    return [g.strip() for g in m.group(1).split(",") if g.strip()]


def convert_rule(agents_text: str) -> str:
    """Antigravity(trigger/globs) → Claude Code(paths)."""
    fm_raw, body = split_frontmatter(agents_text)
    if "trigger:" not in fm_raw:
        return body

    trigger_m = re.search(r"^trigger:\s*(\S+)", fm_raw, re.MULTILINE)
    trigger = trigger_m.group(1) if trigger_m else ""

    if trigger == "glob":
        globs = extract_globs(fm_raw)
        if not globs:
            return body
        paths_yaml = "\n".join(f'  - "{g}"' for g in globs)
        return f"---\npaths:\n{paths_yaml}\n---\n{body}"

    # model_decision / always_on → frontmatter 생략 (Claude Code는 항상 로드)
    return body


def convert_workflow(agents_text: str, fallback_desc: str) -> str:
    fm_raw, body = split_frontmatter(agents_text)
    if fm_raw:
        return agents_text
    m = re.search(r"^#\s+(.+)$", body, re.MULTILINE)
    desc = m.group(1).strip() if m else fallback_desc
    return f"---\ndescription: {desc}\n---\n\n{body}"


class SyncStats:
    def __init__(self) -> None:
        self.written = 0
        self.drift = 0


def _write_or_check(target: Path, content: str, check: bool, stats: SyncStats) -> None:
    if target.exists() and target.read_text(encoding="utf-8") == content:
        return
    if check:
        print(f"[DRIFT] {target.relative_to(ROOT)}")
        stats.drift += 1
        return
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(content, encoding="utf-8")
    print(f"[write] {target.relative_to(ROOT)}")
    stats.written += 1


def sync_rules(check: bool, stats: SyncStats) -> None:
    src_dir = AGENTS / "rules"
    if not src_dir.is_dir():
        return
    for src in sorted(src_dir.glob("*.md")):
        converted = convert_rule(src.read_text(encoding="utf-8"))
        _write_or_check(CLAUDE / "rules" / src.name, converted, check, stats)


def sync_skills(check: bool, stats: SyncStats) -> None:
    src_dir = AGENTS / "skills"
    if not src_dir.is_dir():
        return
    for skill_dir in sorted(src_dir.iterdir()):
        if not (skill_dir / "SKILL.md").is_file():
            continue
        dst_dir = CLAUDE / "skills" / skill_dir.name
        for f in skill_dir.rglob("*"):
            if not f.is_file():
                continue
            target = dst_dir / f.relative_to(skill_dir)
            if check:
                if not target.exists() or target.read_bytes() != f.read_bytes():
                    print(f"[DRIFT] {target.relative_to(ROOT)}")
                    stats.drift += 1
            else:
                target.parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(f, target)
                stats.written += 1


def sync_workflows(check: bool, stats: SyncStats) -> None:
    src_dir = AGENTS / "workflows"
    if not src_dir.is_dir():
        return
    for src in sorted(src_dir.glob("*.md")):
        converted = convert_workflow(
            src.read_text(encoding="utf-8"), fallback_desc=src.stem
        )
        _write_or_check(CLAUDE / "commands" / src.name, converted, check, stats)


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--check", action="store_true", help="drift만 감지 (CI/pre-commit)")
    args = ap.parse_args()

    stats = SyncStats()
    sync_rules(args.check, stats)
    sync_skills(args.check, stats)
    sync_workflows(args.check, stats)

    if args.check:
        print(f"\nCheck: {stats.drift} drift(s)")
        return 1 if stats.drift else 0

    print(f"\nSync complete: {stats.written} file(s) written")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def is_backend(path: str) -> bool:
    if not path:
        return False
    normalized = path.replace("\\", "/")
    return normalized.startswith(BACKEND_PREFIX) and normalized.endswith((".h", ".cpp"))


def main() -> int:
    try:
        data = json.load(sys.stdin)
    except json.JSONDecodeError:
        return 0  # 파싱 실패는 allow (fail-open)

    tool_input = data.get("tool_input", {})
    path = (
        tool_input.get("path")
        or tool_input.get("file_path")
        or tool_input.get("filePath")
        or ""
    )

    if not is_backend(path):
        return 0

    if MARKER.is_file():
        return 0

    print(
        json.dumps(
            {
                "decision": "deny",
                "reason": (
                    f"백엔드 수정 전 승인 필요: {path}\n"
                    "AGENTS.md 원칙 1: 목적·기능변화를 정리해 사용자 승인 후 "
                    ".agents/state/backend-approval.json 생성하세요."
                ),
            }
        )
    )
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# tools/split_audit_log.py — 1회 실행 (idempotent)
"""CONTRACT_0203.md §7.1~§7.10 → AUDIT_LOG_0203.md 이관."""
import re
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
CONTRACT = ROOT / "src/v0203/contract_doc_0203/CONTRACT_0203.md"
AUDIT_LOG = ROOT / "src/v0203/contract_doc_0203/AUDIT_LOG_0203.md"

text = CONTRACT.read_text(encoding="utf-8")

# §7.1 시작부터 §7.11 직전까지 (감사 이력)
m_start = re.search(r"^### 7\.1\b", text, re.MULTILINE)
m_end = re.search(r"^### 7\.11\b", text, re.MULTILINE)

if not (m_start and m_end):
    print("SKIP: 이미 분리됨 또는 패턴 불일치")
    raise SystemExit(0)

historical = text[m_start.start():m_end.start()].rstrip()
print(f"추출: {len(historical):,} chars")

# AUDIT_LOG 파일 골격
skeleton = (ROOT / "tools/_audit_log_skeleton.md").read_text(encoding="utf-8")
AUDIT_LOG.write_text(skeleton.replace("<!-- HISTORICAL_BODY -->", historical), encoding="utf-8")

# CONTRACT 재작성
replacement = (
    "## 7. 감사 관측 및 이슈 추적표\n\n"
    "> **감사 이력 (F-01~Phase 6, 모두 해결)은 "
    "[`AUDIT_LOG_0203.md`](AUDIT_LOG_0203.md)로 이관**되었습니다.\n"
    "> 본 §7은 **현재 활성 계약** (§7.11~§7.18)만 유지합니다.\n\n"
    "### 7.0 이력 참조 인덱스\n\n"
    "| 시리즈 | 대상 | 상세 링크 |\n|---|---|---|\n"
    "| F-01~I-6 | 초기 감사 | AUDIT_LOG §1 |\n"
    "| N/G 시리즈 | N-1~N-4, G-1~G-3 | AUDIT_LOG §2 |\n"
    "| Batch A~F | A-1~F-10 | AUDIT_LOG §3 |\n"
    "| Batch G | C-1~C-5, B-1~B-13 | AUDIT_LOG §4, §5 |\n"
    "| Batch H/I | LOAD, OVR, OFFLINE | AUDIT_LOG §6~§8 |\n"
    "| SYNC | SYNC-1~3 | AUDIT_LOG §9 |\n"
    "| Phase 1~6 | N-5~N-30, P-1~P-5 | AUDIT_LOG §10 |\n"
    "| PHY | PHY-1~4 | AUDIT_LOG §11 |\n\n---\n\n"
)
new_text = text[: m_start.start()] + replacement + text[m_end.start():]
CONTRACT.write_text(new_text, encoding="utf-8")
print(f"CONTRACT 축소: {len(text):,} → {len(new_text):,} chars")
```

4. tools/graphify_init.py

graphify-out/graph.json 이 없는 초기 환경에서 1회 실행. PowerShell 의존 없이 cross-platform.

```python
#!/usr/bin/env python3
"""v0203 graphify 초기 빌드 래퍼 (cross-platform).

Usage:
    python tools/graphify_init.py             # full build
    python tools/graphify_init.py --check     # 기존 graph 존재 확인만
    python tools/graphify_init.py --update    # 증분 빌드

기존 .agents/skills/graphify/SKILL.md 는 PowerShell 예제 기준이므로,
이 래퍼는 Windows/macOS/Linux 공통 엔트리를 제공한다.
"""
from __future__ import annotations

import argparse
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
GRAPH_JSON = ROOT / "graphify-out" / "graph.json"


def have_graphify() -> str | None:
    for name in ("graphify", "graphifyy"):
        p = shutil.which(name)
        if p:
            return p
    return None


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--check", action="store_true", help="graph.json 존재만 확인")
    ap.add_argument("--update", action="store_true", help="증분 빌드")
    ap.add_argument("--mode", default="normal", choices=("normal", "deep"),
                    help="추출 모드 (deep: 더 풍부한 INFERRED 엣지)")
    args = ap.parse_args()

    if args.check:
        if GRAPH_JSON.is_file():
            size_kb = GRAPH_JSON.stat().st_size / 1024
            print(f"OK: {GRAPH_JSON.relative_to(ROOT)} ({size_kb:.0f} KB)")
            return 0
        print(f"MISSING: {GRAPH_JSON.relative_to(ROOT)}")
        return 1

    cli = have_graphify()
    if not cli:
        print("ERROR: graphify CLI 미설치. `pip install graphifyy` 또는 `uv tool install graphifyy`", file=sys.stderr)
        return 2

    cmd = [cli]
    if args.update:
        cmd += [".", "--update"]
    else:
        cmd += [".", "--mode", args.mode]
    print(f"$ {' '.join(cmd)}")
    return subprocess.call(cmd, cwd=ROOT)


if __name__ == "__main__":
    sys.exit(main())
```

5. tools/approve_backend.py

원칙 1(백엔드 승인) UX. backend-approval.json을 안전하게 생성.

```python
#!/usr/bin/env python3
"""백엔드 C++ 수정 승인 마커 생성 (AGENTS.md 원칙 1).

Usage:
    python tools/approve_backend.py \\
        --purpose "Session FSM pause 버그 수정" \\
        --files "src/v0203/CT10_Ctl_Ctl_0203.cpp,src/v0203/CT10_Ctl_0203.h"

마커: .agents/state/backend-approval.json
Hook(`.agents/hooks/require_backend_approval.py`)이 이 마커 존재 여부로 allow/deny 판정.

작업 완료 후 반드시:
    python tools/approve_backend.py --revoke
"""
from __future__ import annotations

import argparse
import json
import sys
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
STATE_DIR = ROOT / ".agents" / "state"
MARKER = STATE_DIR / "backend-approval.json"


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--purpose", help="수정 목적·기능변화 요약 (필수, 10자 이상)")
    ap.add_argument("--files", help="대상 파일 (comma-separated)")
    ap.add_argument("--revoke", action="store_true", help="승인 마커 삭제")
    args = ap.parse_args()

    if args.revoke:
        if MARKER.is_file():
            MARKER.unlink()
            print(f"REVOKED: {MARKER.relative_to(ROOT)}")
        else:
            print("NO-OP: 마커 없음")
        return 0

    if not args.purpose or len(args.purpose.strip()) < 10:
        print("ERROR: --purpose 는 10자 이상 필요 (AGENTS.md 원칙 1: 목적·기능변화 명확화)", file=sys.stderr)
        return 1

    if not args.files:
        print("ERROR: --files 필요", file=sys.stderr)
        return 1

    files = [f.strip() for f in args.files.split(",") if f.strip()]
    bad = [f for f in files if not f.startswith("src/v0203/")]
    if bad:
        print(f"WARN: 백엔드 경로 아님: {bad}", file=sys.stderr)

    STATE_DIR.mkdir(parents=True, exist_ok=True)
    marker = {
        "approved": True,
        "purpose": args.purpose.strip(),
        "files": files,
        "approved_at": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "approved_by": "user",
    }
    MARKER.write_text(json.dumps(marker, ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"APPROVED: {MARKER.relative_to(ROOT)}")
    print(f"  purpose: {marker['purpose']}")
    print(f"  files:   {len(files)}개")
    print("\n작업 완료 후: python tools/approve_backend.py --revoke")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
