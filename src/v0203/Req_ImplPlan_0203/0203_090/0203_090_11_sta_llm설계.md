## 1. 아키텍처 개요

### 1.1 전체 구조

    ┌──────────────────────────────────────────────────────────────┐
    │  ESP32-S3 (Standalone Voice)                                 │
    │                                                              │
    │  [1] ESP-SR 웨이크워드 감지 (상시)                            │
    │       ↓ "Hi Wind"                                            │
    │  [2] LED 녹색 + 비프 100ms                                   │
    │       ↓                                                      │
    │  [3] I2S 녹음 5초 (PSRAM 160KB)                              │
    │       ↓                                                      │
    │  [4] WAV 인코딩 (44-byte header + PCM)                       │
    │       ↓                                                      │
    │  [5] Base64 인코딩 (213KB)                                   │
    │       ↓                                                      │
    │  [6] HTTPS POST → Gemini API (inline_data: audio/wav)        │
    │       ↓                                                      │
    │  [7] FunctionCall 수신 → CT10 실행 (기존 12개 재사용)         │
    │       ↓                                                      │
    │  [8] 2-turn → 자연어 응답 텍스트                              │
    │       ↓                                                      │
    │  [9] Cloud TTS or Gemini TTS → PCM 24kHz                     │
    │       ↓                                                      │
    │  [10] I2S 재생 (마이크 mute)                                 │
    │       ↓                                                      │
    │  [11] 상태 LED 복귀                                          │
    └──────────────────────────────────────────────────────────────┘

### 1.2 핵심 결정

| 항목 | 결정 | 이유 |
|---|---|---|
| 웨이크워드 | ESP-SR (프리셋 or 커스텀) | ESP32-S3 전용 최적 |
| 녹음 | 5초 일괄 (비 스트리밍) | ESP32 오디오 중계 불가 |
| STT | Gemini audio inline_data | 통합 단일 API |
| TTS | Google Cloud TTS | Gemini TTS는 별도 API |
| 재생 | 반이중 (mute 필수) | AEC 없음 |
| Live API | 사용 안 함 | ESP32 오디오 중계 불가 |

### 1.3 제약 기반 설계 원칙

| 원칙 | 이유 |
|---|---|
| **PSRAM 8MB 필수** | WAV 160KB + ESP-SR 모델 200KB |
| **5초 일괄 처리** | ESP32 CPU/메모리 한계 |
| **재생 중 mute** | AEC 부재 |
| **로컬 피드백 강제** | 4초 지연 체감 완화 |
| **GO/NO-GO 검증** | 상품화 리스크 |

## 4. 상태 머신

### 4.1 상태 정의

    typedef enum : uint8_t {
        EN_VOICE_IDLE         = 0,   // 웨이크워드 감지 대기
        EN_VOICE_WAKE_DETECTED,      // 감지 → LED + 비프
        EN_VOICE_RECORDING,          // 5초 녹음
        EN_VOICE_UPLOADING,          // WAV 인코딩 + Gemini 전송
        EN_VOICE_PROCESSING,         // FunctionCall 실행
        EN_VOICE_TTS_FETCHING,       // TTS 오디오 수신
        EN_VOICE_PLAYING,            // 스피커 재생
        EN_VOICE_ERROR,              // 오류
    } EN_VOICE_state_t;

### 4.2 상태 전이

    [IDLE]
      │ 웨이크워드 감지
      ▼
    [WAKE_DETECTED]  ← LED 녹색 + 비프 100ms (300ms)
      │
      ▼
    [RECORDING]  ← 5초 녹음 (PSRAM 버퍼)
      │
      ▼
    [UPLOADING]  ← WAV 인코딩 + Base64 + HTTPS POST
      │
      ▼
    [PROCESSING] ← FunctionCall 실행 (CT10 12개 재사용)
      │
      ▼
    [TTS_FETCHING] ← 2-turn 응답 + TTS 요청
      │
      ▼
    [PLAYING]  ← I2S 재생 (마이크 mute)
      │
      ▼
    [IDLE]

    (오류 발생 시)
      ▼
    [ERROR] → LED 보라 3회 점멸 → IDLE

| IDLE | 파랑 (호흡) | - |

### 5.2 웨이크워드 (P_Voice_WakeWord_0203.cpp)

    #include "esp_sr.h"
    #include "model_path.h"

    static srmodel_list_t* models = nullptr;
    static esp_sr_iface_t* sr_handle = nullptr;
    static esp_sr_data_t*  sr_data = nullptr;

    bool voice_wakeword_init() {
        models = esp_srmodel_init("model");   // Flash 파티션
        if (!models) return false;

        sr_handle = esp_sr_handle_from_name("wn9_hilexin");
        if (!sr_handle) return false;

        sr_data = sr_handle->create(models, nullptr);
        return (sr_data != nullptr);
    }

    // 웨이크워드 감지 태스크 (Core 0, Priority 5)
    void voice_wakeword_task(void* arg) {
        int16_t buf[512];
        size_t bytesRead = 0;
        while (1) {
            i2s_read(I2S_NUM_0, buf, sizeof(buf), &bytesRead, portMAX_DELAY);
            int wakeIdx = sr_handle->detect(sr_data, buf);
            if (wakeIdx >= 0) {
                // 감지 → 상태머신에 신호
                voice_fsm_post_event(EN_VOICE_EVT_WAKE);
            }
        }
    }

### 5.3 WAV 인코딩 (P_Voice_WAV_0203.cpp)

    typedef struct __attribute__((packed)) {
        char     riff[4];       // "RIFF"
        uint32_t fileSize;      // 총 - 8
        char     wave[4];       // "WAVE"
        char     fmt[4];        // "fmt "
        uint32_t fmtSize;       // 16
        uint16_t audioFormat;   // 1
        uint16_t numChannels;   // 1
        uint32_t sampleRate;    // 16000
        uint32_t byteRate;      // 32000
        uint16_t blockAlign;    // 2
        uint16_t bitsPerSample; // 16
        char     data[4];       // "data"
        uint32_t dataSize;      // 160000
    } WAVHeader_t;

    uint8_t* voice_wav_encode(const uint8_t* pcm, size_t pcmLen) {
        size_t total = 44 + pcmLen;
        uint8_t* wav = (uint8_t*)heap_caps_malloc(total, MALLOC_CAP_SPIRAM);
        if (!wav) return nullptr;

        WAVHeader_t* h = (WAVHeader_t*)wav;
        memcpy(h->riff, "RIFF", 4);
        h->fileSize = total - 8;
        memcpy(h->wave, "WAVE", 4);
        memcpy(h->fmt,  "fmt ", 4);
        h->fmtSize = 16; h->audioFormat = 1;
        h->numChannels = 1; h->sampleRate = 16000;
        h->byteRate = 32000; h->blockAlign = 2;
        h->bitsPerSample = 16;
        memcpy(h->data, "data", 4);
        h->dataSize = pcmLen;
        memcpy(wav + 44, pcm, pcmLen);
        return wav;
    }

### 5.4 Gemini audio (P_Voice_GeminiAudio_0203.cpp)

    String voice_send_to_gemini(const uint8_t* wav, size_t wavLen) {
        // Base64 인코딩 (PSRAM)
        size_t b64Len = 4 * ((wavLen + 2) / 3) + 1;
        char* b64 = (char*)heap_caps_malloc(b64Len, MALLOC_CAP_SPIRAM);
        if (!b64) return "";
        _base64_encode(wav, wavLen, b64, b64Len);

        // 요청 Body
        JsonDocument req;
        req["system_instruction"]["parts"][0]["text"] =
            CT10_buildAIPromptStrict("ko", true);

        req["contents"][0]["parts"][0]["text"] =
            "Interpret this audio as a fan control command.";
        req["contents"][0]["parts"][1]["inline_data"]["mime_type"] = "audio/wav";
        req["contents"][0]["parts"][1]["inline_data"]["data"]      = b64;

        // Function Declarations (12개 재사용)
        JsonArray tools = req["tools"].to<JsonArray>();
        JsonArray decls = tools.add<JsonObject>()["function_declarations"].to<JsonArray>();
        CT10_appendAIFunctionDeclarations(decls);

        // HTTPS POST
        String body;
        serializeJson(req, body);
        free(b64);

        String resp = _gemini_post(body, apiKey);
        return resp;
    }

### 5.5 TTS 재생 (P_Voice_TTS_0203.cpp)

    // 옵션 A: Google Cloud TTS (권장)
    bool voice_tts_fetch(const char* text, uint8_t** outPcm, size_t* outLen) {
        String url = "https://texttospeech.googleapis.com/v1/text:synthesize";
        JsonDocument req;
        req["input"]["text"] = text;
        req["voice"]["languageCode"] = "ko-KR";
        req["voice"]["name"]         = "ko-KR-Neural2-A";
        req["audioConfig"]["audioEncoding"]   = "LINEAR16";
        req["audioConfig"]["sampleRateHertz"] = 24000;

        String body; serializeJson(req, body);
        String resp = _http_post(url, body, "x-goog-api-key", cloudApiKey);

        // 응답: { audioContent: "base64 PCM" }
        JsonDocument respDoc;
        deserializeJson(respDoc, resp);
        const char* b64 = respDoc["audioContent"] | "";
        size_t b64Len = strlen(b64);
        size_t pcmLen = 3 * b64Len / 4;

        uint8_t* pcm = (uint8_t*)heap_caps_malloc(pcmLen, MALLOC_CAP_SPIRAM);
        _base64_decode(b64, b64Len, pcm, pcmLen);

        *outPcm = pcm;
        *outLen = pcmLen;
        return true;
    }

    // 옵션 B: Gemini TTS (별도 API, 실험적)
    // (Phase 9C에서 검토)

    // 재생 (I2S TX)
    void voice_play_pcm(const uint8_t* pcm, size_t len) {

### 5.6 상태머신 (P_Voice_Main_0203.cpp)

    static EN_VOICE_state_t s_state = EN_VOICE_IDLE;

    void voice_fsm_post_event(EN_VOICE_event_t evt) {
        switch (s_state) {
            case EN_VOICE_IDLE:
                if (evt == EN_VOICE_EVT_WAKE) {
                    voice_led_green();
                    voice_beep(1000, 100);
                    s_state = EN_VOICE_WAKE_DETECTED;
                }
                break;
            case EN_VOICE_WAKE_DETECTED:
                // 300ms 후 녹음 시작
                s_state = EN_VOICE_RECORDING;
                break;
            // ... 이하 상태
        }
    }

    void voice_fsm_tick() {
        switch (s_state) {
            case EN_VOICE_RECORDING:
                voice_led_green_blink();
                if (voice_record_5s()) {
                    s_state = EN_VOICE_UPLOADING;
                }
                break;
            case EN_VOICE_UPLOADING:
                voice_led_yellow();
                if (voice_upload_process()) {
                    s_state = EN_VOICE_PROCESSING;
                }
                break;
            // ...
        }
    }
