```cpp
// ======================================================
// [신규 P0] API Key 검사 + 실패 시 401 즉시 응답
//  기존 checkApiKey는 false 반환 후 호출부가 return만 하면
//  HTTP 응답이 전송되지 않아 클라이언트가 timeout까지 hang.

// ======================================================
bool CL_W10_WebAPI::requireApiKey(AsyncWebServerRequest* p_request) {
    if (checkApiKey(p_request)) return true;
    p_request->send(401, "application/json; charset=utf-8",
                    "{\"error\":\"unauthorized\",\"hint\":\"X-API-Key header required\"}");
    return false;
}
```

이유: AI 엔드포인트 7곳 + 기타 라우트에서 checkApiKey → return 패턴이 클라이언트 hang → 서버 소켓 고갈 유발.

```cpp
bool CL_W10_WebAPI::parseJsonBody(AsyncWebServerRequest* p_request, uint8_t* p_data, size_t p_len, JsonDocument& p_doc) {
    // [신규] null/empty 방어 (ArduinoJson 내부 strnlen(nullptr) 크래시 방지)
    if (!p_data || p_len == 0) {
        CL_D10_Logger::log(EN_L10_LOG_WARN, "[W10] parseJsonBody: empty body");
        return false;
    }

    // [신규] 크기 한도 (DoS 방지) — 일반 body 1MB (schedules+profiles+windDict 여유)
    static constexpr size_t W10_MAX_GENERIC_BODY = 1024 * 1024;
    if (p_len > W10_MAX_GENERIC_BODY) {
        CL_D10_Logger::log(EN_L10_LOG_WARN,
                           "[W10] parseJsonBody: body too large (%u > %u)",
                           (unsigned)p_len, (unsigned)W10_MAX_GENERIC_BODY);
        return false;
    }

    // [신규] contentLength 사전 검사 (버퍼링 이전 조기 차단)
    if (p_request) {
        const size_t v_cl = p_request->contentLength();
        if (v_cl > 0 && v_cl > W10_MAX_GENERIC_BODY) {
            CL_D10_Logger::log(EN_L10_LOG_WARN,
                               "[W10] parseJsonBody: contentLength %u exceeds limit",
                               (unsigned)v_cl);
            return false;
        }
    }

    auto v_err = deserializeJson(p_doc, (const char*)p_data, p_len);
    if (v_err) {
        CL_D10_Logger::log(EN_L10_LOG_WARN, "[W10] JSON parse error: %s", v_err.c_str());
        return false;
    }
    return true;
}
```

이유: null 크래시 + 1MB 초과 힙 고갈 DoS 방지.

```cpp
// ======================================================
// [신규 P0] 단편화 body 누적 헬퍼
//  문제: onBody 콜백은 p_data가 각 TCP 조각의 포인터.
//        p_index+p_len==p_total에서 처리하면 "마지막 조각만" 파싱 → 잘린 JSON.
//  해결: _tempObject에 누적 버퍼를 할당, 각 조각을 index 오프셋에 복사.
//  사용 패턴:
//    onBody(r, data, len, index, total) {
//        if (index == 0) beginBody(r, total);
//        if (!appendBody(r, data, len, index, total)) return;  // 미완결
//        auto* acc = takeBody(r);  // (uint8_t*)*buf, *len
//        ... deserializeJson(doc, buf, len) ...
//        freeBody(r);
//    }
// ======================================================
namespace {
    struct ST_W10_BodyAccum { uint8_t* buf; size_t cap; size_t written; };
}

bool CL_W10_WebAPI::beginBody(AsyncWebServerRequest* r, size_t total) {
    if (!r || total == 0) return false;
    auto* acc = new (std::nothrow) ST_W10_BodyAccum{nullptr, total, 0};
    if (!acc) return false;
    acc->buf = (uint8_t*)malloc(total);
    if (!acc->buf) { delete acc; return false; }
    r->_tempObject = acc;
    return true;
}

bool CL_W10_WebAPI::appendBody(AsyncWebServerRequest* r, uint8_t* data, size_t len, size_t index, size_t total) {
    auto* acc = (ST_W10_BodyAccum*)r->_tempObject;
    if (!acc || !acc->buf) return false;
    if (index + len > acc->cap) return false;
    memcpy(acc->buf + index, data, len);
    acc->written = index + len;
    return (acc->written >= total);
}

uint8_t* CL_W10_WebAPI::takeBodyBuf(AsyncWebServerRequest* r, size_t& r_len) {
    auto* acc = (ST_W10_BodyAccum*)r->_tempObject;
    if (!acc || !acc->buf) { r_len = 0; return nullptr; }
    r_len = acc->written;
    return acc->buf;
}

void CL_W10_WebAPI::freeBody(AsyncWebServerRequest* r) {
    auto* acc = (ST_W10_BodyAccum*)r->_tempObject;
    if (!acc) return;
    if (acc->buf) free(acc->buf);
    delete acc;
    r->_tempObject = nullptr;
}
```

W10_Web_0203.h에 선언 추가 (public 섹션):

```cpp
// [신규 P0] 접미사 정확 매칭 (strstr 오탐지 방지: "/config.json" → ".js" 매치 버그)
static bool W10_pathEndsWith(const char* p_path, const char* p_suffix) {
    if (!p_path || !p_suffix) return false;
    const size_t v_ls = strlen(p_path);
    const size_t v_lf = strlen(p_suffix);
    if (v_lf == 0 || v_lf > v_ls) return false;
    return strcmp(p_path + (v_ls - v_lf), p_suffix) == 0;
}

const char* CL_W10_WebAPI::W10_guessMime(const char* p_path) {
    if (!p_path || p_path[0] == '\0') return "application/octet-stream";

    if (W10_pathEndsWith(p_path, ".html"))  return "text/html; charset=utf-8";
    if (W10_pathEndsWith(p_path, ".css"))   return "text/css; charset=utf-8";
    if (W10_pathEndsWith(p_path, ".json"))  return "application/json; charset=utf-8";
    if (W10_pathEndsWith(p_path, ".js"))    return "application/javascript; charset=utf-8";
    if (W10_pathEndsWith(p_path, ".svg"))   return "image/svg+xml";
    if (W10_pathEndsWith(p_path, ".png"))   return "image/png";
    if (W10_pathEndsWith(p_path, ".jpg") ||
        W10_pathEndsWith(p_path, ".jpeg"))  return "image/jpeg";
    if (W10_pathEndsWith(p_path, ".ico"))   return "image/x-icon";
    if (W10_pathEndsWith(p_path, ".woff2")) return "font/woff2";
    if (W10_pathEndsWith(p_path, ".woff"))  return "font/woff";
    if (W10_pathEndsWith(p_path, ".ttf"))   return "font/ttf";
    if (W10_pathEndsWith(p_path, ".txt"))   return "text/plain; charset=utf-8";
    if (W10_pathEndsWith(p_path, ".map"))   return "application/json; charset=utf-8";

    return "application/octet-stream";
}
```

```cpp
static bool W10_pathEndsWith(const char* p_path, const char* p_suffix) {
    if (!p_path || !p_suffix) return false;
    const size_t v_ls = strlen(p_path);
    const size_t v_lf = strlen(p_suffix);
    if (v_lf == 0 || v_lf > v_ls) return false;
```

```cpp
// [신규 P0] 단편화 body 누적 헬퍼
//  문제: AsyncWebServer onBody 콜백의 p_data는 "현재 TCP 조각"만
//        가리킨다. `if (p_index + p_len != p_total) return;`은
//        마지막 조각만 파싱하는 버그 → 4KB+ JSON은 반드시 실패.
//  해결: _tempObject에 전체 크기 버퍼를 할당하고 index 오프셋에
//        memcpy. 마지막 조각까지 다 도달하면 완결.
//  사용 예:
//    if (index == 0) beginBody(r, total);
//    if (!appendBody(r, data, len, index, total)) return;
//    size_t n; uint8_t* buf = takeBodyBuf(r, n);
//    ... deserializeJson(doc, buf, n) ...
//    freeBody(r);
// ======================================================
namespace {
    struct ST_W10_BodyAccum {
        uint8_t* buf;
        size_t   cap;
        size_t   written;
    };
}

bool CL_W10_WebAPI::beginBody(AsyncWebServerRequest* r, size_t total) {
    if (!r || total == 0) return false;
    if (r->_tempObject) return true;   // 이미 할당됨 (방어)
    auto* acc = new (std::nothrow) ST_W10_BodyAccum{nullptr, total, 0};
    if (!acc) return false;
    acc->buf = (uint8_t*)malloc(total);
    if (!acc->buf) { delete acc; return false; }
    r->_tempObject = acc;
    return true;
}

bool CL_W10_WebAPI::appendBody(AsyncWebServerRequest* r, uint8_t* data, size_t len,
                               size_t index, size_t total) {
    auto* acc = (ST_W10_BodyAccum*)r->_tempObject;
    if (!acc || !acc->buf) return false;
    if (index + len > acc->cap) return false;
    memcpy(acc->buf + index, data, len);
    acc->written = index + len;
    return (acc->written >= total);
}

uint8_t* CL_W10_WebAPI::takeBodyBuf(AsyncWebServerRequest* r, size_t& r_len) {
    auto* acc = (ST_W10_BodyAccum*)r->_tempObject;
    if (!acc || !acc->buf) { r_len = 0; return nullptr; }
    r_len = acc->written;
    return acc->buf;
}

void CL_W10_WebAPI::freeBody(AsyncWebServerRequest* r) {
    if (!r) return;
    auto* acc = (ST_W10_BodyAccum*)r->_tempObject;
    if (!acc) return;
    if (acc->buf) free(acc->buf);
    delete acc;
    r->_tempObject = nullptr;
}
```

```cpp
        [](AsyncWebServerRequest* p_request, uint8_t* p_data, size_t p_len,
           size_t p_index, size_t p_total) {
            if (!requireApiKey(p_request)) return;

            // [Stage 13] 첫 조각에서 버퍼 할당 + 크기 검사
            if (p_index == 0) {
                if (p_total == 0) {
                    p_request->send(400, "application/json", "{\"error\":\"empty body\"}");
                    return;
                }
                // 일반 body 한도 1MB (schedules + profiles + windDict 여유)
                if (p_total > (1024 * 1024)) {
                    p_request->send(413, "application/json", "{\"error\":\"payload too large\"}");
                    return;
                }
                if (!beginBody(p_request, p_total)) {
                    p_request->send(500, "application/json", "{\"error\":\"oom\"}");
                    return;
                }
            }
            if (!appendBody(p_request, p_data, p_len, p_index, p_total)) return;

            size_t   v_bodyLen = 0;
            uint8_t* v_bodyBuf = takeBodyBuf(p_request, v_bodyLen);
            if (!v_bodyBuf) {
                freeBody(p_request);
                p_request->send(500, "application/json", "{\"error\":\"acc\"}");
                return;
            }

            JsonDocument v_doc;
            if (!parseJsonBody(p_request, v_bodyBuf, v_bodyLen, v_doc)) {
                freeBody(p_request);
                p_request->send(400, "application/json", "{\"error\":\"json parse\"}");
                return;
            }
            freeBody(p_request);

            // [B-2] async_tcp → getRootSnapshot
            ST_A20_ConfigRoot_t v_snap;
            CL_C10_ConfigManager::getRootSnapshot(v_snap);

            bool v_changed = false;
            if (v_snap.userProfiles) {
                v_changed = CL_C10_ConfigManager::patchUserProfilesFromJson(*v_snap.userProfiles, v_doc);
            }
            if (v_changed) {
                CL_C10_ConfigManager::saveDirtyConfigs();
            }

            JsonDocument v_res;
            v_res["updated"] = v_changed;
            sendJson(p_request, v_res);
        });
}
```

문제
