2.5.1 /ws/ai 핸들러

```cpp
// W10_Web_AIStream_0203.cpp
static AsyncWebSocket s_wsAi("/ws/ai");

// 스트리밍 요청 큐
struct AIStreamRequest {
    uint32_t   clientId;
    String     message;
    JsonDocument history;
};

static QueueHandle_t s_aiStreamQueue = nullptr;

// WS 수신 핸들러 (async_tcp 태스크)
static void onWsAiEvent(AsyncWebSocket* server, AsyncWebSocketClient* client,
                        AwsEventType type, void* arg, uint8_t* data, size_t len) {
    if (type != WS_EVT_DATA) return;

    AwsFrameInfo* info = (AwsFrameInfo*)arg;
    if (!info->final || info->index != 0) return;

    JsonDocument req;
    if (deserializeJson(req, data, len) != DeserializationError::Ok) return;

    // 큐에 push (blocking 방지)
    AIStreamRequest* r = new AIStreamRequest;
    r->clientId = client->id();
    r->message  = A40_ComFunc::Json_getStr(req.as<JsonObjectConst>(), "message", "");
    r->history.set(req["history"]);

    if (xQueueSend(s_aiStreamQueue, &r, 0) != pdTRUE) {
        // 큐 풀: 에러 응답
        JsonDocument err;
        err["type"] = "error";
        err["code"] = "queue_full";
        String out;
        serializeJson(err, out);
        client->text(out);
        delete r;
    }
}

// 스트리밍 태스크 (Gemini SSE 수신)
static void aiStreamTask(void* pv) {
    AIStreamRequest* req = nullptr;
    if (xQueueReceive(s_aiStreamQueue, &req, portMAX_DELAY) != pdTRUE) {
        vTaskDelete(NULL);
        return;
    }

    AsyncWebSocketClient* client = s_wsAi.client(req->clientId);
    if (!client) {
        delete req;
        vTaskDelete(NULL);
        return;
    }

    String apiKey = g_A20_config_root.system->security.geminiApiKey;

    // Gemini streamGenerateContent
    String url = "https://generativelanguage.googleapis.com/v1beta/models/"
                 "gemini-2.0-flash-exp:streamGenerateContent?alt=sse";

    WiFiClientSecure tls;
    tls.setInsecure();
    HTTPClient http;
    http.begin(tls, url);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("x-goog-api-key", apiKey);
    http.setTimeout(20000);

    String body = _buildGeminiRequestBody(req->message, req->history);
    int httpCode = http.POST(body);

    if (httpCode != 200) {
        _sendWsError(client, "gemini_error", httpCode);
        http.end();
        delete req;
        vTaskDelete(NULL);
        return;
    }

    // SSE 파싱
    WiFiClient* stream = http.getStreamPtr();
    String buffer;
    String fullText;

    while (stream->connected() || stream->available()) {
        if (!stream->available()) {
            delay(10);
            continue;
        }

        char c = stream->read();
        buffer += c;

        if (buffer.endsWith("\r\n\r\n") || buffer.endsWith("\n\n")) {
            // SSE 이벤트 처리
            int dataPos = buffer.indexOf("data: ");
            if (dataPos >= 0) {
                String dataLine = buffer.substring(dataPos + 6);
                dataLine.trim();

                JsonDocument chunk;
                if (deserializeJson(chunk, dataLine) == DeserializationError::Ok) {
                    JsonArray cands = chunk["candidates"].as<JsonArray>();
                    if (!cands.isNull() && cands.size() > 0) {
                        JsonObject part = cands[0]["content"]["parts"][0];
                        const char* t = A40_ComFunc::Json_getStr(part, "text", "");

                        if (t && t[0]) {
                            fullText += t;
                            // WS push
                            JsonDocument push;
                            push["type"] = "chunk";
                            push["text"] = t;
                            String out;
                            serializeJson(push, out);
                            client->text(out);
                        }
                    }
                }
            }
            buffer = "";
        }

        // 오버플로 방지
        if (buffer.length() > 4096) buffer = buffer.substring(2048);

        // WDT feed
        if ((millis() % 1000) == 0) vTaskDelay(1);
    }

    http.end();

    // 액션 실행 (텍스트 응답 완료 후)
    // ⚠️ 2-turn은 별도 처리 필요 (8.1 참고)

    // 완료 신호
    JsonDocument done;
    done["type"] = "done";
    done["text"] = fullText;
    String outDone;
    serializeJson(done, outDone);
    client->text(outDone);

    delete req;
    vTaskDelete(NULL);
}
```

```javascript
// 145.ai.client.js 확장
async sendStreaming(message, history, { onChunk, onActions, onDone }) {
    const ws = SNW.WsManager.get('ai');
    if (!ws) throw new Error('AI WS unavailable');

    return new Promise((resolve, reject) => {
        let unsub = null;
        const timeout = setTimeout(() => {
            if (unsub) unsub();
            reject(new Error('timeout'));
        }, 30000);

        unsub = ws.onMessage((msg) => {
            if (msg.type === 'chunk')   onChunk(msg.text);
            if (msg.type === 'actions') onActions(msg.actions || []);
            if (msg.type === 'error')   { clearTimeout(timeout); unsub(); reject(new Error(msg.code)); }
            if (msg.type === 'done') {
                clearTimeout(timeout);
                unsub();
                onDone(msg.text);
                resolve(msg.text);
            }
        });

        ws.send(JSON.stringify({ message, history: (history || []).slice(-6) }));
    });
}
```

```cpp
// P_Voice_I2S_0203.cpp
#include <driver/i2s.h>

#define I2S_MIC_BCK   GPIO_NUM_4
#define I2S_MIC_WS    GPIO_NUM_5
#define I2S_MIC_DATA  GPIO_NUM_6

#define SAMPLE_RATE   16000
#define RECORD_SECONDS 5
#define SAMPLE_BYTES 2
#define BUFFER_SIZE (SAMPLE_RATE * SAMPLE_BYTES * RECORD_SECONDS)  // 160KB

void voice_i2s_mic_init() {
    i2s_config_t cfg = {
        .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
        .sample_rate = SAMPLE_RATE,
        .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
        .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
        .communication_format = I2S_COMM_FORMAT_STAND_I2S,
        .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
        .dma_buf_count = 8,
        .dma_buf_len = 1024,
        .use_apll = false,
    };
    i2s_driver_install(I2S_NUM_0, &cfg, 0, NULL);

    i2s_pin_config_t pins = {
        .bck_io_num   = I2S_MIC_BCK,
        .ws_io_num    = I2S_MIC_WS,
        .data_out_num = I2S_PIN_NO_CHANGE,
        .data_in_num  = I2S_MIC_DATA,
    };
    i2s_set_pin(I2S_NUM_0, &pins);
}

// PSRAM 버퍼에 녹음
uint8_t* voice_record_5s() {
    size_t bytesRead = 0;
    uint8_t* buf = (uint8_t*)heap_caps_malloc(BUFFER_SIZE, MALLOC_CAP_SPIRAM);
    if (!buf) return nullptr;

    size_t totalRead = 0;
    while (totalRead < BUFFER_SIZE) {
        i2s_read(I2S_NUM_0, buf + totalRead, BUFFER_SIZE - totalRead, &bytesRead, 1000);
        totalRead += bytesRead;
    }
    return buf;
}
```

```cpp
// P_Voice_WAV_0203.cpp
typedef struct {
    char     riff[4];       // "RIFF"
    uint32_t fileSize;      // 총 파일 크기 - 8
    char     wave[4];       // "WAVE"
    char     fmt[4];        // "fmt "
    uint32_t fmtSize;       // 16
    uint16_t audioFormat;   // 1 (PCM)
    uint16_t numChannels;   // 1 (mono)
    uint32_t sampleRate;    // 16000
    uint32_t byteRate;      // sampleRate * numChannels * 2
    uint16_t blockAlign;    // numChannels * 2
    uint16_t bitsPerSample; // 16
    char     data[4];       // "data"
    uint32_t dataSize;      // PCM bytes
} __attribute__((packed)) WAVHeader_t;

uint8_t* voice_wav_encode(const uint8_t* pcm, size_t pcmLen) {
    size_t totalSize = 44 + pcmLen;
    uint8_t* wav = (uint8_t*)heap_caps_malloc(totalSize, MALLOC_CAP_SPIRAM);
    if (!wav) return nullptr;

    WAVHeader_t* h = (WAVHeader_t*)wav;
    memcpy(h->riff, "RIFF", 4);
    h->fileSize = totalSize - 8;
    memcpy(h->wave, "WAVE", 4);
    memcpy(h->fmt, "fmt ", 4);
    h->fmtSize = 16;
    h->audioFormat = 1;
    h->numChannels = 1;
    h->sampleRate = 16000;
    h->byteRate = 16000 * 1 * 2;
    h->blockAlign = 2;
    h->bitsPerSample = 16;
    memcpy(h->data, "data", 4);
    h->dataSize = pcmLen;

    memcpy(wav + 44, pcm, pcmLen);
    return wav;
}
```

3.1.3 백엔드 — Ephemeral Token 발급

W10_Web_Routes_AILive_0203.cpp

```cpp
#include "W10_Web_0203.h"
#include "A40_Com_Utils_0203.h"
#include <HTTPClient.h>
#include <WiFiClientSecure.h>
#include <ArduinoJson.h>

// ======================================================
// POST /api/v0203/ai/live-token
// - Gemini Ephemeral Token 발급 (Auth API)
// - 응답: { token, expiresAt, model }
// ======================================================
static void handleLiveToken(AsyncWebServerRequest* req) {
    if (!W10_checkApiKey(req)) { req->send(401); return; }

    if (!g_A20_config_root.system ||
        !g_A20_config_root.system->security.geminiApiKey[0]) {
        req->send(400, "application/json",
            "{\"error\":\"gemini_api_key_not_set\"}");
        return;
    }

    const char* apiKey = g_A20_config_root.system->security.geminiApiKey;

    // Gemini Auth Tokens API (Ephemeral)
    String url = "https://generativelanguage.googleapis.com/v1alpha/auth_tokens";

    JsonDocument authReq;
    authReq["uses"]  = 1;   // 1회용 (Live 세션 1회)
    authReq["expireTime"] = "";   // 기본 30분

    String body;
    serializeJson(authReq, body);

    WiFiClientSecure tls;
    tls.setInsecure();

    HTTPClient http;
    http.begin(tls, url);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("x-goog-api-key", apiKey);
    http.setTimeout(8000);

    int httpCode = http.POST(body);
    String resp = http.getString();
    http.end();

    if (httpCode != 200) {
        JsonDocument err;
        err["error"] = "token_mint_failed";
        err["httpCode"] = httpCode;
        err["detail"] = resp.substring(0, 200);
        W10_sendJson(req, 502, err);
        return;
    }

    // 응답 파싱
    JsonDocument tokenResp;
    DeserializationError e = deserializeJson(tokenResp, resp);
    if (e) {
        req->send(502, "application/json", "{\"error\":\"parse_failed\"}");
        return;
    }

    JsonDocument out;
    out["token"]  = tokenResp["name"];   // 실제 token string
    out["expire"] = tokenResp["expireTime"];
    out["model"]  = "gemini-2.0-flash-exp";
    W10_sendJson(req, 200, out);
}

// ======================================================
// POST /api/v0203/ai/live-function
// - 브라우저가 Live tool_call을 위임
// - 기존 CT10_executeAIAction 재사용
// ======================================================
static void handleLiveFunction(AsyncWebServerRequest* req) {
    if (!W10_checkApiKey(req)) { req->send(401); return; }

    JsonDocument in;
    uint8_t buf[2048];
```

```javascript
// 147a.audio.capture.js — 마이크 PCM 캡처 (16kHz, AEC)

window.SNW = window.SNW || {};
(function(SNW) {
    'use strict';

    const SAMPLE_RATE = 16000;
    const CHUNK_MS = 100;                        // 100ms 청크
    const CHUNK_SAMPLES = SAMPLE_RATE * CHUNK_MS / 1000;   // 1600
    const CHUNK_BYTES = CHUNK_SAMPLES * 2;       // 3200 bytes

    SNW.AudioCapture = {
        _stream: null,
        _audioCtx: null,
        _workletNode: null,
        _sourceNode: null,

        async start(onChunk) {
            // ★ AEC 필수 — 스피커 출력이 마이크로 재유입되면 barge-in 오탐
            this._stream = await navigator.mediaDevices.getUserMedia({
                audio: {
                    echoCancellation: true,       // ★
                    noiseSuppression: true,
                    autoGainControl:    true,
                    sampleRate:         SAMPLE_RATE,
                    channelCount:       1,
                },
            });

            this._audioCtx = new AudioContext({ sampleRate: SAMPLE_RATE });
            await this._audioCtx.audioWorklet.addModule('./js/147a.worklet.js');

            this._sourceNode  = this._audioCtx.createMediaStreamSource(this._stream);
            this._workletNode = new AudioWorkletNode(this._audioCtx, 'pcm-processor');

            this._workletNode.port.onmessage = (e) => {
                const float32 = e.data;   // Float32Array (1600 samples)
                const pcm16 = this._float32ToPCM16(float32);
                onChunk(pcm16);   // 3200 bytes (Int16Array)
            };

            this._sourceNode.connect(this._workletNode);
            this._workletNode.connect(this._audioCtx.destination);
        },

        stop() {
            if (this._stream) {
                this._stream.getTracks().forEach(t => t.stop());
                this._stream = null;
            }
            if (this._audioCtx) {
                this._audioCtx.close();
                this._audioCtx = null;
            }
        },

        _float32ToPCM16(float32) {
            const pcm = new Int16Array(float32.length);
            for (let i = 0; i < float32.length; i++) {
                const s = Math.max(-1, Math.min(1, float32[i]));
                pcm[i] = s < 0 ? s * 0x8000 : s * 0x7FFF;
            }
            return pcm;
        },
    };
})(window.SNW);
```

```javascript
// =========================================================
// 147a.worklet.js — AudioWorklet (100ms 청크 생성)
// =========================================================
class PCMProcessor extends AudioWorkletProcessor {
    constructor() {
        super();
        this._bufferSize = 1600;   // 100ms @ 16kHz
        this._buffer = new Float32Array(this._bufferSize);
        this._idx = 0;
    }

    process(inputs) {
        const input = inputs[0][0];
        if (!input) return true;

        for (let i = 0; i < input.length; i++) {
            this._buffer[this._idx++] = input[i];
            if (this._idx >= this._bufferSize) {
                // 청크 완성 → 메인 스레드로 전송
                const chunk = new Float32Array(this._buffer);
                this.port.postMessage(chunk);
                this._idx = 0;
            }
        }
        return true;
    }
}

registerProcessor('pcm-processor', PCMProcessor);
```

```javascript
// =========================================================
// 147b.audio.playback.js — PCM 재생 (24kHz, 버퍼 큐)
// =========================================================
window.SNW = window.SNW || {};
(function(SNW) {
    'use strict';

    const OUTPUT_RATE = 24000;

    SNW.AudioPlayback = {
        _audioCtx: null,
        _queue: [],          // 재생 대기 청크
        _playing: false,
        _nextStartMs: 0,

        start() {
            if (this._audioCtx) return;
            this._audioCtx = new AudioContext({ sampleRate: OUTPUT_RATE });
        },

        // Gemini에서 받은 PCM 청크 (24kHz, Int16 → Base64 디코딩)
        push(pcmBytes) {   // Uint8Array
            const pcm16 = new Int16Array(pcmBytes.buffer);
            const float32 = new Float32Array(pcm16.length);
            for (let i = 0; i < pcm16.length; i++) {
                float32[i] = pcm16[i] / 32768.0;
            }

            this._queue.push(float32);
            if (!this._playing) this._playNext();
        },

        _playNext() {
            if (this._queue.length === 0) {
                this._playing = false;
                return;
            }
            this._playing = true;

            const chunk = this._queue.shift();
            const buf = this._audioCtx.createBuffer(1, chunk.length, OUTPUT_RATE);
            buf.copyToChannel(chunk, 0);

            const src = this._audioCtx.createBufferSource();
            src.buffer = buf;
            src.connect(this._audioCtx.destination);

            const now = this._audioCtx.currentTime;
            const startAt = Math.max(now, this._nextStartMs);
            src.start(startAt);
            this._nextStartMs = startAt + buf.duration;

            src.onended = () => this._playNext();
        },

        // ★ Barge-in 시 즉시 정지
        flush() {
            this._queue = [];
            this._playing = false;
            this._nextStartMs = 0;
            if (this._audioCtx) {
                // 현재 재생 중인 노드들 중단 (모든 source.stop)
                // ⚠️ 실제로는 각 source를 추적해서 stop 필요
                // Phase 2에서는 _sources 배열로 관리
            }
        },

        stop() {
            this.flush();
            if (this._audioCtx) {
                this._audioCtx.close();
                this._audioCtx = null;
            }
        },
    };
})(window.SNW);
```
