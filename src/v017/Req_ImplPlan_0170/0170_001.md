A-3. docs/contracts/tools/commit_contracts.sh

```bash
#!/usr/bin/env bash
# ==============================================================
# docs/contracts 커밋 스크립트 (Linux / macOS / Git-Bash)
#
# 기능:
#   1. 저장소 루트 자동 감지 (git rev-parse --show-toplevel)
#   2. docs/contracts/ 파일 존재 검증 (SPEC/CONTRACT/STATE/FLOW/BUDGET/TEST_PLAN/README)
#   3. 각 문서를 staging 후 단일 커밋
#   4. 커밋 후 요약 출력
#
# 사용:
#   bash docs/contracts/tools/commit_contracts.sh
#   bash docs/contracts/tools/commit_contracts.sh "docs(contracts): 커스텀 메시지"
#
# 종료 코드:
#   0 성공, 1 파일 누락, 2 git 오류
# ==============================================================

set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
CONTRACTS_DIR="$(cd "${SCRIPT_DIR}/.." && pwd)"
REPO_ROOT="$(git -C "${CONTRACTS_DIR}" rev-parse --show-toplevel 2>/dev/null || true)"

if [[ -z "${REPO_ROOT}" ]]; then
    echo "[ERROR] git 저장소 루트를 찾을 수 없습니다." >&2
    exit 2
fi

cd "${REPO_ROOT}"

# ---- 커밋 메시지 (인자 없으면 기본값) -------------------------
DEFAULT_MSG="docs(contracts): 계약 문서 갱신 (F-01~F-06, I-1~I-4 반영)"
COMMIT_MSG="${1:-${DEFAULT_MSG}}"

# ---- 필수 파일 검증 ------------------------------------------
REQUIRED_FILES=(
    "docs/contracts/README.md"
    "docs/contracts/SPEC.md"
    "docs/contracts/CONTRACT.md"
    "docs/contracts/STATE.md"
    "docs/contracts/FLOW.md"
    "docs/contracts/BUDGET.md"
    "docs/contracts/TEST_PLAN.md"
)

MISSING=()
for f in "${REQUIRED_FILES[@]}"; do
    [[ -f "${f}" ]] || MISSING+=("${f}")
done

if [[ ${#MISSING[@]} -gt 0 ]]; then
    echo "[ERROR] 필수 문서 누락:" >&2
    for f in "${MISSING[@]}"; do echo "  - ${f}" >&2; done
    exit 1
fi

echo "============================================================"
echo " 커밋 대상: docs/contracts/"
echo " 저장소   : ${REPO_ROOT}"
echo " 메시지   : ${COMMIT_MSG}"
echo "============================================================"

# ---- 현재 git 상태 요약 --------------------------------------
echo ""
echo "[1/4] 변경 사항 확인"
git status --short docs/contracts/ || true

# ---- staging ------------------------------------------------
echo ""
echo "[2/4] staging"
git add docs/contracts/

# ---- staged 파일 목록 ---------------------------------------
STAGED=$(git diff --cached --name-only -- docs/contracts/ || true)
if [[ -z "${STAGED}" ]]; then
    echo ""
    echo "[INFO] staging된 변경 사항이 없습니다. 커밋을 생략합니다."
    exit 0
fi

echo ""
echo "[3/4] staged 파일:"
echo "${STAGED}" | sed 's/^/  + /'

# ---- 커밋 ---------------------------------------------------
echo ""
echo "[4/4] 커밋"
if git commit -m "${COMMIT_MSG}"; then
    echo ""
    echo "============================================================"
    echo " ✅ 커밋 완료"
    echo "============================================================"
    git log -1 --pretty=format:"  %h %s%n  author: %an <%ae>%n  date  : %ad" --date=iso
    echo ""
    echo ""
    echo " 다음 단계:"
    echo "   git push origin $(git rev-parse --abbrev-ref HEAD)"
else
    echo "[ERROR] 커밋 실패" >&2
    exit 2
fi
```

A-4. docs/contracts/tools/commit_contracts.ps1

```powershell
<#
.SYNOPSIS
    docs/contracts 커밋 스크립트 (Windows PowerShell)

.DESCRIPTION
    1. 저장소 루트 자동 감지 (git rev-parse --show-toplevel)
    2. docs/contracts/ 파일 존재 검증
    3. staging 후 단일 커밋
    4. 커밋 요약 출력

.PARAMETER Message
    커밋 메시지 (지정하지 않으면 기본 메시지 사용)

.EXAMPLE
    .\docs\contracts\tools\commit_contracts.ps1
    .\docs\contracts\tools\commit_contracts.ps1 -Message "docs(contracts): F-05 reload 위임 반영"

.NOTES
    실행 정책 오류 시:
        powershell -ExecutionPolicy Bypass -File .\docs\contracts\tools\commit_contracts.ps1
#>

[CmdletBinding()]
param(
    [string]$Message = "docs(contracts): 계약 문서 갱신 (F-01~F-06, I-1~I-4 반영)"
)

$ErrorActionPreference = 'Stop'

# ---- 저장소 루트 감지 ---------------------------------------
try {
    $repoRoot = (& git -C $PSScriptRoot rev-parse --show-toplevel 2>$null)
    if (-not $repoRoot) { throw "저장소 루트 없음" }
    $repoRoot = $repoRoot.Trim()
} catch {
    Write-Error "[ERROR] git 저장소 루트를 찾을 수 없습니다."
    exit 2
}

Push-Location $repoRoot

try {
    # ---- 필수 파일 검증 -------------------------------------
    $requiredFiles = @(
        "docs\contracts\README.md",
        "docs\contracts\SPEC.md",
        "docs\contracts\CONTRACT.md",
        "docs\contracts\STATE.md",
        "docs\contracts\FLOW.md",
        "docs\contracts\BUDGET.md",
        "docs\contracts\TEST_PLAN.md"
    )

    $missing = @()
    foreach ($f in $requiredFiles) {
        if (-not (Test-Path -LiteralPath $f -PathType Leaf)) {
            $missing += $f
        }
    }

    if ($missing.Count -gt 0) {
        Write-Host "[ERROR] 필수 문서 누락:" -ForegroundColor Red
        foreach ($f in $missing) { Write-Host "  - $f" -ForegroundColor Red }
        exit 1
    }

    Write-Host "============================================================"
    Write-Host " 커밋 대상: docs/contracts/"
    Write-Host " 저장소   : $repoRoot"
    Write-Host " 메시지   : $Message"
    Write-Host "============================================================"

    # ---- 변경 사항 확인 -------------------------------------
    Write-Host ""
    Write-Host "[1/4] 변경 사항 확인"
    & git status --short docs/contracts/

    # ---- staging --------------------------------------------
    Write-Host ""
    Write-Host "[2/4] staging"
    & git add docs/contracts/

    # ---- staged 파일 목록 -----------------------------------
    $staged = & git diff --cached --name-only -- docs/contracts/
    if (-not $staged) {
        Write-Host ""
        Write-Host "[INFO] staging된 변경 사항이 없습니다. 커밋을 생략합니다." -ForegroundColor Yellow
        exit 0
    }

    Write-Host ""
    Write-Host "[3/4] staged 파일:"
    $staged | ForEach-Object { Write-Host "  + $_" }

    # ---- 커밋 -----------------------------------------------
    Write-Host ""
    Write-Host "[4/4] 커밋"
    & git commit -m $Message
    if ($LASTEXITCODE -ne 0) {
        Write-Host "[ERROR] 커밋 실패" -ForegroundColor Red
        exit 2
    }

    Write-Host ""
    Write-Host "============================================================"
    Write-Host " ✅ 커밋 완료" -ForegroundColor Green
    Write-Host "============================================================"
    & git log -1 --pretty=format:"  %h %s%n  author: %an <%ae>%n  date  : %ad" --date=iso
    Write-Host ""
    Write-Host ""
    $branch = (& git rev-parse --abbrev-ref HEAD).Trim()
    Write-Host " 다음 단계:"
    Write-Host "   git push origin $branch"
}
finally {
    Pop-Location
}
```

```markdown
## 5. 락 순서(Lock Ordering) — 데드락 방지
```
CT10::s_stateMutex (최상위, T-loop)
  └→ S10::_recursiveMutex      (CT10 tick → sim.tick)
  └→ C10::s_rootSwapMux        (portMUX, 짧게)
  └→ C10::s_recursiveMutex     (재진입 경유, 짧게)

C10::s_recursiveMutex (독립 경로, T-async 직진입 가능)

```
## 8. PIR 피드 흐름 — F-03 반영
```
[1] T-async POST /motion/pir/feed
      → M10::feedPIR(pir)  [mutex]
        → notifyPIRDetected: _pir.active=true, _pir.lastDetected_ms=millis
[2] T-loop A00_run → g_M10_motionLogic->tick()  ← F-03
      [mutex]
      → holdSec 초과: _pir.active=false
      → 상태 변화 시 _state.pirActive/lastChange_ms 갱신
      → _onChange 콜백 (있으면)
[3] T-loop CT10::decideRunSource()
      → isMotionBlocked() → motion->isActive()
        isActive: _state.pirActive || (millis - lastChange_ms < holdSec*1000)
```
**F-03 후**: 부팅 후 `holdSec` 동안 false-positive (F-09) 자연 해소 — `_state.lastChange_ms`가 tick마다 갱신.

## 9. Motion patch 흐름 — F-06 반영 (async_tcp → loopTask 위임)
```
