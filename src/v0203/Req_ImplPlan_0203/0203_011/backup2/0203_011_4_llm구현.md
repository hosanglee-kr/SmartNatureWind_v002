```cpp
/*
 * ------------------------------------------------------
 * 소스명 : W10_Web_Routes_AILive_0203.cpp
 * 모듈약어 : W10
 * 모듈명 : Gemini Live API 브리지 (Ephemeral Token + Function Bridge)
 * ------------------------------------------------------
 * 아키텍처:
 *  - Browser = Voice Terminal (Live WS 직접 연결, 오디오는 ESP32 미경유)
 *  - ESP32   = Command Executor (Token 발급 + Function Call 실행)
 *
 * 엔드포인트:
 *  POST /api/v0203/ai/live-token      : Ephemeral Token 발급
 *  POST /api/v0203/ai/live-function   : Function Call 실행 위임
 *  GET  /api/v0203/ai/live-config     : Setup 파라미터 (프롬프트 + tools)
 * ------------------------------------------------------
 */

#include "W10_Web_0203.h"
#include "CT10_Ctl_0203.h"
#include "A40_Com_Utils_0203.h"
#include <HTTPClient.h>
#include <WiFiClientSecure.h>

// ======================================================
// 1) POST /api/v0203/ai/live-token
//    - Gemini Ephemeral Token 발급
//    - 응답: { token, expire, wsUrl }
// ======================================================
static void handleLiveToken(AsyncWebServerRequest* req) {
    if (!W10_checkApiKey(req)) { req->send(401); return; }

    if (!g_A20_config_root.system ||
        !g_A20_config_root.system->security.geminiApiKey[0]) {
        JsonDocument err;
        err["error"] = "gemini_api_key_not_set";
        err["message"] = "Gemini API Key를 설정에 입력하세요";
        W10_sendJson(req, 400, err);
        return;
    }

    const char* apiKey = g_A20_config_root.system->security.geminiApiKey;

    // Gemini Auth Tokens API
    String url = "https://generativelanguage.googleapis.com/v1alpha/auth_tokens";

    JsonDocument authReq;
    authReq["uses"] = 1;   // 1회용 (Live 세션 1회)

    String body;
    serializeJson(authReq, body);

    WiFiClientSecure tls;
    tls.setInsecure();
    tls.setTimeout(10000);

    HTTPClient http;
    http.begin(tls, url);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("x-goog-api-key", apiKey);
    http.setTimeout(10000);

    int httpCode = http.POST(body);
    String resp = http.getString();
    http.end();

    CL_D10_Logger::log(EN_L10_LOG_INFO,
        "[AI Live] token mint: http=%d len=%u",
        httpCode, resp.length());

    if (httpCode != 200) {
        JsonDocument err;
        err["error"] = "token_mint_failed";
        err["httpCode"] = httpCode;
        err["detail"] = resp.substring(0, 200);
        W10_sendJson(req, 502, err);
        return;
    }

    // 응답 파싱
    // 실제 응답 예: { "name": "authtokens/xxx...", "expireTime": "...", "usesRemaining": 1 }
    JsonDocument tokenResp;
    DeserializationError errParse = deserializeJson(tokenResp, resp);
    if (errParse) {
        JsonDocument err;
        err["error"] = "parse_failed";
        err["detail"] = errParse.c_str();
        W10_sendJson(req, 502, err);
        return;
    }

    const char* tokenName = tokenResp["name"] | "";
    const char* expireTime = tokenResp["expireTime"] | "";

    if (!tokenName[0]) {
        JsonDocument err;
        err["error"] = "empty_token";
        err["detail"] = resp.substring(0, 100);
        W10_sendJson(req, 502, err);
        return;
    }

    JsonDocument out;
    out["token"]   = tokenName;
    out["expire"]  = expireTime;
    out["model"]   = "gemini-2.0-flash-exp";
    out["wsUrl"]   = "wss://generativelanguage.googleapis.com/ws/"
                     "google.ai.generativelanguage.v1beta.GenerativeService."
                     "BidiGenerateContent";

    W10_sendJson(req, 200, out);

    CL_D10_Logger::log(EN_L10_LOG_INFO,
        "[AI Live] token issued (expire=%s)", expireTime);
}

// ======================================================
// 2) POST /api/v0203/ai/live-function
//    - 브라우저가 Live toolCall을 위임
//    - 8.1의 CT10_executeAIAction 재사용
// ======================================================
static void handleLiveFunction(AsyncWebServerRequest* req) {
    if (!W10_checkApiKey(req)) { req->send(401); return; }

    JsonDocument in;
    uint8_t buf[2048];
    if (!W10_parseJsonBody(req, in, buf, sizeof(buf))) {
        req->send(400, "application/json", "{\"error\":\"invalid_body\"}");
        return;
    }

    const char* fnName = A40_ComFunc::Json_getStr(
        in.as<JsonObjectConst>(), "name", "");
    JsonObjectConst args = in["args"].as<JsonObjectConst>();

    if (!fnName[0]) {
        req->send(400, "application/json", "{\"error\":\"missing_name\"}");
        return;
    }

    CL_D10_Logger::log(EN_L10_LOG_INFO,
        "[AI Live] fn=%s", fnName);

    // 8.1의 액션 실행 재사용
    String result = CT10_executeAIAction(fnName, args);

    JsonDocument out;
    out["ok"]     = true;
    out["name"]   = fnName;
    out["result"] = result;
    W10_sendJson(req, 200, out);
}

// ======================================================
// 3) GET /api/v0203/ai/live-config
//    - Live setup에 필요한 systemInstruction + tools
//    - 브라우저가 setup 메시지 조립에 사용
// ======================================================
static void handleLiveConfig(AsyncWebServerRequest* req) {
    if (!W10_checkApiKey(req)) { req->send(401); return; }

    JsonDocument out;

    // system_instruction
    String systemPrompt = CT10_buildAIPrompt();
    out["systemInstruction"]["parts"][0]["text"] = systemPrompt;

    // function declarations (10개)
    JsonArray decls = out["tools"].to<JsonArray>()
        .add<JsonObject>()["function_declarations"].to<JsonArray>();
    CT10_appendAIFunctionDeclarations(decls);

    // 기본 voice
    out["voiceName"] = "Puck";

    W10_sendJson(req, 200, out);

    CL_D10_Logger::log(EN_L10_LOG_INFO,
        "[AI Live] config served (prompt=%u chars, %u funcs)",
        systemPrompt.length(), decls.size());
}

// ======================================================
// 등록
// ======================================================
void CL_W10_WebAPI::registerAILiveRoutes(AsyncWebServer& server) {
    server.on("/api/v0203/ai/live-token",    HTTP_POST, handleLiveToken);
    server.on("/api/v0203/ai/live-function", HTTP_POST, handleLiveFunction);
    server.on("/api/v0203/ai/live-config",   HTTP_GET,  handleLiveConfig);
}
```

2.3 147.ai.live.js — Live 클라이언트 (Phase 1 완성) ★

```javascript
// =========================================================
// 147.ai.live.js — Gemini Live API 클라이언트
// Phase 1: 텍스트 + Function Call (오디오는 8.3A-2)
// =========================================================
window.SNW = window.SNW || {};
(function(SNW) {
    'use strict';

    SNW.AILive = {
        // =========================================================
        // 상태
        // =========================================================
        _ws: null,
        _token: null,
        _config: null,          // systemInstruction + tools (캐시)
        _state: 'idle',         // idle | connecting | ready | error | closed
        _resolveConnect: null,
        _rejectConnect: null,

        // 세션 타이머 (8.3C)
        _sessionStartMs: 0,
        _sessionTimer: null,
        _warnTimer: null,

        // 이벤트 핸들러 (외부 주입)
        _onTextChunk: null,
        _onAction: null,
        _onError: null,
        _onClose: null,

        // =========================================================
        // 시작
        // =========================================================
        async start(opts = {}) {
            if (this._state === 'ready') return true;
            if (this._state === 'connecting') {
                return new Promise((resolve, reject) => {
                    this._resolveConnect = resolve;
                    this._rejectConnect = reject;
                });
            }

            this._onTextChunk = opts.onTextChunk || null;
            this._onAction    = opts.onAction    || null;
            this._onError     = opts.onError     || null;
            this._onClose     = opts.onClose     || null;

            this._state = 'connecting';

            try {
                // 1) Live 설정 캐시 (systemInstruction + tools)
                if (!this._config) {
                    this._config = await SNW.Api.getLiveConfig();
                    SNW.LogUtil.info('[AILive] config loaded',
                        (this._config.tools?.[0]?.function_declarations?.length || 0) + ' funcs');
                }

                // 2) Ephemeral Token 발급
                const tokenResp = await SNW.Api.getLiveToken();
                if (!tokenResp.token) {
                    throw new Error('no_token');
                }
                this._token = tokenResp.token;
                SNW.LogUtil.info('[AILive] token acquired');

                // 3) WS 연결
                const wsUrl = `${SNW.CFG.AI_LIVE.WS_URL}?access_token=${encodeURIComponent(this._token)}`;
                this._ws = new WebSocket(wsUrl);
                this._ws.binaryType = 'arraybuffer';

                this._ws.onopen    = () => this._onOpen();
                this._ws.onmessage = (e) => this._onMessage(e);
                this._ws.onerror   = (e) => this._onError_ws(e);
                this._ws.onclose   = (e) => this._onClose_ws(e);

                return new Promise((resolve, reject) => {
                    const timeout = setTimeout(() => {
                        if (this._state === 'connecting') {
                            this._cleanup();
                            this._state = 'error';
                            reject(new Error('ws_connect_timeout'));
                        }
                    }, SNW.CFG.AI_LIVE.CONNECT_TIMEOUT_MS);

                    this._resolveConnect = (v) => { clearTimeout(timeout); resolve(v); };
                    this._rejectConnect  = (e) => { clearTimeout(timeout); reject(e); };
                });

            } catch (e) {
                this._state = 'error';
                SNW.LogUtil.warn('[AILive] start failed:', e.message);
                throw e;
            }
        },

        // =========================================================
        // Setup 메시지 (연결 직후 1회)
        // =========================================================
        _onOpen() {
            SNW.LogUtil.info('[AILive] ws open → sending setup');

            const cfg = this._config || {};

            const setup = {
                setup: {
                    model: 'models/gemini-2.0-flash-exp',
                    generation_config: {
                        response_modalities: ['AUDIO'],
                        speech_config: {
                            voice_config: {
                                prebuilt_voice_config: {
                                    voice_name: cfg.voiceName || SNW.CFG.AI_LIVE.DEFAULT_VOICE,
                                },
                            },
                        },
                    },
                },
            };

            // systemInstruction
            if (cfg.systemInstruction) {
                setup.setup.system_instruction = cfg.systemInstruction;
            }

            // tools
            if (cfg.tools && cfg.tools.length > 0) {
                setup.setup.tools = cfg.tools;
            }

            this._ws.send(JSON.stringify(setup));
        },

        // =========================================================
        // 수신 처리
        // =========================================================
        async _onMessage(event) {
            let data;
            try {
                data = JSON.parse(event.data);
            } catch {
                return;
            }

            // 1) Setup 완료
            if (data.setupComplete) {
                this._state = 'ready';
                this._startSessionTimer();
                if (this._resolveConnect) {
                    this._resolveConnect(true);
                    this._resolveConnect = null;
                }
                SNW.LogUtil.info('[AILive] setup complete, ready');
                return;
            }

            // 2) Tool Call (Function Call)
            if (data.toolCall) {
                await this._handleToolCall(data.toolCall);
                return;
            }

            // 3) 서버 콘텐츠 (Phase 1: 텍스트만)
            if (data.serverContent) {
                await this._handleServerContent(data.serverContent);
                return;
            }

            // 4) 세션 종료
            if (data.goAway) {
                SNW.LogUtil.info('[AILive] server goAway:', data.goAway.timeLeft);
                this._scheduleReconnect();
                return;
            }
        },

        // =========================================================
        // Tool Call → ESP32 위임
        // =========================================================
        async _handleToolCall(toolCall) {
            const calls = toolCall.functionCalls || [];
            const responses = [];

            for (const call of calls) {
                const { id, name, args } = call;

                SNW.LogUtil.info('[AILive] toolCall:', name);

                let result = 'ok';
                try {
                    const res = await SNW.Api.postLiveFunction(name, args || {});
                    result = res.result || 'ok';
                } catch (e) {
                    result = 'error:' + (e.message || 'unknown');
                    SNW.LogUtil.warn('[AILive] fn exec failed:', name, e);
                }

                // 외부 콜백 (UI 표시용)
                if (this._onAction) {
                    try { this._onAction({ name, args, result }); } catch {}
                }

                responses.push({
                    id,
                    name,
                    response: { result },
                });
            }

            // 결과를 Gemini에 반환
            if (this._ws && this._state === 'ready') {
                this._ws.send(JSON.stringify({
                    toolResponse: {
                        functionResponses: responses,
                    },
                }));
            }
        },

        // =========================================================
        // 서버 콘텐츠 (텍스트/오디오)
        // =========================================================
        async _handleServerContent(sc) {
            // Barge-in (Phase 2에서 오디오 flush)
            if (sc.interrupted) {
                SNW.LogUtil.info('[AILive] interrupted by user');
                return;
            }

            if (!sc.modelTurn || !sc.modelTurn.parts) return;

            for (const part of sc.modelTurn.parts) {
                // 텍스트
                if (part.text && this._onTextChunk) {
                    try { this._onTextChunk(part.text); } catch {}
                }

                // 오디오 (Phase 2)
                if (part.inlineData && part.inlineData.mimeType === 'audio/pcm') {
                    // Phase 1: 무시 (재생 안 함)
                    // Phase 2에서 SNW.AudioPlayback.push() 호출
                }
            }
        },

        // =========================================================
        // 세션 타이머 (8.3C)
        // =========================================================
        _startSessionTimer() {
            this._stopSessionTimer();
            this._sessionStartMs = Date.now();

            const maxMs  = SNW.CFG.AI_LIVE.MAX_SESSION_MS;
            const warnMs = SNW.CFG.AI_LIVE.SESSION_WARN_MS;

            // 경고 (1분 전)
            this._warnTimer = setTimeout(() => {
                if (this._state === 'ready' && SNW.ToastUtil) {
                    SNW.ToastUtil.info('60초 후 세션이 자동 재연결됩니다');
                }
            }, maxMs - warnMs);

            // 자동 재연결
            this._sessionTimer = setTimeout(() => {
                if (this._state === 'ready') {
                    SNW.LogUtil.info('[AILive] session max → reconnecting');
                    this._scheduleReconnect();
                }
            }, maxMs);
        },

        _stopSessionTimer() {
            if (this._warnTimer) {
                clearTimeout(this._warnTimer);
                this._warnTimer = null;
            }
            if (this._sessionTimer) {
                clearTimeout(this._sessionTimer);
                this._sessionTimer = null;
            }
        },

        // =========================================================
        // 재연결 (Exponential Backoff)
        // =========================================================
        _reconnectAttempt: 0,

        _scheduleReconnect() {
            this._cleanup();

            const delays = SNW.CFG.AI_LIVE.RECONNECT_MS;
            const wait = delays[Math.min(this._reconnectAttempt, delays.length - 1)];
            this._reconnectAttempt++;

            SNW.LogUtil.info(`[AILive] reconnecting in ${wait}ms (attempt ${this._reconnectAttempt})`);

            setTimeout(async () => {
                try {
                    await this.start({
                        onTextChunk: this._onTextChunk,
                        onAction:    this._onAction,
                        onError:     this._onError,
                        onClose:     this._onClose,
                    });
                    this._reconnectAttempt = 0;
                } catch (e) {
                    if (this._reconnectAttempt < 5) {
                        this._scheduleReconnect();
                    } else {
                        this._state = 'error';
                        if (this._onError) this._onError(e);
                    }
                }
            }, wait);
        },

        // =========================================================
        // WS 에러/종료
        // =========================================================
        _onError_ws(e) {
            SNW.LogUtil.warn('[AILive] ws error');
        },

        _onClose_ws(e) {
            SNW.LogUtil.info('[AILive] ws closed code=' + e.code);
            this._stopSessionTimer();

            if (this._state === 'connecting') {
                if (this._rejectConnect) {
                    this._rejectConnect(new Error('ws_closed_during_setup'));
                    this._rejectConnect = null;
                }
            }

            this._state = 'closed';
            if (this._onClose) try { this._onClose(e); } catch {}
        },

        // =========================================================
        // 정리 / 종료
        // =========================================================
        _cleanup() {
            this._stopSessionTimer();
            if (this._ws) {
                try { this._ws.close(); } catch {}
                this._ws = null;
            }
            this._state = 'idle';
        },

        stop() {
            this._cleanup();
            this._token = null;
            SNW.LogUtil.info('[AILive] stopped');
        },

        // =========================================================
        // 상태 조회
        // =========================================================
        isReady()      { return this._state === 'ready'; },
        isConnecting() { return this._state === 'connecting'; },
        getState()     { return this._state; },

        // =========================================================
        // 텍스트 입력 (Live 세션에 직접 주입)
        // =========================================================
        sendText(text) {
            if (!this._ws || this._state !== 'ready') return false;

            this._ws.send(JSON.stringify({
                clientContent: {
                    turns: [{
                        role: 'user',
                        parts: [{ text }],
                    }],
                    turnComplete: true,
                },
            }));
            return true;
        },
    };
})(window.SNW);
```

```javascript
                    <div class="snw-ai-sheet__quick">
                        <button class="snw-quick-chip" data-quick="1시간 후 꺼줘">⏱️ 1시간 후 꺼줘</button>
                        <button class="snw-quick-chip" data-quick="지금 상태 어때">📊 지금 상태</button>
                        <button class="snw-quick-chip" data-quick="잠깐 멈춰">⏸️ 멈춰</button>
                        <button class="snw-quick-chip" data-quick="다시 틀어줘">▶️ 다시</button>
                    </div>

                    <div class="snw-ai-sheet__input-wrap">
                        <button class="snw-ai-mic" data-action="voice" title="음성 입력">🎤</button>
                        <input type="text" class="snw-ai-sheet__input" data-slot="input"
                               placeholder="메시지 입력..." maxlength="500">
                        <button class="snw-ai-send" data-action="send">전송</button>
                    </div>

                    <div class="snw-ai-sheet__status" data-slot="status"></div>
                </div>
            `;

            document.body.appendChild(el);
            this._messagesEl = el.querySelector('[data-slot="messages"]');

            // 이력 로드
            this._history = SNW.AIClient.loadHistory();

            // 이벤트
            el.querySelectorAll('[data-action="close"]').forEach(b => b.onclick = () => this.close());
            el.querySelector('[data-action="send"]').onclick  = () => this._sendFromInput();
            el.querySelector('[data-action="voice"]').onclick = () => this._startVoice();
            el.querySelector('[data-action="clear-history"]').onclick = () => this._clearHistory();

            const input = el.querySelector('[data-slot="input"]');
            input.addEventListener('keydown', (e) => {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    this._sendFromInput();
                }
            });

            el.querySelectorAll('[data-quick]').forEach(b => {
                b.onclick = () => {
                    input.value = b.getAttribute('data-quick');
                    this._sendFromInput();
                };
            });

            // 이력 렌더
            if (this._history.length === 0) {
                this._appendMessage('assistant',
                    '안녕하세요! 음성이나 텍스트로 자연풍 선풍기를 제어하세요. 🌿');
            } else {
                this._history.forEach(m => this._appendMessage(m.role, m.text, false));
            }

            // Live 모드면 연결
            if (this._mode === 'live') {
                await this._ensureLiveConnected();
            }

            requestAnimationFrame(() => el.classList.add('is-open'));
            setTimeout(() => input.focus(), 300);
        },

        close() {
            this._visible = false;
            SNW.AIVoice.stopSpeaking();
            this._stopTyping();

            const sheet = document.getElementById('ai-sheet');
            if (sheet) {
                sheet.classList.remove('is-open');
                setTimeout(() => sheet.remove(), 200);
            }
        },

        // =========================================================
        // 모드 배지 갱신
        // =========================================================
        _updateModeBadge() {
            const badge = document.querySelector('[data-slot="mode-badge"]');
            if (!badge) return;

            if (this._mode === 'live') {
                const st = SNW.AILive.getState();
                let label = 'LIVE';
                let cls = 'snw-ai-badge--live';
                if (st === 'connecting') { label = 'LIVE ...'; }
                else if (st === 'error') { label = 'LIVE ✕'; cls = 'snw-ai-badge--error'; }
                else if (st === 'ready') { label = 'LIVE ●'; }
                badge.textContent = label;
                badge.className = `snw-ai-badge ${cls}`;
            } else {
                badge.textContent = 'REST';
                badge.className = 'snw-ai-badge snw-ai-badge--rest';
            }
        },

        // =========================================================
        // Live 연결 보장
        // =========================================================
        async _ensureLiveConnected() {
            if (SNW.AILive.isReady()) return true;

            this._setStatus('🔗 Live 연결 중...');
            this._updateModeBadge();

            try {
                await SNW.AILive.start({
                    onTextChunk: (chunk) => this._onLiveTextChunk(chunk),
                    onAction:    (act) => this._onLiveAction(act),
                    onError:     (e) => {
                        this._setStatus('⚠️ Live 오류 · REST 폴백');
                        this._fallbackToRest();
                    },
                    onClose:     () => {
                        if (this._mode === 'live') {
                            this._setStatus('🔌 Live 연결 종료');
                            this._updateModeBadge();
                        }
                    },
                });
                this._setStatus('');
                this._updateModeBadge();
                return true;
            } catch (e) {
                this._setStatus(`⚠️ Live 연결 실패: ${e.message} · REST 폴백`);
                this._fallbackToRest();
                return false;
            }
        },

        _fallbackToRest() {
            this._mode = 'rest';
            localStorage.setItem(SNW.CFG.LS_KEY.AI_MODE, 'rest');
            this._updateModeBadge();
            SNW.AILive.stop();
        },

        // =========================================================
        // Live 콜백
        // =========================================================
        _onLiveTextChunk(chunk) {
            if (!this._typingBubble) {
                this._typingBubble = this._appendEmptyAssistant();
                this._typingBuffer = '';
            }
            this._typingBuffer += chunk;
            this._typingBubble.querySelector('.snw-ai-bubble').textContent = this._typingBuffer;
            this._messagesEl.scrollTop = this._messagesEl.scrollHeight;
        },

        _onLiveAction(act) {
            const labels = {
                start_playlist: '▶ 플레이리스트',
                start_preset:   '▶ 프리셋',
                start_fixed:    '🌀 일반 선풍기',
                stop_session:   '⏹ 정지',
                pause_session:  '⏸ 일시정지',
                resume_session: '▶ 재개',
                set_timer:      '⏱️ 타이머',
                add_schedule:   '⏰ 스케줄',
                get_status:     '📊 상태 조회',
                list_profiles:  '📋 프로필 목록',
            };
            this._setStatus(`⚙️ ${labels[act.name] || act.name}`);
            setTimeout(() => {
                if (this._typingBubble) return;
                this._setStatus('');
            }, 2000);
        },

        _stopTyping() {
            this._typingBubble = null;
            this._typingBuffer = '';
        },

        // =========================================================
        // 전송 (모드 분기)
        // =========================================================
        async _sendFromInput(overrideText) {
            const input = document.querySelector('[data-slot="input"]');
            const text = (overrideText || (input && input.value) || '').trim();
            if (!text) return;

            if (input) input.value = '';

            this._appendMessage('user', text);
            this._history.push({ role: 'user', text });
            SNW.AIClient.saveHistory(this._history);

            if (this._mode === 'live' && SNW.AILive.isReady()) {
                // Live: clientContent로 텍스트 주입
                this._setStatus('💭 응답 대기...');
                this._stopTyping();
                SNW.AILive.sendText(text);
                // 응답은 onTextChunk로 도착
                setTimeout(() => {
                    if (this._typingBuffer) {
                        this._history.push({ role: 'model', text: this._typingBuffer });
                        SNW.AIClient.saveHistory(this._history);
                        this._stopTyping();
                        this._setStatus('');
                    }
                }, 5000);
            } else {
                // REST
                await this._sendRest(text);
            }
        },

        async _sendRest(text) {
            this._setStatus('🤔 생각 중...');

            try {
                const res = await SNW.AIClient.send(text, this._history.slice(-6));

                const responseText = res.text || '(응답 없음)';
                this._appendMessage('assistant', responseText);
                this._history.push({ role: 'model', text: responseText });
                SNW.AIClient.saveHistory(this._history);

                if (res.actions && res.actions.length > 0) {
                    const labels = res.actions.map(a => {
                        const m = {
                            start_playlist: '플레이리스트', start_preset: '프리셋',
                            start_fixed: '일반 선풍기', stop_session: '정지',
                            pause_session: '일시정지', resume_session: '재개',
                            set_timer: '타이머', add_schedule: '스케줄',
                        };
                        return m[a.name] || a.name;
                    }).join(', ');
                    this._setStatus(`✓ ${labels}`);
                    setTimeout(() => this._setStatus(''), 2500);
                } else {
                    this._setStatus('');
                }

                // TTS
                if (SNW.AIVoice.isTTSSupported() && responseText.length > 0) {
                    SNW.AIVoice.speak(responseText, {
                        lang: SNW.AIVoice.detectLang(responseText),
                        rate: 1.05,
                    });
                }
            } catch (e) {
                this._appendMessage('assistant', `❌ ${this._errorText(e)}`);
                this._setStatus('');
            }
        },

        // =========================================================
        // 음성 (8.1 그대로)
        // =========================================================
        async _startVoice() {
            if (this._listening) return;

            if (!SNW.AIVoice.isSTTSupported()) {
                SNW.ToastUtil.warn('음성 인식 미지원 (Chrome/Edge 권장)');
                document.querySelector('[data-slot="input"]')?.focus();
                return;
            }

            this._listening = true;
            const micBtn = document.querySelector('.snw-ai-mic');
            if (micBtn) micBtn.classList.add('is-listening');
            this._setStatus('🎤 듣고 있어요...');

            try {
                const result = await SNW.AIVoice.startListening({
                    lang: SNW.I18n.getLang() === 'ko' ? 'ko-KR' : 'en-US',
                    timeoutMs: 10000,
                });
                this._setStatus('');
                await this._sendFromInput(result.transcript);
            } catch (e) {
                this._setStatus('');
                const input = document.querySelector('[data-slot="input"]');
                if (input) {
                    input.focus();
                    input.placeholder = '음성 인식 실패 · 직접 입력하세요';
                }
                SNW.ToastUtil.warn(e.message === 'not-allowed'
                    ? '🚫 마이크 권한 필요' : `⚠️ ${e.message}`);
            } finally {
                this._listening = false;
                if (micBtn) micBtn.classList.remove('is-listening');
            }
        },

        // =========================================================
        // 유틸
        // =========================================================
        _appendMessage(role, text, save = true) {
            if (!this._messagesEl) return;
            const div = document.createElement('div');
            div.className = `snw-ai-msg snw-ai-msg--${role}`;
            div.innerHTML = `<div class="snw-ai-bubble">${SNW.DomUtil.escape(text)}</div>`;
            this._messagesEl.appendChild(div);
            this._messagesEl.scrollTop = this._messagesEl.scrollHeight;
            return div;
        },

        _appendEmptyAssistant() {
            if (!this._messagesEl) return null;
            const div = document.createElement('div');
            div.className = 'snw-ai-msg snw-ai-msg--assistant';
            div.innerHTML = `<div class="snw-ai-bubble"></div>`;
            this._messagesEl.appendChild(div);
            this._messagesEl.scrollTop = this._messagesEl.scrollHeight;
            return div;
        },

        _setStatus(text) {
            const el = document.querySelector('[data-slot="status"]');
            if (el) el.textContent = text;
        },

        _clearHistory() {
            if (!confirm('대화 이력을 삭제하시겠습니까?')) return;
            this._history = [];
            SNW.AIClient.clearHistory();
            if (this._messagesEl) {
                this._messagesEl.innerHTML = '';
                this._appendMessage('assistant', '대화가 초기화되었습니다.');
            }
        },

        _errorText(e) {
            if (e.status === 429) return '요청 과다 · 30초 후 재시도';
            if (e.status === 403) {
                setTimeout(() => SNW.Router.navigate('/settings'), 2000);
                return 'API Key 무효 · 설정으로 이동';
            }
            if (e.status === 502) return '네트워크 오류';
            if (e.status === 400) return '요청 형식 오류';
            return e.message || '알 수 없는 오류';
        },
    };
})(window.SNW);
```

Part 1. 오디오 파이프라인 아키텍처

```
┌─────────────────────────────────────────────────────────┐
│  마이크 (MediaDevices)                                   │
│  ↓ AEC 필수: echoCancellation:true                       │
│  ↓ noiseSuppression / autoGainControl                    │
│  ┌─────────────────────────────────────────────┐         │
│  │ AudioContext (16kHz 시도 → 폴백 리샘플)      │         │
│  │ ↓ AudioWorklet "pcm-capture-processor"       │         │
│  │ ↓ 100ms 청크 (1600 samples @ 16kHz)          │         │
│  └─────────────────────────────────────────────┘         │
│  ↓ Int16LE 변환 → Base64                                 │
│  ↓ WS realtimeInput.mediaChunks                          │
│  ═══════════════════════════════════════════════════     │
│  ↑ WS serverContent.modelTurn.parts[].inlineData         │
│  ↓ Base64 → Int16 → Float32                              │
│  ┌─────────────────────────────────────────────┐         │
│  │ AudioContext (24kHz 시도 → 폴백 리샘플)      │         │
│  │ ↓ 재생 큐 (AudioBufferSourceNode)            │         │
│  │ ↓ interrupted:true → 즉시 flush              │         │
│  └─────────────────────────────────────────────┘         │
│  ↓ 스피커 (AEC가 에코 제거 → Barge-in 오탐 방지)          │
└─────────────────────────────────────────────────────────┘
```

Part 2. 147a.worklet.js — AudioWorklet 프로세서

```javascript
// =========================================================
// 147a.worklet.js — PCM 캡처 AudioWorklet (100ms 청크)
// 등록: AudioContext.audioWorklet.addModule('./js/147a.worklet.js')
// =========================================================
class PCMCaptureProcessor extends AudioWorkletProcessor {
    constructor() {
        super();
        // 100ms @ 16kHz = 1600 samples
        // ⚠️ 실제 컨텍스트 sampleRate를 모를 수 있음 → 런타임에 리사이즈
        this._targetSamples = 1600;
        this._buffer = new Float32Array(this._targetSamples);
        this._idx = 0;
        this._actualRate = sampleRate;   // AudioWorkletGlobalScope
    }

    process(inputs) {
        const input = inputs[0];
        if (!input || !input[0]) return true;

        const channel = input[0];   // mono

        // 실제 컨텍스트 rate가 16kHz가 아니면 청크 크기 조정
        if (this._actualRate !== 16000) {
            // 리샘플은 상위(캡처 측)에서 처리
            // 여기서는 원본 rate 그대로 100ms에 해당하는 샘플 수만큼 모음
            const desired = Math.floor(this._actualRate * 0.1);   // 100ms
            if (this._buffer.length !== desired) {
                this._buffer = new Float32Array(desired);
                this._idx = 0;
            }
        }

        for (let i = 0; i < channel.length; i++) {
            this._buffer[this._idx++] = channel[i];
            if (this._idx >= this._buffer.length) {
                // 청크 완성 → 메인 스레드로 전송
                const chunk = new Float32Array(this._buffer);
                this.port.postMessage({
                    samples: chunk,
                    sampleRate: this._actualRate,
                }, [chunk.buffer]);   // transferable
                this._idx = 0;
            }
        }

        return true;
    }
}

registerProcessor('pcm-capture-processor', PCMCaptureProcessor);
```

설계 근거:

항목 선택 이유
100ms 청크 1600 samples @ 16kHz 참조 문서 §6 (100~200ms 최적)
transferable [chunk.buffer] 메모리 복사 회피 (-5ms)
동적 버퍼 크기 sampleRate * 0.1 브라우저 sampleRate 차이 대응

Part 3. 147a.audio.capture.js — 마이크 캡처

```javascript
// =========================================================
// 147a.audio.capture.js — 마이크 캡처 (16kHz + AEC)
// - 우선 16kHz AudioContext 시도
// - 실패 시 native rate + 오프라인 리샘플
// =========================================================
window.SNW = window.SNW || {};
(function(SNW) {
    'use strict';

    const TARGET_RATE = 16000;

    SNW.AudioCapture = {
        _stream: null,
        _audioCtx: null,
        _workletNode: null,
        _sourceNode: null,
        _muted: false,
        _onChunkCb: null,
        _resampleRatio: 1.0,

        // =========================================================
        // 시작
        // =========================================================
        async start(onChunk) {
            if (this._audioCtx) {
                SNW.LogUtil.info('[AudioCapture] already started');
                return true;
            }

            // 마이크 권한 + AEC 필수
            try {
                this._stream = await navigator.mediaDevices.getUserMedia({
                    audio: {
                        echoCancellation: true,       // ★ 필수 (참조 §6)
                        noiseSuppression: true,
                        autoGainControl:  true,
                        channelCount:     1,
                        sampleRate:       TARGET_RATE,
                    },
                });
            } catch (e) {
                throw new Error('mic_denied:' + (e.name || 'unknown'));
            }

            // AudioContext 생성 (16kHz 시도)
            let ctx;
            try {
                ctx = new AudioContext({ sampleRate: TARGET_RATE });
            } catch {
                ctx = new AudioContext();   // native rate 폴백
            }

            if (ctx.state === 'suspended') {
                await ctx.resume();
            }

            this._audioCtx = ctx;
            const actualRate = ctx.sampleRate;

            if (actualRate !== TARGET_RATE) {
                // 리샘플 필요
                this._resampleRatio = TARGET_RATE / actualRate;
                SNW.LogUtil.info(`[AudioCapture] resample ${actualRate} → ${TARGET_RATE} (ratio=${this._resampleRatio.toFixed(3)})`);
            } else {
                this._resampleRatio = 1.0;
            }

            // Worklet 로드
            try {
                await ctx.audioWorklet.addModule('./js/147a.worklet.js');
            } catch (e) {
                this.stop();
                throw new Error('worklet_load_failed');
            }

            this._sourceNode  = ctx.createMediaStreamSource(this._stream);
            this._workletNode = new AudioWorkletNode(ctx, 'pcm-capture-processor');

            this._onChunkCb = onChunk;

            this._workletNode.port.onmessage = (e) => {
                if (this._muted) return;

                const { samples, sampleRate } = e.data;

                // 리샘플 (필요 시)
                let resampled = samples;
                if (sampleRate !== TARGET_RATE) {
                    resampled = this._resampleLinear(samples, sampleRate, TARGET_RATE);
                }

                // Float32 → Int16 (LE)
                const pcm16 = this._float32ToInt16(resampled);

                // 콜백 (Int16Array, 1600 samples = 3200 bytes)
                if (this._onChunkCb) {
                    try { this._onChunkCb(pcm16); } catch (e) {
                        SNW.LogUtil.warn('[AudioCapture] onChunk error', e);
                    }
                }
            };

            this._sourceNode.connect(this._workletNode);
            // 스피커로 연결하지 않음 (에코 방지)

            SNW.LogUtil.info('[AudioCapture] started (rate=' + actualRate + ')');
            return true;
        },

        // =========================================================
        // 중지
        // =========================================================
        stop() {
            if (this._stream) {
                this._stream.getTracks().forEach(t => { try { t.stop(); } catch {} });
                this._stream = null;
            }
            if (this._workletNode) {
                try { this._workletNode.disconnect(); } catch {}
                this._workletNode = null;
            }
            if (this._sourceNode) {
                try { this._sourceNode.disconnect(); } catch {}
                this._sourceNode = null;
            }
            if (this._audioCtx) {
                try { this._audioCtx.close(); } catch {}
                this._audioCtx = null;
            }
            this._onChunkCb = null;
            this._muted = false;
            SNW.LogUtil.info('[AudioCapture] stopped');
        },

        // =========================================================
        // 음소거 (모델 재생 중 마이크 일시 차단)
        // =========================================================
        mute() {
            this._muted = true;
        },
        unmute() {
            this._muted = false;
        },
        isMuted() {
            return this._muted;
        },

        // =========================================================
        // Linear 리샘플 (단순, 품질 충분)
        // =========================================================
        _resampleLinear(input, fromRate, toRate) {
            if (fromRate === toRate) return input;

            const ratio = fromRate / toRate;
            const outLen = Math.floor(input.length / ratio);
            const out = new Float32Array(outLen);

            for (let i = 0; i < outLen; i++) {
                const srcPos = i * ratio;
                const srcIdx = Math.floor(srcPos);
                const frac = srcPos - srcIdx;
                const s0 = input[srcIdx] || 0;
                const s1 = input[srcIdx + 1] || s0;
                out[i] = s0 + (s1 - s0) * frac;
            }
            return out;
        },

        // =========================================================
        // Float32 → Int16 LE
        // =========================================================
        _float32ToInt16(float32) {
            const int16 = new Int16Array(float32.length);
            for (let i = 0; i < float32.length; i++) {
                const s = Math.max(-1, Math.min(1, float32[i]));
                int16[i] = s < 0 ? s * 0x8000 : s * 0x7FFF;
            }
            return int16;
        },
    };
})(window.SNW);
```

Part 4. 147b.audio.playback.js — PCM 재생 + Barge-in

```javascript
// =========================================================
// 147b.audio.playback.js — PCM 재생 (24kHz, 버퍼 큐, Barge-in)
// =========================================================
window.SNW = window.SNW || {};
(function(SNW) {
    'use strict';

    const TARGET_RATE = 24000;

    SNW.AudioPlayback = {
        _audioCtx: null,
        _sources: [],           // 활성 AudioBufferSourceNode 목록
        _queue: [],             // 대기 중인 Float32Array[]
        _playing: false,
        _nextStartMs: 0,
        _totalChunks: 0,
        _flushedChunks: 0,

        // =========================================================
        // 시작
        // =========================================================
        async start() {
            if (this._audioCtx) return true;

            let ctx;
            try {
                ctx = new AudioContext({ sampleRate: TARGET_RATE });
            } catch {
                ctx = new AudioContext();
            }

            if (ctx.state === 'suspended') {
                await ctx.resume();
            }

            this._audioCtx = ctx;
            const actualRate = ctx.sampleRate;

            if (actualRate !== TARGET_RATE) {
                SNW.LogUtil.info(`[AudioPlayback] browser rate=${actualRate} → resample to ${TARGET_RATE}`);
            }

            this._queue = [];
            this._playing = false;
            this._nextStartMs = 0;
            this._sources = [];
            this._totalChunks = 0;
            this._flushedChunks = 0;

            SNW.LogUtil.info('[AudioPlayback] started (rate=' + actualRate + ')');
            return true;
        },

        // =========================================================
        // 청크 push (Int16 24kHz)
        // =========================================================
        push(pcm16) {
            if (!this._audioCtx) return;

            // Int16 → Float32
            const float32 = new Float32Array(pcm16.length);
            for (let i = 0; i < pcm16.length; i++) {
                float32[i] = pcm16[i] / 32768.0;
            }

            this._queue.push(float32);
            this._totalChunks++;

            if (!this._playing) this._playNext();
        },

        // =========================================================
        // 다음 청크 예약 재생
        // =========================================================
        _playNext() {
            if (!this._audioCtx) return;
            if (this._queue.length === 0) {
                this._playing = false;
                return;
            }

            this._playing = true;
            const chunk = this._queue.shift();

            const buf = this._audioCtx.createBuffer(1, chunk.length, TARGET_RATE);
            buf.copyToChannel(chunk, 0);

            const src = this._audioCtx.createBufferSource();
            src.buffer = buf;
            src.connect(this._audioCtx.destination);

            this._sources.push(src);

            const now = this._audioCtx.currentTime;
            const startAt = Math.max(now, this._nextStartMs);

            try {
                src.start(startAt);
            } catch (e) {
                SNW.LogUtil.warn('[AudioPlayback] start error', e);
                return;
            }

            this._nextStartMs = startAt + buf.duration;

            src.onended = () => {
                // sources 배열에서 제거
                const idx = this._sources.indexOf(src);
                if (idx >= 0) this._sources.splice(idx, 1);

                // 다음 재생
                this._playNext();
            };
        },

        // =========================================================
        // ★ Barge-in: 즉시 flush
        // =========================================================
        flush() {
            if (!this._audioCtx) return;

            // 대기 큐 비우기
            const pending = this._queue.length;
            this._queue = [];
            this._flushedChunks += pending;

            // 재생 중인 소스 모두 즉시 중단
            const active = this._sources.length;
            for (const src of this._sources) {
                try { src.stop(0); } catch {}
            }
            this._sources = [];

            this._playing = false;
            this._nextStartMs = 0;

            SNW.LogUtil.info(`[AudioPlayback] flush (pending=${pending}, active=${active})`);
        },

        // =========================================================
        // 중지
        // =========================================================
        stop() {
            this.flush();
            if (this._audioCtx) {
                try { this._audioCtx.close(); } catch {}
                this._audioCtx = null;
            }
            SNW.LogUtil.info('[AudioPlayback] stopped');
        },

        // =========================================================
        // 상태 조회
        // =========================================================
        isPlaying() {
            return this._playing || this._queue.length > 0;
        },

        getStats() {
            return {
                totalChunks: this._totalChunks,
                flushedChunks: this._flushedChunks,
                queueLen: this._queue.length,
                activeSources: this._sources.length,
            };
        },
    };
})(window.SNW);
```

```javascript
// 147.ai.live.js — Phase 2 확장 (오디오 파이프라인)

    // =========================================================
    // [수정] start() — 오디오 초기화 추가
    // =========================================================
    // (기존 start 메서드 내부, setup 응답 후 오디오 시작)

    // =========================================================
    // [수정] _onMessage — setupComplete 후 오디오 시작
    // =========================================================
    // 기존 _onMessage의 setupComplete 처리 부분을 아래로 교체:

    /*
    if (data.setupComplete) {
        this._state = 'ready';
        this._startSessionTimer();

    // =========================================================
    // [신규] 오디오 파이프라인 시작

    SNW.AILive._startAudioPipeline = async function() {
        // 1) 재생 준비 (출력 먼저)
        try {
            await SNW.AudioPlayback.start();
        } catch (e) {
            throw new Error('playback_init_failed:' + e.message);

        // 2) 마이크 캡처 시작 (AEC)
        try {
            await SNW.AudioCapture.start((pcm16) => {
                this._sendAudioChunk(pcm16);
            });
        } catch (e) {
            SNW.AudioPlayback.stop();
            throw new Error('capture_init_failed:' + e.message);
        }

        SNW.LogUtil.info('[AILive] audio pipeline ready');
    };

    // =========================================================
    // [신규] PCM 청크 전송
    // =========================================================
    SNW.AILive._sendAudioChunk = function(pcm16) {
        if (!this._ws || this._state !== 'ready') return;
        if (!this._audioEnabled) return;

        // Int16Array → Uint8Array (LE 강제)
        const bytes = new Uint8Array(pcm16.buffer);
        const b64 = this._arrayBufferToBase64(bytes);

        try {
            this._ws.send(JSON.stringify({
                realtimeInput: {
                    mediaChunks: [{
                        mimeType: 'audio/pcm',
                        data: b64,
                    }],
                },
            }));
        } catch (e) {
            SNW.LogUtil.warn('[AILive] send audio failed', e);
        }
    };

    // =========================================================
    // [신규] Base64 인코딩 (Bytes → b64)
    // =========================================================
    SNW.AILive._arrayBufferToBase64 = function(bytes) {
        // 대용량은 배치 처리 (3200 bytes는 한 번에 OK)
        let binary = '';
        const len = bytes.length;
        for (let i = 0; i < len; i++) {
            binary += String.fromCharCode(bytes[i]);
        }
        return btoa(binary);
    };

    // =========================================================
    // [신규] Base64 디코딩 (b64 → Uint8Array)
    // =========================================================
    SNW.AILive._base64ToBytes = function(b64) {
        const binary = atob(b64);
        const len = binary.length;
        const bytes = new Uint8Array(len);
        for (let i = 0; i < len; i++) {
            bytes[i] = binary.charCodeAt(i);
        }
        return bytes;
    };

    // =========================================================
    // [수정] _handleServerContent — 오디오 재생 + Barge-in
    // =========================================================
    SNW.AILive._handleServerContent = function(sc) {
        // ★ Barge-in: 사용자 발화 감지 → 즉시 재생 중단

            SNW.AudioPlayback.flush();
            if (this._onInterrupt) {
                try { this._onInterrupt(); } catch {}
            }
            return;
        }

        if (!sc.modelTurn || !sc.modelTurn.parts) return;

        for (const part of sc.modelTurn.parts) {
            // 텍스트
            if (part.text) {
                if (this._onTextChunk) {
                    try { this._onTextChunk(part.text); } catch {}
                }
            }

            // ★ 오디오 (24kHz PCM)
            if (part.inlineData && part.inlineData.mimeType === 'audio/pcm') {
                try {
                    const bytes = this._base64ToBytes(part.inlineData.data);
                    const pcm16 = new Int16Array(bytes.buffer, bytes.byteOffset, bytes.byteLength / 2);
                    SNW.AudioPlayback.push(pcm16);
                } catch (e) {
                    SNW.LogUtil.warn('[AILive] audio decode failed', e);
                }
            }
        }
    };

    // =========================================================
    // [수정] _cleanup — 오디오 정리 추가
    // =========================================================
    SNW.AILive._cleanup = function() {
        this._stopSessionTimer();

        // ★ 오디오 중지
        try { SNW.AudioCapture.stop(); } catch {}
        try { SNW.AudioPlayback.stop(); } catch {}

        if (this._ws) {
            try { this._ws.close(); } catch {}
            this._ws = null;
        }
        this._state = 'idle';
    };

    // =========================================================
    // [신규] 마이크 음소거 (모델 재생 중 선택 사항)
    // =========================================================
    SNW.AILive.setMicMuted = function(muted) {
        if (muted) SNW.AudioCapture.mute();
        else       SNW.AudioCapture.unmute();
```

```cpp
static String   s_cachedPrompt;

static uint32_t s_cachedSessionHash = 0;

// 세션 상태 해시 (30초 캐시 유효성 판단)
static uint32_t _computeSessionHash() {
    auto sess = CL_CT10_ControlManager::getSessionSnapshot();
    uint32_t h = 0;
    h = h * 31 + sess.type;
    h = h * 31 + sess.state;
    h = h * 31 + sess.payload.playlist.currentStepIdx;
    h = h * 31 + (uint32_t)sess.payload.fixed.percent;
    h = h * 31 + sess.payload.playlist.profileId;
    return h;
}

String CT10_buildAIPrompt() {
    uint32_t now = millis();
    uint32_t sessionHash = _computeSessionHash();

    // 캐시 유효: 30초 + 세션 변화 없음
    if (s_cachedPrompt.length() > 0 &&
        (now - s_cachedPromptMs) < 30000 &&
        sessionHash == s_cachedSessionHash) {
        return s_cachedPrompt;
    }

    // 신규 생성
    extern String _buildPromptImpl();
    String p = _buildPromptImpl();

    s_cachedPrompt       = p;
    s_cachedPromptMs     = now;
    s_cachedSessionHash  = sessionHash;

    CL_D10_Logger::log(EN_L10_LOG_DEBUG,
        "[AI] prompt cache regenerated (%u chars)", p.length());

    return p;
```

```javascript
// (기존 WsClient 클래스에 추가)

class WsClient {
    // 기존...

    _startHeartbeat() {
        this._stopHeartbeat();
        this._heartbeatId = setInterval(() => {
            if (this.ws && this.ws.readyState === WebSocket.OPEN) {
                try {
                    this.ws.send(JSON.stringify({ type: 'ping', t: Date.now() }));
                } catch {}
            }
            // Pong 타임아웃 체크
            if (this._lastPongMs && Date.now() - this._lastPongMs > 30000) {
                SNW.LogUtil.warn(`[WS] ${this.channel} heartbeat timeout`);
                try { this.ws.close(); } catch {}
            }
        }, 10000);
    }

    _stopHeartbeat() {
        if (this._heartbeatId) {
            clearInterval(this._heartbeatId);
            this._heartbeatId = null;
        }
    }

    // onmessage 내부에서 pong 처리
    // if (data.type === 'pong') { this._lastPongMs = Date.now(); return; }
}
```

```cpp
static void handleDiag(AsyncWebServerRequest* req) {
    if (!W10_checkApiKey(req)) { req->send(401); return; }

    JsonDocument doc;
    doc["heap"]       = ESP.getFreeHeap();
    doc["heapMin"]    = ESP.getMinFreeHeap();
    doc["psram"]      = ESP.getFreePsram();
    doc["psramMin"]   = ESP.getMinFreePsram();
    doc["uptimeMs"]   = millis();
    doc["aiUsage"]    = JsonObject();   // 별도

    auto usage = CL_N10_NvsManager::getAIDailyUsage();
    JsonObject u = doc["aiUsage"].to<JsonObject>();
    u["audioSec"]   = usage.audioSec;
    u["turnCount"]  = usage.turnCount;
    u["estCostUsd"] = usage.estCostUsdX1000 / 1000.0;

    W10_sendJson(req, 200, doc);
}
```
