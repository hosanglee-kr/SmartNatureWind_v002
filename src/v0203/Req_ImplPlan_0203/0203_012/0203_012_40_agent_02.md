C-2. require_backend_approval.py 핵심 로직

```python
import json, sys, os

data = json.load(sys.stdin)
tool = data.get("tool_name", "")
tool_input = data.get("tool_input", {})

# 대상: src/v0203/**/*.h, *.cpp
path = tool_input.get("path") or tool_input.get("file_path") or ""
is_backend = path.startswith("src/v0203/") and path.endswith((".h", ".cpp"))

if is_backend:
    marker = ".agents/state/backend-approval.json"
    if not os.path.exists(marker):
        print(json.dumps({
            "decision": "deny",
            "reason": "백엔드 수정 전 승인이 필요합니다. 목적·기능변화를 정리해 사용자 승인 후 .agents/state/backend-approval.json 생성"
        }))
        sys.exit(0)

sys.exit(0)  # allow
```

4. 동기화 스크립트 (tools/sync-agent-configs.py)

Windows/macOS/Linux 크로스 플랫폼, symlink 불필요.

```python
#!/usr/bin/env python3
"""`.agents/` → `.claude/` 미러. pre-commit hook 또는 수동 실행."""
import re, shutil, sys
from pathlib import Path

ROOT = Path(__file__).parent.parent
AGENTS = ROOT / ".agents"
CLAUDE  = ROOT / ".claude"

def parse_frontmatter(text: str):
    """(frontmatter_dict_raw, body) 반환. `---` 블록 단순 파싱."""
    m = re.match(r"^---\n(.*?)\n---\n(.*)$", text, re.DOTALL)
    if not m:
        return {}, text
    fm_raw, body = m.group(1), m.group(2)
    return {"raw": fm_raw}, body

def convert_rule_frontmatter(raw: str) -> str:
    """Antigravity(trigger+globs) → Claude Code(paths)."""
    if "trigger:" not in raw or "globs:" not in raw:
        return ""  # model_decision 등은 paths 없음 → frontmatter 생략
    m = re.search(r'globs:\s*"([^"]+)"', raw)
    if not m:
        return ""
    globs = [g.strip() for g in m.group(1).split(",") if g.strip()]
    lines = ["paths:"]
    for g in globs:
        lines.append(f'  - "{g}"')
    return "\n".join(lines)

def sync_rules():
    src_dir = AGENTS / "rules"
    dst_dir = CLAUDE / "rules"
    if not src_dir.exists(): return
    dst_dir.mkdir(parents=True, exist_ok=True)
    for src in src_dir.glob("*.md"):
        fm, body = parse_frontmatter(src.read_text(encoding="utf-8"))
        new_fm = convert_rule_frontmatter(fm.get("raw", ""))
        out = f"---\n{new_fm}\n---\n{body}" if new_fm else body
        (dst_dir / src.name).write_text(out, encoding="utf-8")
        print(f"[rule] {src.name} → {dst_dir/ src.name}")

def sync_skills():
    src_dir = AGENTS / "skills"
    dst_dir = CLAUDE / "skills"
    if not src_dir.exists(): return
    for skill_dir in src_dir.iterdir():
        if not (skill_dir / "SKILL.md").exists(): continue
        target = dst_dir / skill_dir.name
        if target.exists(): shutil.rmtree(target)
        shutil.copytree(skill_dir, target)
        print(f"[skill] {skill_dir.name}")

def sync_workflows():
    src_dir = AGENTS / "workflows"
    dst_dir = CLAUDE / "commands"
    if not src_dir.exists(): return
    dst_dir.mkdir(parents=True, exist_ok=True)
    for src in src_dir.glob("*.md"):
        body = src.read_text(encoding="utf-8")
        # 첫 H1을 description으로
        m = re.search(r"^#\s+(.+)$", body, re.MULTILINE)
        desc = m.group(1).strip() if m else src.stem
        out = f"---\ndescription: {desc}\n---\n\n{body}"
        (dst_dir / src.name).write_text(out, encoding="utf-8")
        print(f"[workflow] {src.name}")

if __name__ == "__main__":
    CLAUDE.mkdir(exist_ok=True)
    sync_rules()
    sync_skills()
    sync_workflows()
    print("Sync complete.")
```

1. tools/sync-agent-configs.py

```python
#!/usr/bin/env python3
"""`.agents/` → `.claude/` 동기화 (SmartNatureWind v0203).

`.agents/` 를 canonical로 하고 `.claude/` 는 생성물로 취급한다.
CI/pre-commit 에서 `--check` 로 drift 감지.

Usage:
    python tools/sync-agent-configs.py           # sync
    python tools/sync-agent-configs.py --check   # drift 검사 (exit 1 if drift)
"""
from __future__ import annotations

import argparse
import re
import shutil
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
AGENTS = ROOT / ".agents"
CLAUDE = ROOT / ".claude"

FRONTMATTER_RE = re.compile(r"^---\r?\n(.*?)\r?\n---\r?\n(.*)$", re.DOTALL)


def split_frontmatter(text: str) -> tuple[str, str]:
    m = FRONTMATTER_RE.match(text)
    return (m.group(1), m.group(2)) if m else ("", text)


def extract_globs(fm_raw: str) -> list[str]:
    m = re.search(r'^globs:\s*"?([^"\n]+)"?', fm_raw, re.MULTILINE)
    if not m:
        return []
    return [g.strip() for g in m.group(1).split(",") if g.strip()]


def convert_rule(agents_text: str) -> str:
    """Antigravity(trigger/globs) → Claude Code(paths)."""
    fm_raw, body = split_frontmatter(agents_text)
    if "trigger:" not in fm_raw:
        return body

    trigger_m = re.search(r"^trigger:\s*(\S+)", fm_raw, re.MULTILINE)
    trigger = trigger_m.group(1) if trigger_m else ""

    if trigger == "glob":
        globs = extract_globs(fm_raw)
        if not globs:
            return body
        paths_yaml = "\n".join(f'  - "{g}"' for g in globs)
        return f"---\npaths:\n{paths_yaml}\n---\n{body}"

    # model_decision / always_on → frontmatter 생략 (Claude Code는 항상 로드)
    return body


def convert_workflow(agents_text: str, fallback_desc: str) -> str:
    fm_raw, body = split_frontmatter(agents_text)
    if fm_raw:
        return agents_text
    m = re.search(r"^#\s+(.+)$", body, re.MULTILINE)
    desc = m.group(1).strip() if m else fallback_desc
    return f"---\ndescription: {desc}\n---\n\n{body}"


class SyncStats:
    def __init__(self) -> None:
        self.written = 0
        self.drift = 0


def _write_or_check(target: Path, content: str, check: bool, stats: SyncStats) -> None:
    if target.exists() and target.read_text(encoding="utf-8") == content:
        return
    if check:
        print(f"[DRIFT] {target.relative_to(ROOT)}")
        stats.drift += 1
        return
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(content, encoding="utf-8")
    print(f"[write] {target.relative_to(ROOT)}")
    stats.written += 1


def sync_rules(check: bool, stats: SyncStats) -> None:
    src_dir = AGENTS / "rules"
    if not src_dir.is_dir():
        return
    for src in sorted(src_dir.glob("*.md")):
        converted = convert_rule(src.read_text(encoding="utf-8"))
        _write_or_check(CLAUDE / "rules" / src.name, converted, check, stats)


def sync_skills(check: bool, stats: SyncStats) -> None:
    src_dir = AGENTS / "skills"
    if not src_dir.is_dir():
        return
    for skill_dir in sorted(src_dir.iterdir()):
        if not (skill_dir / "SKILL.md").is_file():
            continue
        dst_dir = CLAUDE / "skills" / skill_dir.name
        for f in skill_dir.rglob("*"):
            if not f.is_file():
                continue
            target = dst_dir / f.relative_to(skill_dir)
            if check:
                if not target.exists() or target.read_bytes() != f.read_bytes():
                    print(f"[DRIFT] {target.relative_to(ROOT)}")
                    stats.drift += 1
            else:
                target.parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(f, target)
                stats.written += 1


def sync_workflows(check: bool, stats: SyncStats) -> None:
    src_dir = AGENTS / "workflows"
    if not src_dir.is_dir():
        return
    for src in sorted(src_dir.glob("*.md")):
        converted = convert_workflow(
            src.read_text(encoding="utf-8"), fallback_desc=src.stem
        )
        _write_or_check(CLAUDE / "commands" / src.name, converted, check, stats)


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--check", action="store_true", help="drift만 감지 (CI/pre-commit)")
    args = ap.parse_args()

    stats = SyncStats()
    sync_rules(args.check, stats)
    sync_skills(args.check, stats)
    sync_workflows(args.check, stats)

    if args.check:
        print(f"\nCheck: {stats.drift} drift(s)")
        return 1 if stats.drift else 0

    print(f"\nSync complete: {stats.written} file(s) written")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

### 7.1 `.agents/hooks/require_backend_approval.py`

```python
#!/usr/bin/env python3
"""PreToolUse hook — 백엔드 C++ 수정 전 승인 검증.

Antigravity: .agents/hooks.json
Claude Code: .claude/settings.json

승인 마커: .agents/state/backend-approval.json
{
  "approved": true,
  "purpose": "...",
  "files": ["src/v0203/CT10_Ctl_0203.cpp"],
  "approved_at": "2026-11-09T10:30:00Z"
}
"""
import json
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent.parent
MARKER = ROOT / ".agents" / "state" / "backend-approval.json"
BACKEND_PREFIX = "src/v0203/"


def is_backend(path: str) -> bool:
    if not path:
        return False
    normalized = path.replace("\\", "/")
    return normalized.startswith(BACKEND_PREFIX) and normalized.endswith((".h", ".cpp"))


def main() -> int:
    try:
        data = json.load(sys.stdin)
    except json.JSONDecodeError:
        return 0  # 파싱 실패는 allow (fail-open)

    tool_input = data.get("tool_input", {})
    path = (
        tool_input.get("path")
        or tool_input.get("file_path")
        or tool_input.get("filePath")
        or ""
    )

    if not is_backend(path):
        return 0

    if MARKER.is_file():
        return 0

    print(
        json.dumps(
            {
                "decision": "deny",
                "reason": (
                    f"백엔드 수정 전 승인 필요: {path}\n"
                    "AGENTS.md 원칙 1: 목적·기능변화를 정리해 사용자 승인 후 "
                    ".agents/state/backend-approval.json 생성하세요."
                ),
            }
        )
    )
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

````markdown
2.4 이관 자동화 스크립트 (선택)

수동 복붙이 부담되면 다음 1회성 스크립트 사용:

```python
# tools/split_audit_log.py — 1회 실행 (idempotent)
"""CONTRACT_0203.md §7.1~§7.10 → AUDIT_LOG_0203.md 이관."""
import re
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
CONTRACT = ROOT / "src/v0203/contract_doc_0203/CONTRACT_0203.md"
AUDIT_LOG = ROOT / "src/v0203/contract_doc_0203/AUDIT_LOG_0203.md"

text = CONTRACT.read_text(encoding="utf-8")

# §7.1 시작부터 §7.11 직전까지 (감사 이력)
m_start = re.search(r"^### 7\.1\b", text, re.MULTILINE)
m_end = re.search(r"^### 7\.11\b", text, re.MULTILINE)

if not (m_start and m_end):
    print("SKIP: 이미 분리됨 또는 패턴 불일치")
    raise SystemExit(0)

historical = text[m_start.start():m_end.start()].rstrip()
print(f"추출: {len(historical):,} chars")

# AUDIT_LOG 파일 골격
skeleton = (ROOT / "tools/_audit_log_skeleton.md").read_text(encoding="utf-8")
AUDIT_LOG.write_text(skeleton.replace("<!-- HISTORICAL_BODY -->", historical), encoding="utf-8")

# CONTRACT 재작성
replacement = (
    "## 7. 감사 관측 및 이슈 추적표\n\n"
    "> **감사 이력 (F-01~Phase 6, 모두 해결)은 "
    "[`AUDIT_LOG_0203.md`](AUDIT_LOG_0203.md)로 이관**되었습니다.\n"
    "> 본 §7은 **현재 활성 계약** (§7.11~§7.18)만 유지합니다.\n\n"
    "### 7.0 이력 참조 인덱스\n\n"
    "| 시리즈 | 대상 | 상세 링크 |\n|---|---|---|\n"
    "| F-01~I-6 | 초기 감사 | AUDIT_LOG §1 |\n"
    "| N/G 시리즈 | N-1~N-4, G-1~G-3 | AUDIT_LOG §2 |\n"
    "| Batch A~F | A-1~F-10 | AUDIT_LOG §3 |\n"
    "| Batch G | C-1~C-5, B-1~B-13 | AUDIT_LOG §4, §5 |\n"
    "| Batch H/I | LOAD, OVR, OFFLINE | AUDIT_LOG §6~§8 |\n"
    "| SYNC | SYNC-1~3 | AUDIT_LOG §9 |\n"
    "| Phase 1~6 | N-5~N-30, P-1~P-5 | AUDIT_LOG §10 |\n"
    "| PHY | PHY-1~4 | AUDIT_LOG §11 |\n\n---\n\n"
)
new_text = text[: m_start.start()] + replacement + text[m_end.start():]
CONTRACT.write_text(new_text, encoding="utf-8")
print(f"CONTRACT 축소: {len(text):,} → {len(new_text):,} chars")
````

2. .git/hooks/pre-commit (크로스 플랫폼)

.git/hooks/pre-commit — Unix는 chmod +x 필요, Windows는 Git for Windows가 sh로 실행.

```bash
#!/bin/sh
# ============================================================
# SmartNatureWind v0203 — .agents/ → .claude/ drift 방지
# ============================================================

set -e

# 1) drift 검사
if ! python tools/sync-agent-configs.py --check > /tmp/sync_check.log 2>&1; then
    echo "[pre-commit] .claude/ drift 감지 → 자동 sync"
    cat /tmp/sync_check.log
    python tools/sync-agent-configs.py
    git add .claude/rules .claude/skills .claude/commands 2>/dev/null || true
fi

# 2) 계약 문서 변경 감지 시 strict 검증
if git diff --cached --name-only | grep -q "contract_doc_0203/.*\.md"; then
    echo "[pre-commit] 계약 문서 변경 감지 → strict 검증"
    python src/v0203/tools_0203/commit_contracts_0203.py --dry-run --strict
fi

# 3) FE 자산 변경 감지 시 i18n 검증
if git diff --cached --name-only | grep -q "data_v0203_src/html_v0203/.*\.js"; then
    echo "[pre-commit] FE 자산 변경 감지 → i18n 검증"
    python src/v0203/tools_0203/check_i18n.py --strict
fi

exit 0
```

```yaml
          else
            echo "validate_contracts.py 미존재 → skip"
          fi

  diff-size-guard:
    name: PR Diff Size Guard (경고)
    runs-on: ubuntu-latest
    if: github.event_name == 'pull_request'
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Warn if huge diff
        run: |
          BASE=${{ github.event.pull_request.base.sha }}
          HEAD=${{ github.event.pull_request.head.sha }}
          FILES=$(git diff --name-only "$BASE" "$HEAD" | wc -l)
          LINES=$(git diff --numstat "$BASE" "$HEAD" | awk '{s+=$1+$2} END {print s+0}')
          echo "Changed files: $FILES, changed lines: $LINES"
          if [ "$LINES" -gt 3000 ]; then
            echo "::warning::PR diff가 3000줄을 초과합니다 (karpathy-guidelines §3 Surgical Changes 참조)"
          fi
```

4. tools/graphify_init.py

graphify-out/graph.json 이 없는 초기 환경에서 1회 실행. PowerShell 의존 없이 cross-platform.

```python
#!/usr/bin/env python3
"""v0203 graphify 초기 빌드 래퍼 (cross-platform).

Usage:
    python tools/graphify_init.py             # full build
    python tools/graphify_init.py --check     # 기존 graph 존재 확인만
    python tools/graphify_init.py --update    # 증분 빌드

기존 .agents/skills/graphify/SKILL.md 는 PowerShell 예제 기준이므로,
이 래퍼는 Windows/macOS/Linux 공통 엔트리를 제공한다.
"""
from __future__ import annotations

import argparse
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
GRAPH_JSON = ROOT / "graphify-out" / "graph.json"


def have_graphify() -> str | None:
    for name in ("graphify", "graphifyy"):
        p = shutil.which(name)
        if p:
            return p
    return None


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--check", action="store_true", help="graph.json 존재만 확인")
    ap.add_argument("--update", action="store_true", help="증분 빌드")
    ap.add_argument("--mode", default="normal", choices=("normal", "deep"),
                    help="추출 모드 (deep: 더 풍부한 INFERRED 엣지)")
    args = ap.parse_args()

    if args.check:
        if GRAPH_JSON.is_file():
            size_kb = GRAPH_JSON.stat().st_size / 1024
            print(f"OK: {GRAPH_JSON.relative_to(ROOT)} ({size_kb:.0f} KB)")
            return 0
        print(f"MISSING: {GRAPH_JSON.relative_to(ROOT)}")
        return 1

    cli = have_graphify()
    if not cli:
        print("ERROR: graphify CLI 미설치. `pip install graphifyy` 또는 `uv tool install graphifyy`", file=sys.stderr)
        return 2

    cmd = [cli]
    if args.update:
        cmd += [".", "--update"]
    else:
        cmd += [".", "--mode", args.mode]
    print(f"$ {' '.join(cmd)}")
    return subprocess.call(cmd, cwd=ROOT)


if __name__ == "__main__":
    sys.exit(main())
```

5. tools/approve_backend.py

원칙 1(백엔드 승인) UX. backend-approval.json을 안전하게 생성.

```python
#!/usr/bin/env python3
"""백엔드 C++ 수정 승인 마커 생성 (AGENTS.md 원칙 1).

Usage:
    python tools/approve_backend.py \\
        --purpose "Session FSM pause 버그 수정" \\
        --files "src/v0203/CT10_Ctl_Ctl_0203.cpp,src/v0203/CT10_Ctl_0203.h"

마커: .agents/state/backend-approval.json
Hook(`.agents/hooks/require_backend_approval.py`)이 이 마커 존재 여부로 allow/deny 판정.

작업 완료 후 반드시:
    python tools/approve_backend.py --revoke
"""
from __future__ import annotations

import argparse
import json
import sys
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
STATE_DIR = ROOT / ".agents" / "state"
MARKER = STATE_DIR / "backend-approval.json"


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--purpose", help="수정 목적·기능변화 요약 (필수, 10자 이상)")
    ap.add_argument("--files", help="대상 파일 (comma-separated)")
    ap.add_argument("--revoke", action="store_true", help="승인 마커 삭제")
    args = ap.parse_args()

    if args.revoke:
        if MARKER.is_file():
            MARKER.unlink()
            print(f"REVOKED: {MARKER.relative_to(ROOT)}")
        else:
            print("NO-OP: 마커 없음")
        return 0

    if not args.purpose or len(args.purpose.strip()) < 10:
        print("ERROR: --purpose 는 10자 이상 필요 (AGENTS.md 원칙 1: 목적·기능변화 명확화)", file=sys.stderr)
        return 1

    if not args.files:
        print("ERROR: --files 필요", file=sys.stderr)
        return 1

    files = [f.strip() for f in args.files.split(",") if f.strip()]
    bad = [f for f in files if not f.startswith("src/v0203/")]
    if bad:
        print(f"WARN: 백엔드 경로 아님: {bad}", file=sys.stderr)

    STATE_DIR.mkdir(parents=True, exist_ok=True)
    marker = {
        "approved": True,
        "purpose": args.purpose.strip(),
        "files": files,
        "approved_at": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "approved_by": "user",
    }
    MARKER.write_text(json.dumps(marker, ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"APPROVED: {MARKER.relative_to(ROOT)}")
    print(f"  purpose: {marker['purpose']}")
    print(f"  files:   {len(files)}개")
    print("\n작업 완료 후: python tools/approve_backend.py --revoke")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

1. src/v0203/tools_0203/enhance_index_links_0203.py

TOC anchor를 신뢰 소스로 사용합니다. 헤딩에서 재계산하지 않음 → GitHub/Antigravity 렌더링과 100% 일치.

```python
#!/usr/bin/env python3
"""계약 문서 "AI 빠른 참조" 블록의 §섹션을 실제 헤딩 앵커로 링크화.

전략:
- 각 문서의 TOC에서 `- [N.M 제목](#anchor)` 패턴을 파싱 → anchor 맵 구축
- "## 🔍 AI 빠른 참조" 블록 영역에서 `§N.M` 패턴만 `[§N.M](#anchor)`로 변환
- "연관 계약" 컬럼은 다른 문서 참조이므로 그대로 유지 (in-place 링크 X)
- 이미 링크화된 항목은 skip (idempotent)

Usage:
    python src/v0203/tools_0203/enhance_index_links_0203.py --dry-run
    python src/v0203/tools_0203/enhance_index_links_0203.py
    python src/v0203/tools_0203/enhance_index_links_0203.py --verify      # anchor 유효성 검증
    python src/v0203/tools_0203/enhance_index_links_0203.py --docs path1 path2
"""
from __future__ import annotations

import argparse
import re
import sys
from pathlib import Path

THIS_DIR = Path(__file__).resolve().parent
DOC_DIR = THIS_DIR.parent / "contract_doc_0203"
ROOT = THIS_DIR.parent.parent.parent

DEFAULT_DOCS = [
    "SPEC_0203.md",
    "CONTRACT_0203.md",
    "STATE_0203.md",
    "FLOW_0203.md",
    "BUDGET_0203.md",
    "TEST_PLAN_0203.md",
]

# "- [5.1.4 AI 제어 ...](#514-ai-제어-...)"  → group1=번호, group2=anchor
TOC_RE = re.compile(
    r"^\s*[-*]\s+\[(\d+(?:\.\d+)*)\s[^\]]*\]\(#([^)]+)\)",
    re.MULTILINE,
)

# "## N.M 제목" / "### N.M.K 제목" — 앵커 유효성 검증용
HEADING_RE = re.compile(r"^#{2,6}\s+(\d+(?:\.\d+)*)\.?\s+(.+)$", re.MULTILINE)

# "## 🔍 AI 빠른 참조" 블록 시작 마커
BLOCK_START_RE = re.compile(r"^##\s+🔍\s*AI 빠른 참조", re.MULTILINE)
# 다음 `## ` 헤딩이 블록 끝
BLOCK_END_RE = re.compile(r"^##\s+", re.MULTILINE)

# §섹션 패턴 (숫자와 점)
SECTION_RE = re.compile(r"§(\d+(?:\.\d+)*)")

# 이미 링크화된 § 패턴
LINKED_RE = re.compile(r"\[§(\d+(?:\.\d+)*)\]\(#[^)]+\)")


def github_anchor(text: str) -> str:
    """GitHub/Antigravity anchor 규칙.

    - 소문자화
    - 공백 → 하이픈
    - 영숫자/한글/하이픈/언더스코어 유지
    - 그 외(.,  ,(), ↔, &, :, — 등) 제거
    """
    out = []
    for ch in text.lower():
        if ch == " ":
            out.append("-")
        elif ch.isalnum() or ch in "-_":
            out.append(ch)
    return "".join(out)


def build_anchor_map(text: str) -> dict[str, str]:
    """TOC에서 (섹션번호 → anchor) 맵."""
    return {m.group(1): m.group(2) for m in TOC_RE.finditer(text)}


def verify_anchors(text: str, anchor_map: dict[str, str]) -> list[str]:
    """TOC anchor와 실제 헤딩 anchor가 일치하는지 검증. 불일치 리스트 반환."""
    heading_map: dict[str, str] = {}
    for m in HEADING_RE.finditer(text):
        num, title = m.group(1), m.group(2).strip()
        heading_map[num] = github_anchor(f"{num} {title}")

    mismatches = []
    for num, toc_anchor in anchor_map.items():
        if num not in heading_map:
            mismatches.append(f"§{num}: 헤딩 없음 (TOC anchor={toc_anchor})")
        elif heading_map[num] != toc_anchor:
            mismatches.append(
                f"§{num}: TOC={toc_anchor} ≠ 계산={heading_map[num]}"
            )
    return mismatches


def find_block(text: str) -> tuple[int, int] | None:
    """'AI 빠른 참조' 블록의 (시작, 끝) 인덱스. 없으면 None."""
    start_m = BLOCK_START_RE.search(text)
    if not start_m:
        return None
    start = start_m.start()
    next_m = BLOCK_END_RE.search(text, start_m.end())
    end = next_m.start() if next_m else len(text)
    return start, end


def linkify_block(block: str, anchor_map: dict[str, str]) -> str:
    """블록 내 §패턴 → [§N](#anchor). 이미 링크화된 것은 skip."""
    # 이미 링크화된 §를 placeholder로 stash
    placeholders: list[str] = []

    def _stash(m: re.Match) -> str:
        placeholders.append(m.group(0))
        return f"__LINK_{len(placeholders) - 1}__"

    block = LINKED_RE.sub(_stash, block)

    def _repl(m: re.Match) -> str:
        num = m.group(1)
        anchor = anchor_map.get(num)
        return f"[§{num}](#{anchor})" if anchor else m.group(0)

    block = SECTION_RE.sub(_repl, block)

    for i, orig in enumerate(placeholders):
        block = block.replace(f"__LINK_{i}__", orig)

    return block


def process_doc(path: Path, dry_run: bool, verify: bool) -> str:
    """한 문서 처리. 상태 문자열 반환: NO-CHANGE / WOULD-CHANGE / WROTE / SKIP-*"""
    if not path.is_file():
        return f"SKIP-NO-FILE: {path}"

    text = path.read_text(encoding="utf-8")
    anchor_map = build_anchor_map(text)

    if not anchor_map:
        return f"SKIP-NO-TOC: {path.name}"

    if verify:
        mismatches = verify_anchors(text, anchor_map)
        if mismatches:
            print(f"[VERIFY] {path.name}: {len(mismatches)}건 불일치")
            for mm in mismatches:
                print(f"  - {mm}")
        else:
            print(f"[VERIFY] {path.name}: OK ({len(anchor_map)} anchors)")

    bounds = find_block(text)
    if bounds is None:
        return f"SKIP-NO-BLOCK: {path.name}"

    start, end = bounds
    new_block = linkify_block(text[start:end], anchor_map)
    new_text = text[:start] + new_block + text[end:]

    if new_text == text:
        return f"NO-CHANGE: {path.name}"

    if dry_run:
        # 변경 라인 수 표시
        import difflib
        diff = list(
            difflib.unified_diff(
                text.splitlines(),
                new_text.splitlines(),
                lineterm="",
                n=0,
            )
        )
        changes = sum(1 for line in diff if line.startswith(("+", "-")) and not line.startswith(("+++", "---")))
        print(f"[DRY-RUN] {path.name}: {changes} lines would change")
        return f"WOULD-CHANGE: {path.name}"

    path.write_text(new_text, encoding="utf-8")
    return f"WROTE: {path.name}"


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--dry-run", action="store_true")
    ap.add_argument("--verify", action="store_true", help="TOC anchor 유효성 검증")
    ap.add_argument("--docs", nargs="*", help="처리할 문서 경로 (기본: 계약 6종)")
    args = ap.parse_args()

    if args.docs:
        docs = [Path(p) if Path(p).is_absolute() else (ROOT / p) for p in args.docs]
    else:
        docs = [DOC_DIR / name for name in DEFAULT_DOCS]

    results = []
    for path in docs:
        results.append(process_doc(path, args.dry_run, args.verify))

    print("\n=== 요약 ===")
    for r in results:
        print(f"  {r}")

    # verify 모드면 불일치 있으면 exit 1
    if args.verify:
        # verify_anchors가 이미 위에서 호출됨. 여기서 재수집
        any_mismatch = False
        for path in docs:
            if not path.is_file():
                continue
            text = path.read_text(encoding="utf-8")
            amap = build_anchor_map(text)
            if verify_anchors(text, amap):
                any_mismatch = True
        return 1 if any_mismatch else 0

    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def split_by_top_sections(text: str) -> tuple[str, list[tuple[str, str]]]:
    """(front_matter, [(section_num, section_text), ...]).

    front_matter = '## 1.' 이전 텍스트 (# 제목, 안내 등).
    """
    lines = text.split("\n")
    preamble: list[str] = []
    sections: list[tuple[str, str]] = []
    current_num: str | None = None
    current_lines: list[str] = []

    for line in lines:
        m = TOP_RE.match(line)
        if m:
            if current_num is None:
                preamble = current_lines
                current_lines = []
            else:
                sections.append((current_num, "\n".join(current_lines).rstrip()))
            current_num = m.group(1)
            current_lines = [line]
        else:
            current_lines.append(line)

    if current_num is not None:
        sections.append((current_num, "\n".join(current_lines).rstrip()))

    return "\n".join(preamble).rstrip(), sections


def split_section_3(text: str) -> tuple[str, dict[str, str]]:
    """§3를 base(§3.1 이전)와 서브섹션 dict로 분할."""
    lines = text.split("\n")
    base_lines: list[str] = []
    subs: dict[str, str] = {}
    current: str | None = None
    cur_lines: list[str] = []

    for line in lines:
        m = SUB3_RE.match(line)
        if m:
            if current is not None:
                subs[current] = "\n".join(cur_lines).rstrip()
            current = m.group(1)
            cur_lines = [line]
        elif current is None:
            base_lines.append(line)
        else:
            cur_lines.append(line)

    if current is not None:
        subs[current] = "\n".join(cur_lines).rstrip()

    return "\n".join(base_lines).rstrip(), subs

def extract_subsection(section: str, title_fragment: str) -> str:
    """§1 안에서 특정 ### 서브섹션만 추출."""
    m = re.search(rf"^(###[^\n]*{re.escape(title_fragment)}[^\n]*)$", section, re.MULTILINE)
    if not m:
        return section
    start = m.start()
    next_h3 = re.search(r"^###\s", section[m.end():], re.MULTILINE)
    end = m.end() + next_h3.start() if next_h3 else len(section)
    # ### → ## 로 승격 (인덱스 구조에 맞춤)
    return section[start:end].replace("###", "##", 1).rstrip()


def extract_common_setup(section: str) -> str:
    """§1의 ### 1.2, ### 1.3, ### 1.4 를 추출 (§계층 구조 제외)."""
    result = []
    for m in re.finditer(r"^###\s+(1\.[234][^\n]*)$", section, re.MULTILINE):
        start = m.start()
        next_h3 = re.search(r"^###\s", section[m.end():], re.MULTILINE)
        end = m.end() + next_h3.start() if next_h3 else len(section)
        block = section[start:end].rstrip()
        # ### 1.2 → ### 2  (재번호)
        block = re.sub(r"^###\s+1\.(\d)", r"### \1", block, count=1)
        result.append(block)
    return "\n\n".join(result) if result else section
```

1. tools/graphify_init.py (확장 — 기존 파일 대체)

기존 Phase 3의 간단 래퍼를 서브커맨드 구조로 대체합니다. 모든 graphify CLI 호출을 통일.

```python
#!/usr/bin/env python3
"""v0203 graphify CLI 래퍼 (cross-platform).

`.agents/skills/graphify/SKILL.md`의 PowerShell 파이프라인을 위임받는다.
fast path(graph.json 존재 시 query 직행)는 이 래퍼를 거치지 않아도 된다.

Subcommands:
    build    그래프 빌드 (기본: full / --update / --cluster-only)
    check    graph.json 존재 확인 (exit 0/1)
    query    자연어 질문 BFS/DFS
    path     두 노드 간 최단 경로
    explain  단일 노드 설명
    export   내보내기 (--html/--obsidian/--wiki/--svg/--graphml/--neo4j/--falkordb)
    mcp      MCP stdio 서버 (--watch)

Usage:
    python tools/graphify_init.py build --mode deep
    python tools/graphify_init.py check
    python tools/graphify_init.py query "Session FSM이 어떻게 동작해?"
    python tools/graphify_init.py query "..." --budget 1500 --dfs
    python tools/graphify_init.py path "Session" "runCtx"
    python tools/graphify_init.py explain "CT10::startSession"
    python tools/graphify_init.py export --obsidian --obsidian-dir ~/vaults/my-project
    python tools/graphify_init.py export --wiki --svg --graphml
    python tools/graphify_init.py mcp --watch
"""
from __future__ import annotations

import argparse
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
GRAPH_JSON = ROOT / "graphify-out" / "graph.json"
SCAN_PATH = "."


# ──────────────────────────────────────────────────────────────
# 공통 유틸
# ──────────────────────────────────────────────────────────────
def have_graphify() -> str | None:
    for name in ("graphify", "graphifyy"):
        p = shutil.which(name)
        if p:
            return p
    return None


def require_cli() -> str:
    cli = have_graphify()
    if not cli:
        print(
            "ERROR: graphify CLI 미설치.\n"
            "  uv tool install graphifyy\n"
            "  또는 pip install graphifyy",
            file=sys.stderr,
        )
        sys.exit(2)
    return cli


def run(cmd: list[str]) -> int:
    print(f"$ {' '.join(cmd)}")
    return subprocess.call(cmd, cwd=ROOT)


def fast_path_hint() -> None:
    """graph.json 존재 시 안내."""
    if GRAPH_JSON.is_file():
        size_kb = GRAPH_JSON.stat().st_size / 1024
        print(f"[hint] graph.json 존재 ({size_kb:.0f} KB) → 'query' 직행 가능")


# ──────────────────────────────────────────────────────────────
# 서브커맨드
# ──────────────────────────────────────────────────────────────
def cmd_check(args: argparse.Namespace) -> int:
    if GRAPH_JSON.is_file():
        size_kb = GRAPH_JSON.stat().st_size / 1024
        print(f"OK: {GRAPH_JSON.relative_to(ROOT)} ({size_kb:.0f} KB)")
        return 0
    print(f"MISSING: {GRAPH_JSON.relative_to(ROOT)}")
    print("  → python tools/graphify_init.py build")
    return 1


def cmd_build(args: argparse.Namespace) -> int:
    cli = require_cli()
    cmd = [cli]
    if args.cluster_only:
        cmd += [SCAN_PATH, "--cluster-only"]
    elif args.update:
        cmd += [SCAN_PATH, "--update"]
    else:
        cmd += [SCAN_PATH, "--mode", args.mode]
    if args.directed:
        cmd.append("--directed")
    if args.no_viz:
        cmd.append("--no-viz")
    return run(cmd)


def cmd_query(args: argparse.Namespace) -> int:
    cli = require_cli()
    if not GRAPH_JSON.is_file():
        print("ERROR: graph.json 없음. 먼저 build 실행.", file=sys.stderr)
        return 1
    cmd = [cli, "query", args.question]
    if args.dfs:
        cmd.append("--dfs")
    if args.budget:
        cmd += ["--budget", str(args.budget)]
    return run(cmd)


def cmd_path(args: argparse.Namespace) -> int:
    cli = require_cli()
    if not GRAPH_JSON.is_file():
        print("ERROR: graph.json 없음. 먼저 build 실행.", file=sys.stderr)
        return 1
    return run([cli, "path", args.source, args.target])


def cmd_explain(args: argparse.Namespace) -> int:
    cli = require_cli()
    if not GRAPH_JSON.is_file():
        print("ERROR: graph.json 없음. 먼저 build 실행.", file=sys.stderr)
        return 1
    return run([cli, "explain", args.symbol])


def cmd_export(args: argparse.Namespace) -> int:
    cli = require_cli()
    if not GRAPH_JSON.is_file():
        print("ERROR: graph.json 없음. 먼저 build 실행.", file=sys.stderr)
        return 1

    rc = 0
    if args.html or args.svg or args.graphml:
        cmd = [cli, "export", "html"]
        if args.svg:
            cmd.append("--svg")
        if args.graphml:
            cmd.append("--graphml")
        rc = rc or run(cmd)

    if args.obsidian:
        cmd = [cli, "export", "obsidian"]
        if args.obsidian_dir:
            cmd += ["--dir", args.obsidian_dir]
        rc = rc or run(cmd)

    if args.wiki:
        rc = rc or run([cli, "export", "wiki"])

    if args.neo4j:
        rc = rc or run([cli, "export", "neo4j"])

    if args.falkordb:
        rc = rc or run([cli, "export", "falkordb"])

    if not any([args.html, args.svg, args.graphml, args.obsidian,
                args.wiki, args.neo4j, args.falkordb]):
        print("ERROR: 내보내기 옵션 중 하나 이상 필요.", file=sys.stderr)
        return 2
    return rc


def cmd_mcp(args: argparse.Namespace) -> int:
    cli = require_cli()
    cmd = [cli, SCAN_PATH, "--mcp"]
    if args.watch:
        cmd.append("--watch")
    return run(cmd)


# ──────────────────────────────────────────────────────────────
# main
# ──────────────────────────────────────────────────────────────
def build_parser() -> argparse.ArgumentParser:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    sub = ap.add_subparsers(dest="command", required=True)

    # build
    p = sub.add_parser("build", help="그래프 빌드 (full/update/cluster-only)")
    p.add_argument("--mode", default="normal", choices=("normal", "deep"))
    p.add_argument("--update", action="store_true", help="증분 빌드")
    p.add_argument("--cluster-only", action="store_true", help="기존 그래프 클러스터링만")
    p.add_argument("--directed", action="store_true", help="방향 그래프")
    p.add_argument("--no-viz", action="store_true", help="HTML 시각화 생략")
    p.set_defaults(func=cmd_build)

    # check
    p = sub.add_parser("check", help="graph.json 존재 확인")
    p.set_defaults(func=cmd_check)

    # query
    p = sub.add_parser("query", help="자연어 질문 (BFS 기본)")
    p.add_argument("question")
    p.add_argument("--dfs", action="store_true", help="DFS 모드")
    p.add_argument("--budget", type=int, default=None, help="응답 토큰 상한")
    p.set_defaults(func=cmd_query)

    # path
    p = sub.add_parser("path", help="두 개념 간 최단 경로")
    p.add_argument("source")
    p.add_argument("target")
    p.set_defaults(func=cmd_path)

    # explain
    p = sub.add_parser("explain", help="단일 노드 설명")
    p.add_argument("symbol")
    p.set_defaults(func=cmd_explain)

    # export
    p = sub.add_parser("export", help="내보내기")
    p.add_argument("--html", action="store_true", default=False)
    p.add_argument("--svg", action="store_true")
    p.add_argument("--graphml", action="store_true")
    p.add_argument("--obsidian", action="store_true")
    p.add_argument("--obsidian-dir", default=None)
    p.add_argument("--wiki", action="store_true")
    p.add_argument("--neo4j", action="store_true")
    p.add_argument("--falkordb", action="store_true")
    p.set_defaults(func=cmd_export)

    # mcp
    p = sub.add_parser("mcp", help="MCP stdio 서버")
    p.add_argument("--watch", action="store_true")
    p.set_defaults(func=cmd_mcp)

    return ap


def main() -> int:
    ap = build_parser()
    args = ap.parse_args()
    if args.command in ("query", "path", "explain", "export", "mcp"):
        fast_path_hint()
    return args.func(args)


if __name__ == "__main__":
    sys.exit(main())
```
