# Phase 2 백엔드 확장 — 전체 설계 (승인 반영)

Phase 2 전체 승인에 따라, **구현 착수 전 완전한 설계**를 단계별로 정리했습니다. 이 설계가 확정되면 G1(데이터 구조)부터 순차 구현합니다.

---

## 0. 설계 개요

### 0.1 범위

| 구분 | 대상 | 방식 |
|---|---|---|
| **데이터 구조** | enum/struct 정의 | 헤더 확장 및 신규 |
| **C10 로더/저장** | profile/schedule `rule` 파싱·직렬화 | 단일 규격 (마이그레이션 배제) |
| **CT10 FSM** | `checkTriggers()` 리팩토링, `decideRunSource()` 우선순위 | Pause/Stop 이원화 |
| **AUTO 모드** | `AUTO_FULL` / `AUTO_SELECT` 세션 | 신규 위임 로직 |
| **API** | `/session/start` payload 확장 | 하위 호환 배제 |
| **계약 문서** | 6문서 동시 갱신 | 원칙 4 준수 |

### 0.2 설계 원칙

1. **하위 호환 배제 (Clean Break)**: `autoOff` → `rule` 단일 스키마, 레거시 분기 없음
2. **`ENV_PAUSED` 단일화**: `MOTION_BLOCKED` 완전 폐기
3. **온도/DI 이원화**: `mode = PAUSE | STOP` 사용자 선택
4. **세션 시작 시 규칙 스냅샷**: `_session.rule`에 복사 후 참조 (일관성)
5. **계약 문서 동시 갱신**: 코드 변경 = 6문서 원자 커밋

static void C10_fromJson_TriggerRule(const JsonObjectConst& p_js,
                                      ST_A20_TriggerRule_t& p_out,
                                      const char* p_caller) {
    // timer
    JsonObjectConst t = p_js["timer"].as<JsonObjectConst>();
    p_out.timer.enabled = A40_ComFunc::Json_getBool(t, "enabled", false);
    p_out.timer.minutes = A40_ComFunc::Json_getNum<uint32_t>(t, "minutes", 0);
    
    // offTime
    JsonObjectConst ot = p_js["offTime"].as<JsonObjectConst>();
    p_out.offTime.enabled = A40_ComFunc::Json_getBool(ot, "enabled", false);
    A40_ComFunc::Json_copyStr(ot, "time", p_out.offTime.time,
                              sizeof(p_out.offTime.time), "00:00", p_caller);
    
    // tempLow
    JsonObjectConst tl = p_js["tempLow"].as<JsonObjectConst>();
    p_out.tempLow.enabled = A40_ComFunc::Json_getBool(tl, "enabled", false);
    p_out.tempLow.mode = C10_parseRuleMode(A40_ComFunc::Json_getStr(tl, "mode", "PAUSE"));
    p_out.tempLow.threshold = A40_ComFunc::Json_getNum<float>(tl, "threshold", 22.0f);
    p_out.tempLow.hysteresis = A40_ComFunc::Json_getNum<float>(tl, "hysteresis", 0.5f);
    
    // 검증: mode=PAUSE 시 hysteresis 최소값
    if (p_out.tempLow.mode == EN_A20_RULE_MODE_PAUSE &&
        p_out.tempLow.hysteresis < 0.5f) {
        CL_D10_Logger::log(EN_L10_LOG_WARN,
            "[C10][%s] tempLow.hysteresis < 0.5 clamped to 0.5", p_caller);
        p_out.tempLow.hysteresis = 0.5f;
    }
    
    // diLow, pirPause, smartComfort 동일 패턴
    // ...
}

static EN_A20_ruleMode_t C10_parseRuleMode(const char* p_str) {
    if (!p_str || !p_str[0]) return EN_A20_RULE_MODE_PAUSE;
    if (strcasecmp(p_str, "STOP") == 0) return EN_A20_RULE_MODE_STOP;
    return EN_A20_RULE_MODE_PAUSE;
}

void CL_CT10_ControlManager::initRuleFromProfile(const ST_A20_UserProfileItem_t& p_up) {
    memset(&triggerRt, 0, sizeof(triggerRt));
    _persistOffTimeLastYday = -1;
    
    // 세션 규칙 스냅샷
    _session.rule = p_up.rule;
    
    // Runtime 초기화
    triggerRt.timerArmed   = _session.rule.timer.enabled;
    triggerRt.timerStartMs = millis();
    triggerRt.timerMinutes = _session.rule.timer.minutes;
    triggerRt.offTimeEnabled = _session.rule.offTime.enabled;
    triggerRt.offTimeMinutes = A40_parseHHMMtoMin_24h(_session.rule.offTime.time);
    triggerRt.tempPaused = false;
    triggerRt.diPaused   = false;
}

void CL_CT10_ControlManager::initRuleFromSchedule(const ST_A20_ScheduleItem_t& p_s) {
    if (p_s.ruleOverride) {
        _session.rule = p_s.rule;
    } else if (g_A20_config_root.userProfiles) {
        // 프로파일 규칙 상속
        auto& profs = *g_A20_config_root.userProfiles;
        for (uint8_t i = 0; i < profs.count; ++i) {
            if (profs.items[i].profileId == p_s.profileId) {
                _session.rule = profs.items[i].rule;
                break;
            }
        }
    }
    // 이후 triggerRt는 initRuleFromProfile과 동일 로직
    // (공통 부분 헬퍼로 추출 가능)
    memset(&triggerRt, 0, sizeof(triggerRt));

    triggerRt.timerArmed   = _session.rule.timer.enabled;
    triggerRt.timerStartMs = millis();
    triggerRt.timerMinutes = _session.rule.timer.minutes;

// ═══════════════════════════════════════════════════════════
// 통합 트리거 검사 (Pause / Stop 이원화)
//  - 호출: tickLoop → checkTriggers(&pauseOut, &stopOut)
//  - 반환: pauseOut != NONE → ENV_PAUSED 진입
//          stopOut  != NONE → AUTOOFF_STOPPED 진입
//  - 우선순위: Stop > Pause (Stop이 우선)
// ═══════════════════════════════════════════════════════════
bool CL_CT10_ControlManager::checkTriggers(EN_CT10_pause_t* p_pauseOut,
                                            EN_CT10_reason_t* p_stopOut) {
    if (p_pauseOut) *p_pauseOut = EN_CT10_PAUSE_NONE;
    if (p_stopOut)  *p_stopOut  = EN_CT10_REASON_NONE;
    
    const ST_A20_TriggerRule_t& r = _session.rule;
    uint32_t now = millis();
    
    // ═══════════════ 1) STOP 검사 (우선순위 최상) ═══════════════
    
    // 1-A. Timer
    if (triggerRt.timerArmed && triggerRt.timerMinutes > 0) {
        uint32_t elapsedMin = (now - triggerRt.timerStartMs) / 60000UL;
        if (elapsedMin >= triggerRt.timerMinutes) {
            if (p_stopOut) *p_stopOut = EN_CT10_REASON_AUTOOFF_TIMER;
            CL_D10_Logger::log(EN_L10_LOG_INFO,
                "[CT10] STOP: Timer %u min expired", triggerRt.timerMinutes);
            return true;
        }
    }
    
    // 1-B. OffTime
    if (triggerRt.offTimeEnabled) {
        struct tm tmv;
        if (CL_TM10_TimeManager::getLocalTime(tmv)) {
            int16_t yday = tmv.tm_yday;
            uint16_t curMin = tmv.tm_hour * 60 + tmv.tm_min;
            if (curMin >= triggerRt.offTimeMinutes && _persistOffTimeLastYday != yday) {
                _persistOffTimeLastYday = yday;
                if (p_stopOut) *p_stopOut = EN_CT10_REASON_AUTOOFF_TIME;
                return true;
            }
        }
    }
    
    // 1-C. TempLow (mode=STOP)
    if (r.tempLow.enabled && r.tempLow.mode == EN_A20_RULE_MODE_STOP) {
        float t = getCurrentTemperatureMock();
        if (t < r.tempLow.threshold) {
            if (p_stopOut) *p_stopOut = EN_CT10_REASON_TEMP_LOW_STOP;
            CL_D10_Logger::log(EN_L10_LOG_INFO,
                "[CT10] STOP: tempLow %.2f < %.2f", t, r.tempLow.threshold);
            return true;
        }
    }
    
    // 1-D. DiLow (mode=STOP)
    if (r.diLow.enabled && r.diLow.mode == EN_A20_RULE_MODE_STOP) {
        float di = getDiscomfortIndex();
        if (di < r.diLow.threshold) {
            if (p_stopOut) *p_stopOut = EN_CT10_REASON_DI_LOW_STOP;
            return true;
        }
    }
    
    // ═══════════════ 2) PAUSE 검사 ═══════════════
    
    // 2-A. PIR
    if (r.pirPause.enabled && motion) {
        if (!motion->isActive()) {
            if (p_pauseOut) *p_pauseOut = EN_CT10_PAUSE_MOTION;
            return true;
        }
    }
    
    // 2-B. TempLow (mode=PAUSE) — 히스테리시스
    if (r.tempLow.enabled && r.tempLow.mode == EN_A20_RULE_MODE_PAUSE) {
        float t = getCurrentTemperatureMock();
        float stopTh = r.tempLow.threshold;
        float resumeTh = stopTh + r.tempLow.hysteresis;
        
        if (!triggerRt.tempPaused && t < stopTh) {
            triggerRt.tempPaused = true;
            if (p_pauseOut) *p_pauseOut = EN_CT10_PAUSE_TEMP_LOW;
            return true;
        }
        // 재개는 decideRunSource에서 triggerRt.tempPaused=false로 판정
    }
    
    // 2-C. DiLow (mode=PAUSE) — 히스테리시스
    if (r.diLow.enabled && r.diLow.mode == EN_A20_RULE_MODE_PAUSE) {
        float di = getDiscomfortIndex();
        float stopTh = r.diLow.threshold;
        
        if (!triggerRt.diPaused && di < stopTh) {
            triggerRt.diPaused = true;
            if (p_pauseOut) *p_pauseOut = EN_CT10_PAUSE_DI_LOW;
            return true;
        }
    }
    
    return false;
}

// ═══════════════════════════════════════════════════════════
// Pause 재개 조건 검사 (decideRunSource에서 호출)
// ═══════════════════════════════════════════════════════════
static bool isPauseResumed(const ST_A20_TriggerRule_t& r,
                            ST_CT10_TriggerRuntime_t& rt,
                            EN_CT10_pause_t pauseReason,
                            CL_CT10_ControlManager& ctrl) {
    switch (pauseReason) {
        case EN_CT10_PAUSE_MOTION:
            return ctrl.motion && ctrl.motion->isActive();
        
        case EN_CT10_PAUSE_TEMP_LOW: {
            float t = ctrl.getCurrentTemperatureMock();
            float resumeTh = r.tempLow.threshold + r.tempLow.hysteresis;
            if (t >= resumeTh) {
                rt.tempPaused = false;
                return true;
            }
            return false;
        }
        case EN_CT10_PAUSE_DI_LOW: {
            float di = ctrl.getDiscomfortIndex();
            float resumeTh = r.diLow.threshold + r.diLow.hysteresis;
            if (di >= resumeTh) {
                rt.diPaused = false;
                return true;
            }
            return false;
        }
        default:
            return true;
    }
}

void CL_CT10_ControlManager::onTriggerStop(EN_CT10_reason_t reason) {
    sim.stop();
    runSource = EN_CT10_RUN_NONE;
    curScheduleIndex = -1;
    curProfileIndex  = -1;
    scheduleSegRt.index = -1;
    profileSegRt.index  = -1;
    
    uint32_t now = millis();
    runCtx.state = EN_CT10_STATE_AUTOOFF_STOPPED;
    runCtx.reason = reason;
    runCtx.lastDecisionMs = now;
    runCtx.lastStateChangeMs = now;
    runCtx.stateHoldUntilMs = now + S_EVENT_HOLD_MS;
    runCtx.stateAckRequired = false;   // 사용자 ACK는 상위에서 요구
    
    _autoOffLatched = true;
    markDirty("state");
    markDirty("metrics");
    markDirty("summary");
    
    CL_D10_Logger::log(EN_L10_LOG_INFO,
        "[CT10] STOP triggered (reason=%u, latched=true)", reason);
}

ST_CT10_Decision_t CL_CT10_ControlManager::decideRunSource() {
    ST_CT10_Decision_t d;
    memset(&d, 0, sizeof(d));
    
    // [1] Override 최우선
    if (overrideState.active) {
        d.nextState = EN_CT10_STATE_OVERRIDE;
        d.reason = EN_CT10_REASON_OVERRIDE_ACTIVE;
        return d;
    }
    
    // [2] AutoOff 래치
    if (_autoOffLatched) {
        d.nextState = EN_CT10_STATE_AUTOOFF_STOPPED;
        d.reason = runCtx.reason;
        d.wantSimStop = true;
        return d;
    }
    
    // [3] ENV_PAUSED 재개 판정 (신규)
    if (runCtx.state == EN_CT10_STATE_ENV_PAUSED) {
        if (!isPauseResumed(_session.rule, triggerRt,
                            runCtx.pauseReason, *this)) {
            d.nextState = EN_CT10_STATE_ENV_PAUSED;
            d.reason = runCtx.reason;
            d.wantSimStop = true;
            return d;
        }
        // 재개: 아래 로직으로 진행
        CL_D10_Logger::log(EN_L10_LOG_INFO,
            "[CT10] PAUSE resumed (was reason=%u)", runCtx.pauseReason);
    }
    
    // [4] 새 Pause/Stop 조건 검사 (신규)
    EN_CT10_pause_t pauseOut = EN_CT10_PAUSE_NONE;
    EN_CT10_reason_t stopOut = EN_CT10_REASON_NONE;
    if (checkTriggers(&pauseOut, &stopOut)) {
        if (stopOut != EN_CT10_REASON_NONE) {
            d.nextState = EN_CT10_STATE_AUTOOFF_STOPPED;
            d.reason = stopOut;
            d.wantSimStop = true;
            return d;
        }
        if (pauseOut != EN_CT10_PAUSE_NONE) {
            d.nextState = EN_CT10_STATE_ENV_PAUSED;
            // pauseReason은 applyDecision에서 설정
            d.wantSimStop = true;
            return d;
        }
    }
    
    // [5] ProfileMode (기존과 동일, MOTION_BLOCKED 제거)
    if (useProfileMode && runSource == EN_CT10_RUN_USER_PROFILE && curProfileIndex >= 0) {
        d.nextState = EN_CT10_STATE_PROFILE_RUN;
        d.reason = EN_CT10_REASON_PROFILE_MODE;
        d.nextRunSource = EN_CT10_RUN_USER_PROFILE;
        d.nextProfileIndex = curProfileIndex;
        return d;
    }
    
    // [6] TimeInvalid
    if (!CL_TM10_TimeManager::isTimeValid()) {
        d.nextState = EN_CT10_STATE_TIME_INVALID;
        d.reason = EN_CT10_REASON_TIME_NOT_VALID;
        d.wantSimStop = true;
        return d;
    }
    
    // [7] Schedule
    if (g_A20_config_root.schedules) {
        int idx = findActiveScheduleIndex(*g_A20_config_root.schedules, true);
        if (idx >= 0) {
            d.nextState = EN_CT10_STATE_SCHEDULE_RUN;
            d.reason = EN_CT10_REASON_SCHEDULE_ACTIVE;
            d.nextRunSource = EN_CT10_RUN_SCHEDULE;
            d.nextScheduleIndex = idx;
            return d;
        }
    }
    
    // [8] IDLE
    d.nextState = EN_CT10_STATE_IDLE;
    d.reason = EN_CT10_REASON_NONE;
    d.wantSimStop = true;
    return d;
}

void CL_CT10_ControlManager::tickAutoComfort() {
    if (!s_autoComfortEnabled) return;
    if (millis() - s_autoComfortLastEvalMs < AUTO_COMFORT_INTERVAL_MS) return;
    s_autoComfortLastEvalMs = millis();
    
    // AUTO_FULL 세션 재평가
    if (_session.type == EN_CT10_SESSION_AUTO_FULL &&
        _session.state == EN_SESSION_ACTIVE) {
        const char* recPreset = getRecommendedPresetCode();
        const char* recStyle  = getRecommendedStyleCode();
        
        // 부드러운 전환: crossfade (S10에서 처리)
        if (strcasecmp(_session.payload.preset.presetCode, recPreset) != 0) {
            CL_D10_Logger::log(EN_L10_LOG_INFO,
                "[CT10] AUTO_FULL switch: %s → %s", 
                _session.payload.preset.presetCode, recPreset);
            instance().startOverridePreset(recPreset, recStyle, nullptr, 0, true);
        }
        return;
    }
    
    // AUTO_SELECT 세션 재평가 (styleCode만 자동, presetCode 고정)
    if (_session.type == EN_CT10_SESSION_AUTO_SELECT &&
        _session.state == EN_SESSION_ACTIVE) {
        const char* newStyle = getRecommendedStyleCode();
        
        if (_session.autoSelect.baseType == EN_CT10_SESSION_PRESET) {
            // styleCode만 갱신
            if (strcasecmp(_session.payload.preset.styleCode, newStyle) != 0) {
                CL_D10_Logger::log(EN_L10_LOG_INFO,
                    "[CT10] AUTO_SELECT style: %s → %s",
                    _session.payload.preset.styleCode, newStyle);
                instance().startOverridePreset(
                    _session.autoSelect.basePresetCode, newStyle, nullptr, 0, true);
            }
        }
        // PLAYLIST base는 step 진행 중 styleCode 오버라이드 (S10에서 처리)
        return;
    }
    
    // 기존 Auto Comfort (스위치 모드) — 유지
    if (instance().isSessionActive()) return;
    if (runCtx.state == EN_CT10_STATE_AUTOOFF_STOPPED || _autoOffLatched) return;
    
    const char* recPreset = getRecommendedPresetCode();
    if (recPreset && strlen(recPreset) > 0) {
        instance().startPresetManual(recPreset, getRecommendedStyleCode());
        instance()._session.trigger = EN_CT10_TRIGGER_AUTO_COMFORT;
    }
}
