```cpp
bool CL_CT10_ControlManager::feedPIR(bool p_detected) {
    CL_A40_MutexGuard_Semaphore v_guard(s_stateMutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_guard.isAcquired()) {
        CL_D10_Logger::log(EN_L10_LOG_DEBUG, "[CT10] feedPIR: Mutex busy");

    }
    if (!motion) return false;
```

```cpp
// ======================================================
// [B] SemVer 비교 헬퍼 (x.y.z 형식)
//  - 반환: a < b → -1 / a == b → 0 / a > b → 1
//  - 파싱 실패 시 0 반환 (안전: 같음으로 판정 → "latest" 응답)
// ======================================================
static int _w10_compareSemVer(const char* p_a, const char* p_b) {
    if (!p_a || !p_b) return 0;

    int v_amaj = 0, v_amin = 0, v_apat = 0;
    int v_bmaj = 0, v_bmin = 0, v_bpat = 0;

    if (sscanf(p_a, "%d.%d.%d", &v_amaj, &v_amin, &v_apat) != 3) return 0;
    if (sscanf(p_b, "%d.%d.%d", &v_bmaj, &v_bmin, &v_bpat) != 3) return 0;

    if (v_amaj != v_bmaj) return (v_amaj < v_bmaj) ? -1 : 1;
    if (v_amin != v_bmin) return (v_amin < v_bmin) ? -1 : 1;
    if (v_apat != v_bpat) return (v_apat < v_bpat) ? -1 : 1;
    return 0;
}
```

```cpp
        int v_cmp = _w10_compareSemVer(v_current_semver, v_latest_semver);
        if (v_cmp < 0) {
            v_doc["status"]         = "available";
            v_doc["latest_version"] = v_latest_version;
            v_doc["url"]            = "/api/v001/fwUpdate";
        } else {
            v_doc["status"]         = "latest";
            v_doc["latest_version"] = v_current_version;
        }

        sendJson(p_request, v_doc);
    });
}
```

```cpp
// [D-1] 배치 소비 구현
//  - WS tick당 mutex 1회 획득으로 4개 dirty 플래그 원자 소비
//  - bit0=state, bit1=metrics, bit2=chart, bit3=summary

uint8_t CL_CT10_ControlManager::consumeAllDirty() {
    CL_A40_MutexGuard_Semaphore v_guard(s_stateMutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_guard.isAcquired()) return 0;

    uint8_t v_ret = 0;
    if (_dirtyState)   { v_ret |= 0x01; _dirtyState   = false; }
    if (_dirtyMetrics) { v_ret |= 0x02; _dirtyMetrics = false; }
    if (_dirtyChart)   { v_ret |= 0x04; _dirtyChart   = false; }
    if (_dirtySummary) { v_ret |= 0x08; _dirtySummary = false; }
    return v_ret;
}
```

```cpp
bool CL_WF10_WiFiManager::isStaConnected() {
    // [C-1] timeout 10ms: applyConfig 재연결 중에도 LED 폴링이 정상 반환
    //  - 0ms는 mutex 보유 중 즉시 false 반환 → LED 오표시(빨강)
    //  - s_staConnected 값은 원자적 읽기로도 안전하나, WiFi.status()까지
    //    일관 조회를 위해 mutex 획득 유지
    CL_A40_MutexGuard_Semaphore v_guard(s_wifiMutex, 10, __func__);
    if (!v_guard.isAcquired()) {
        // mutex 미획득 시 stale 값이라도 반환 (LED 빨강 오표시 방지)
        return s_staConnected;
    }
    return s_staConnected && (WiFi.status() == WL_CONNECTED);
}
```

```cpp
// ======================================================
// [E-2.1] _isTimeSane
// ======================================================
bool CL_TM10_TimeManager::_isTimeSane() {
    time_t v_now = time(nullptr);
    if ((uint32_t)v_now < A20_Const::G_TIME_EPOCH_SANE_MIN) return false;

    struct tm v_tm;
    memset(&v_tm, 0, sizeof(v_tm));

    if (localtime_r(&v_now, &v_tm) == nullptr) return false;

    int v_year = v_tm.tm_year + 1900;
    if (v_year < A20_Const::G_TIME_YEAR_MIN || v_year > A20_Const::G_TIME_YEAR_MAX) return false;

    return true;
}

// ======================================================
// [E-2.1] _onSntpTimeSync (SNTP 콜백)
// ======================================================
void CL_TM10_TimeManager::_onSntpTimeSync(struct timeval* p_tv) {
    (void)p_tv;

    if (!s_mutex) return;
    if (xSemaphoreTakeRecursive(s_mutex, 0) != pdTRUE) return;

    bool v_ok = _isTimeSane();
    if (v_ok) {
        s_timeValid       = true;
        s_waitingCallback = false;
        s_lastSyncMs      = millis();

        _updateSyncedRecently(millis());

        CL_D10_Logger::log(EN_L10_LOG_INFO,
                           "[TM10] SNTP synced OK (server=%s)",
                           s_serverList[s_activeServerIdx]);
    } else {
        s_timeValid = false;
        CL_D10_Logger::log(EN_L10_LOG_WARN,
                           "[TM10] SNTP callback but time is not sane (server=%s)",
                           s_serverList[s_activeServerIdx]);
    }

    xSemaphoreGiveRecursive(s_mutex);
}

// ======================================================
// [E-2.1] _stopSntp / _startSntpWithServer
// ======================================================
void CL_TM10_TimeManager::_stopSntp() {
    if (!s_running) return;
    sntp_stop();
    s_running = false;
}

void CL_TM10_TimeManager::_startSntpWithServer(uint8_t p_idx) {
    if (p_idx >= 4) p_idx = 0;

    setenv("TZ", s_tz, 1);
    tzset();

    _stopSntp();

    sntp_setoperatingmode(SNTP_OPMODE_POLL);
    sntp_setservername(0, s_serverList[p_idx]);
    sntp_set_time_sync_notification_cb(&CL_TM10_TimeManager::_onSntpTimeSync);
    sntp_set_sync_interval((uint32_t)s_syncIntervalMs);
    sntp_init();

    s_running         = true;
    s_activeServerIdx = p_idx;
    s_waitingCallback = true;
    s_waitStartMs     = millis();
    s_lastRequestMs   = s_waitStartMs;

    CL_D10_Logger::log(EN_L10_LOG_INFO,
                       "[TM10] SNTP start: tz=%s, server=%s, intervalMs=%lu",
                       s_tz,
                       s_serverList[p_idx],
                       (unsigned long)s_syncIntervalMs);
}

// ======================================================
// [E-2.1] _requestSync / _switchToNextServer
// ======================================================
void CL_TM10_TimeManager::_requestSync() {
    _startSntpWithServer(s_activeServerIdx);
}

void CL_TM10_TimeManager::_switchToNextServer() {
    uint8_t v_next = (uint8_t)(s_activeServerIdx + 1);
    if (v_next >= 4) v_next = 0;

    CL_D10_Logger::log(EN_L10_LOG_WARN,
                       "[TM10] NTP fallback switch: %s -> %s",
                       s_serverList[s_activeServerIdx],
                       s_serverList[v_next]);

    _startSntpWithServer(v_next);
}

// [E-2.1] begin
// ======================================================
void CL_TM10_TimeManager::begin() {
    CL_A40_MutexGuard_Semaphore v_guard(s_mutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_guard.isAcquired()) {
        CL_D10_Logger::log(EN_L10_LOG_DEBUG, "[TM10] %s: Mutex busy", __func__);
        return;
    }

    // 멱등 가드: SNTP 진행 중이거나 WiFi up 상태면 상태 리셋 금지
    if (s_running || s_wifiUp) {
        CL_D10_Logger::log(EN_L10_LOG_DEBUG,
                           "[TM10] begin ignored (running=%d wifiUp=%d)",
                           (int)s_running, (int)s_wifiUp);
        return;
    }

    _setDefaults();

    s_timeValid       = _isTimeSane();
    s_wifiUp          = false;
    s_lastSyncMs      = 0;
    s_lastRequestMs   = 0;
    s_activeServerIdx = 0;
    s_running         = false;
    s_waitingCallback = false;
    s_waitStartMs     = 0;
    s_nextActionMs    = 0;

    CL_D10_Logger::log(EN_L10_LOG_INFO, "[TM10] begin (timeValid=%d)", (int)s_timeValid);
}

// ======================================================
// [E-2.1] tick
// ======================================================
void CL_TM10_TimeManager::tick(const ST_A20_SystemConfig_t* p_sysOrNull) {
    CL_A40_MutexGuard_Semaphore v_guard(s_mutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_guard.isAcquired()) {
        CL_D10_Logger::log(EN_L10_LOG_DEBUG, "[TM10] %s: Mutex busy", __func__);
        return;
    }

    uint32_t v_now = millis();

    _updateSyncedRecently(v_now);

    if (!s_wifiUp) return;

    if (p_sysOrNull) {
        uint32_t v_ms = (uint32_t)p_sysOrNull->timeCfg.syncIntervalMin * 60000UL;
        if (v_ms == 0) v_ms = (6UL * 60UL * 60UL * 1000UL);
        v_ms = A40_ComFunc::clampVal(v_ms,
                                     (uint32_t)G_TM10_MIN_SYNC_INTERVAL_MS,
                                     (uint32_t)G_TM10_MAX_SYNC_INTERVAL_MS);
        s_syncIntervalMs = v_ms;
        sntp_set_sync_interval((uint32_t)s_syncIntervalMs);
    }

    if (s_waitingCallback) {
        if (s_nextActionMs != 0 && v_now < s_nextActionMs) return;

        if (v_now - s_waitStartMs >= (uint32_t)G_TM10_SYNC_TIMEOUT_MS) {
            s_nextActionMs = v_now + (uint32_t)G_TM10_RETRY_BACKOFF_MS;
            _switchToNextServer();
            return;
        }
    }

    if (s_timeValid) {
        if (!_isTimeSane()) {
            s_timeValid       = false;
            s_waitingCallback = true;
            s_waitStartMs     = v_now;
            s_nextActionMs    = v_now + (uint32_t)G_TM10_RETRY_BACKOFF_MS;

            _requestSync();
            return;
        }
    }
}

    CL_A40_MutexGuard_Semaphore v_guard(s_mutex, G_A40_MUTEX_TIMEOUT_100, __func__);

        CL_D10_Logger::log(EN_L10_LOG_DEBUG, "[TM10] %s: Mutex busy", __func__);
        return false;
    }

}

bool CL_TM10_TimeManager::isTimeSyncedRecently(uint32_t p_windowMs) {
    CL_A40_MutexGuard_Semaphore v_guard(s_mutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_guard.isAcquired()) {
        CL_D10_Logger::log(EN_L10_LOG_DEBUG, "[TM10] %s: Mutex busy", __func__);
        return false;
    }

    uint32_t v_win = (p_windowMs == 0) ? (uint32_t)G_TM10_SYNC_RECENT_WINDOW_MS : p_windowMs;
    if (s_lastSyncMs == 0) return false;

    uint32_t v_now = millis();
    if (v_now >= s_lastSyncMs && (v_now - s_lastSyncMs) <= v_win) return true;

    return false;
}

bool CL_TM10_TimeManager::getLocalTime(struct tm& p_outTm) {
    memset(&p_outTm, 0, sizeof(p_outTm));

    CL_A40_MutexGuard_Semaphore v_guard(s_mutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_guard.isAcquired()) {
        CL_D10_Logger::log(EN_L10_LOG_DEBUG, "[TM10] %s: Mutex busy", __func__);
        return false;
    }

    if (!_isTimeSane()) {
        s_timeValid          = false;
        s_timeSyncedRecently = false;
        return false;
    }

    time_t v_now = time(nullptr);
    if (v_now <= 0) {
        s_timeValid          = false;
        s_timeSyncedRecently = false;
        return false;
    }

    if (localtime_r(&v_now, &p_outTm) == nullptr) {
        s_timeValid          = false;
        s_timeSyncedRecently = false;
        return false;
    }

    s_timeValid = true;
    _updateSyncedRecently(millis());

    return true;
}

// ======================================================
// [E-2.1] _updateSyncedRecently
// ======================================================
void CL_TM10_TimeManager::_updateSyncedRecently(uint32_t p_nowMs) {
    if (s_lastSyncMs == 0) {
        s_timeSyncedRecently = false;
        return;
    }

    uint32_t v_win = (uint32_t)G_TM10_SYNC_RECENT_WINDOW_MS;

    if (p_nowMs >= s_lastSyncMs && (p_nowMs - s_lastSyncMs) <= v_win) {
        s_timeSyncedRecently = true;
    } else {
        s_timeSyncedRecently = false;
    }
}
```
