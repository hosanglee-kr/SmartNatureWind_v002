

Gemini API 제어 가능 기능 검토

1. 현재 구현 상태 진단

W10_Web_Gemini_0202.cpp의 routeGeminiProxy()는 순수 프록시(pass-through) 입니다:

```
프론트 → POST /ai/gemini → Gemini API → 응답 그대로 반환
```

- ❌ Function Calling 미지원: tools / function_declarations 필드 없음
- ❌ 도구 실행 없음: ESP32가 응답을 파싱해 CT10을 호출하는 경로 없음
- ❌ 대화 컨텍스트 없음: 매 요청마다 WiFiClientSecure 재연결 → stateless

즉, 현재는 Gemini가 "말"만 하고 ESP32는 아무것도 제어하지 않습니다.

---

2. Gemini가 제어 가능한 기능 (CT10 API 기준)

현재 CT10/W10이 노출하는 실제 제어 API를 정리하면 다음과 같습니다.

| 카테고리 | CT10 메서드 / W10 엔드포인트 | LLM 자연어 명령 예시 | 실용성 |
|---|---|---|---|
| 바람 고정 | startOverrideFixed(pct, sec, forever) / POST /control/override/fixed | "5분간 60%로" | ⭐⭐⭐⭐⭐ | 
| 프리셋 바람 | startOverridePreset(code, style, ovr, sec, forever) / POST /control/override/preset | "산들바람 10분간" | ⭐⭐⭐⭐⭐ | 
| 오버라이드 해제 | stopOverride() / POST /control/override/clear|  "바람 멈춰"|  ⭐⭐⭐⭐⭐|
| 시뮬레이션 조정 | patchSimFromJson() / POST /simulation|  "강도를 70으로"|  ⭐⭐⭐⭐|
| 프로필 시작 | startUserProfileByNo(no) / POST /control/profile/select|  "밤 프로필 실행"|  ⭐⭐⭐⭐|
| 프로필 정지 | stopUserProfile() / POST /control/profile/stop | "프로필 종료"|  ⭐⭐⭐⭐|
| 모드 전환 | setProfileMode(bool) / POST /control/mode | "스케줄 모드로" | ⭐⭐⭐|
| 상태 조회 | exportStateJson_v02() / GET /state|  "지금 상태?"|  ⭐⭐⭐⭐⭐|
| 이벤트 | ACK ackEvent() / POST /control/ack|  "확인" | ⭐⭐⭐|
| 재부팅 | ESP.restart() / POST /control/reboot|  "재부팅해" ⭐⭐ (위험)|
| 팩토리 리셋 | factoryResetFromDefault() / POST /control/factoryReset | "초기화" | ⭐ (매우 위험)|
| PIR 피드 | feedPIR(bool) / POST /motion/pir/feed|  (수동 개입) | ❌ (LLM 부적합)|
| 설정 CRUD | patchSystemFromJson 등 | "밝기 80으로" | ⭐⭐ (검증 부담)|

3. 현재 코드의 구조적 한계

(1) 요청 본문에 도구 정의 불가

```cpp
// 현재: body를 그대로 전송
v_client.print(p_body);
```

Gemini의 tools 스키마를 넣을 자리가 없습니다.

(2) 응답 파싱 없음

```cpp
if (v_httpCode == 200) {
    sendText(p_request, v_resp, 200, "application/json; charset=utf-8");
}
```

candidates[0].content.parts[].functionCall을 감지/실행하지 않습니다.

(3) Stateless 재연결

```cpp
WiFiClientSecure v_client;
v_client.setInsecure();
v_client.connect(...);  // 매 호출마다
v_client.stop();
```

멀티턴(functionCall → functionResponse → 최종 텍스트)을 위해서는 2회 왕복이 필요합니다.

(4) 응답 캡(4KB)이 빠듯

- Function call 응답 + 두 번째 라운드 응답까지 4KB 이내여야 함
- function_declarations 스키마를 포함하면 요청 본문도 커짐 (현재 8KB 캡)

4. 구현 방안: Gemini Native Function Calling

4-1. 요청 구조 (Phase 1 — 도구 정의 추가)

W10_callGeminiApi()를 다음과 같이 확장:

```cpp
static String W10_buildGeminiRequest(const char* p_userBody) {
    JsonDocument v_req;
    auto v_err = deserializeJson(v_req, p_userBody);
    // ... 검증 생략

    // ── 1. 시스템 지시문 ──
    v_req["system_instruction"]["parts"][0]["text"] =
        "You are a wind machine controller. "
        "Call tools ONLY when the user asks to change device state. "
        "Never invent tool arguments outside the schema.";

    // ── 2. 도구 선언 ──
    JsonArray v_tools = v_req["tools"].to<JsonArray>();
    JsonObject v_tool = v_tools.add<JsonObject>();
    JsonArray v_fns   = v_tool["function_declarations"].to<JsonArray>();

    // [A] set_fixed_wind
    {
        JsonObject v_fn = v_fns.add<JsonObject>();
        v_fn["name"]        = "set_fixed_wind";
        v_fn["description"] = "Set fan PWM to a fixed percent for a duration";
        JsonObject v_p      = v_fn["parameters"].to<JsonObject>();
        v_p["type"]         = "object";
        JsonObject v_props  = v_p["properties"].to<JsonObject>();
        v_props["percent"]["type"]        = "number";
        v_props["percent"]["minimum"]     = 0;
        v_props["percent"]["maximum"]     = 100;
        v_props["seconds"]["type"]        = "integer";
        v_props["seconds"]["minimum"]     = 1;
        v_props["seconds"]["maximum"]     = 7200;
        v_props["forever"]["type"]        = "boolean";
        JsonArray v_req_f = v_p["required"].to<JsonArray>();
        v_req_f.add("percent");
    }

    // [B] set_preset_wind
    {
        JsonObject v_fn = v_fns.add<JsonObject>();
        v_fn["name"]        = "set_preset_wind";
        v_fn["description"] = "Start a preset wind (e.g., BREEZE, GUST) with a style";
        JsonObject v_p      = v_fn["parameters"].to<JsonObject>();
        v_p["type"]         = "object";
        JsonObject v_props  = v_p["properties"].to<JsonObject>();
        v_props["presetCode"]["type"]        = "string";
        v_props["styleCode"]["type"]         = "string";
        v_props["seconds"]["type"]           = "integer";
        v_props["forever"]["type"]           = "boolean";
        JsonArray v_req_f = v_p["required"].to<JsonArray>();
        v_req_f.add("presetCode");
    }

    // [C] stop_wind
    {
        JsonObject v_fn = v_fns.add<JsonObject>();
        v_fn["name"]        = "stop_wind";
        v_fn["description"] = "Clear any override and stop wind";
    }

    // [D] select_profile
    {
        JsonObject v_fn = v_fns.add<JsonObject>();
        v_fn["name"]        = "select_profile";
        v_fn["description"] = "Start a user profile by its profileNo";
        JsonObject v_p      = v_fn["parameters"].to<JsonObject>();
        v_p["type"]         = "object";
        v_p["properties"]["profileNo"]["type"] = "integer";
        v_p["required"].to<JsonArray>().add("profileNo");
    }

    // [E] get_state
    {
        JsonObject v_fn = v_fns.add<JsonObject>();
        v_fn["name"]        = "get_state";
        v_fn["description"] = "Return current wind state, mode, override, and sensors";
    }

    // [F] set_mode
    {
        JsonObject v_fn = v_fns.add<JsonObject>();
        v_fn["name"]        = "set_mode";
        v_fn["description"] = "Switch between SCHEDULE and PROFILE mode";
        JsonObject v_p      = v_fn["parameters"].to<JsonObject>();
        v_p["type"]         = "object";
        v_p["properties"]["mode"]["type"] = "string";
        v_p["properties"]["mode"]["enum"].to<JsonArray>().add("SCHEDULE");
        v_p["properties"]["mode"]["enum"].add("PROFILE");
        v_p["required"].to<JsonArray>().add("mode");
    }

    String v_out;
    serializeJson(v_req, v_out);
    return v_out;
}
```

4-2. 응답 파싱 + 도구 실행 (Phase 2 — 핵심)

routeGeminiProxy()의 마지막 청크 처리부를 다음 흐름으로 교체:

```cpp
// ─── 1차: Gemini 호출 ───
String v_resp = W10_callGeminiApi(v_reqBody.c_str(), v_reqBody.length(), v_httpCode);
if (v_httpCode != 200) { /* 502 에러 */ }

// ─── 2차: functionCall 감지 ───
JsonDocument v_respDoc;
deserializeJson(v_respDoc, v_resp);

const char* v_fnName = v_respDoc["candidates"][0]["content"]["parts"][0]["functionCall"]["name"] | "";
JsonObject  v_fnArgs = v_respDoc["candidates"][0]["content"]["parts"][0]["functionCall"]["args"];

if (v_fnName[0] == '\0') {
    // 도구 호출 없음 → 최종 텍스트 그대로 반환
    sendText(p_request, v_resp, 200, "application/json; charset=utf-8");
    return;
}

// ─── 3차: 로컬 실행 (CT10 위임) ───
JsonDocument v_fnResult;   // functionResponse로 되돌릴 결과
bool v_ok = false;

if (strcmp(v_fnName, "set_fixed_wind") == 0) {
    float    pct     = v_fnArgs["percent"] | 0.0f;
    uint32_t sec     = v_fnArgs["seconds"] | 0;
    bool     forever = v_fnArgs["forever"] | false;
    s_control->startOverrideFixed(pct, sec, forever);
    v_fnResult["result"] = "started";
    v_fnResult["percent"] = pct;
    v_ok = true;

} else if (strcmp(v_fnName, "set_preset_wind") == 0) {
    const char* v_preset = v_fnArgs["presetCode"] | "";
    const char* v_style  = v_fnArgs["styleCode"]  | "BALANCE";
    uint32_t    sec      = v_fnArgs["seconds"]    | 0;
    bool        forever  = v_fnArgs["forever"]    | false;
    // override 없이 프리셋만
    s_control->startOverridePreset(v_preset, v_style, nullptr, sec, forever);
    v_fnResult["result"] = "started";
    v_ok = true;

} else if (strcmp(v_fnName, "stop_wind") == 0) {
    s_control->stopOverride();
    v_fnResult["result"] = "stopped";
    v_ok = true;

} else if (strcmp(v_fnName, "select_profile") == 0) {
    int no = v_fnArgs["profileNo"] | -1;
    v_ok = (no > 0) && s_control->startUserProfileByNo((uint16_t)no);
    v_fnResult["result"] = v_ok ? "profile_started" : "invalid_profile";
    v_fnResult["profileNo"] = no;

} else if (strcmp(v_fnName, "get_state") == 0) {
    s_control->exportStateJson_v02(v_fnResult);
    v_ok = true;

} else if (strcmp(v_fnName, "set_mode") == 0) {
    const char* v_mode = v_fnArgs["mode"] | "";
    bool v_profMode = (strcasecmp(v_mode, "PROFILE") == 0);
    CL_CT10_ControlManager::setMode(v_profMode);
    CL_N10_NvsManager::setRunMode(v_profMode ? 2 : 1, 2);
    v_fnResult["result"] = "mode_set";
    v_ok = true;
}

// ─── 4차: functionResponse 회신 → 자연어 최종 응답 ───
if (v_ok) {
    JsonDocument v_followup;
    // 원본 contents + functionResponse part 추가
    JsonArray v_contents = v_followup["contents"].to<JsonArray>();
    // ... (원본 user content 삽입)

    JsonObject v_fnRespPart = v_contents.add<JsonObject>()
        .add<JsonObject>("parts").add<JsonObject>().add<JsonObject>("functionResponse");
    v_fnRespPart["name"]   = v_fnName;
    v_fnRespPart["response"] = v_fnResult;

    String v_followBody;
    serializeJson(v_followup, v_followBody);

    int v_h2 = 0;
    String v_final = W10_callGeminiApi(v_followBody.c_str(), v_followBody.length(), v_h2);

    if (v_h2 == 200) {
        sendText(p_request, v_final, 200, "application/json; charset=utf-8");
    } else {
        sendText(p_request, v_resp, 200, "application/json; charset=utf-8"); // 함수 결과만
    }
} else {
    JsonDocument v_err;
    v_err["error"] = "function execution failed";
    v_err["fn"]    = v_fnName;
    sendJson(p_request, v_err, 502);
}
```

5. 현재 코드에서 반드시 손봐야 할 부분

| 문제 | 위치 | 조치 | 
|---|---|---|
| WiFiClientSecure v_client | W10_callGeminiApi() | 1차/2차 두 번 재연결 → 핸드셰이크 2배 비용. 세션 재사용 or 팩토리 함수화 | 
| G_W10_GEMINI_MAX_RESP = 4096 | 상수|  Function call 응답 + 도구 스키마를 합치면 4KB 초과 가능 → 8KB로 확대 권장| 
| 요청 본문 8KB 캡 | routeGeminiProxy() | tools 배열만 ~2KB → 12KB로 상향| 
| s_lastGeminiCallMs | 레이트 리밋 | 3초 → 5~10초로 늘리는 것이 안전 (function call 2회 왕복 감안)| 
| checkApiKey() | 현재 코드 | ⚠️ W10_USE_API_KEY=0이고 g_A20_config_root.system->security.apiKey가 비어있으면 인증 없이 누구나 LLM 제어 가능 → 최소한 프론트의 API Key 또는 별도 토큰 검증 필요| 
| 위험 명령 | reboot / factoryReset | ❌ LLM 도구로 노출 비권장 — 사람 확인 UI 필수| 
| setOverridePreset 호출 시 p_override = nullptr | 예제 ⚠️ startOverridePreset은 p_override가 nullptr이면 S20_resolveWindParams에서 내부 deref 위험 — ST_A20_SegmentOverride_t 명시 전달 필수|

6. 아키텍처 권장안

```
┌─────────────────────────────────────────────────────────────┐
│ POST /api/v001/ai/gemini                                    │
│  ├─ 1) checkApiKey()  ← 반드시 활성화                       │
│  ├─ 2) rate limit (5~10s)                                   │
│  ├─ 3) body 파싱 + tools 스키마 주입 (system_instruction)   │
│  ├─ 4) Gemini 1차 호출                                      │
│  ├─ 5) functionCall 검출                                    │
│  │    ├─ 없음 → 텍스트 그대로 반환                          │
│  │    └─ 있음 → s_control->* 실행 (mutex 보호)             │
│  │         ├─ reboot/factoryReset → **거부**                │
│  │         └─ 안전 도구 → 실행                              │
│  ├─ 6) functionResponse로 2차 호출 → 자연어 요약            │
│  └─ 7) 최종 응답                                            │
└─────────────────────────────────────────────────────────────┘
```

7. 결론

항목 평가
현재 Gemini 프록시의 제어 능력 0% — 순수 pass-through, ESP32는 아무것도 안 함
도구 정의만 추가하면 제어 가능한 범위 오버라이드/프로필/모드/상태 — CT10 API의 80% 커버
구현 난이도 중간 (JSON 스키마 작성 + 응답 파싱 + 2-round trip)
메모리 부담 요청 12KB / 응답 8KB 상향 필요 (heap 여유 확인)
응답 지연 1차+2차 왕복 → 5~8초 (15s 타임아웃 유지)
가장 실용적 도구 세트 set_fixed_wind, set_preset_wind, stop_wind, select_profile, get_state, set_mode
제외 권장 reboot, factoryReset, feedPIR, 설정 CRUD 전체
보안 필수 조치 API Key 검증 강제, 위험 명령은 화이트리스트로 차단

다음 단계 제안: 우선 W10_callGeminiApi()에 tools 스키마를 주입하고, 1차 응답에서 functionCall 존재 여부만 파싱해 로그로 확인 → 도구 실행은 별도 커밋으로 분리하여 단계적 적용을 권장합니다.

# Gemini API 기반 추가 가능 기능 검토

현재 시스템의 **CT10 제어 API + W10 REST + 센서 + 스케줄/프로필 구조**를 기반으로, Gemini가 실제로 부가가치를 창출할 수 있는 기능을 카테고리별로 정리합니다.

---

## 1. 자연어 제어 (기본 계층)

이미 앞서 논의한 Function Calling의 연장선. 구현 난이도 낮고 즉시 효과 큼.

| 기능 | 설명 | 필요 API | 난이도 |
|---|---|---|---|
| **자연어 오버라이드** | "5분간 60%로", "산들바람 10분" | `startOverrideFixed/Preset` | ⭐ |
| **자연어 정지** | "멈춰", "그만" | `stopOverride` | ⭐ |
| **자연어 프로필 실행** | "밤 프로필 실행해" | `startUserProfileByNo` | ⭐⭐ (프로필 이름→번호 매핑 필요) |
| **자연어 모드 전환** | "스케줄 모드로 돌아가" | `setMode` | ⭐ |
| **자연어 상태 조회** | "지금 뭐 하고 있어?" | `exportStateJson_v02` | ⭐ |
| **자연어 시뮬레이션 튜닝** | "강도 70, 변동성 30으로" | `patchSimFromJson` | ⭐⭐ |

**구현 포인트**: `system_instruction`에 "도구 호출만 하고 추측 금지" + 프로필 이름은 `userProfiles[].name → profileNo` 매핑을 프롬프트에 동적 주입.

---

## 2. 컨텍스트 인지 자동화

센서값과 현재 상태를 LLM에게 컨텍스트로 제공 → 상황 기반 제안/실행.

| 기능 | 설명 | 필요 데이터 | 난이도 |
|---|---|---|---|
| **"지금 더워?" 판단 후 제안** | DHT 온도/습도 → LLM 판단 → 팬 제어 | `sensor.tempC`, `sensor.humidity` | ⭐⭐ |
| **재실 감지 연동** | PIR `motionActive` → 부재 시 자동 감속/정지 제안 | `sensor.motionActive` | ⭐⭐ |
| **시간대별 추천** | 현재 `time.hour` 기반 "밤에는 조용한 바람" 제안 | `time.hour`, `time.wday` | ⭐⭐ |
| **AutoOff 사유 설명** | `AUTOOFF_STOPPED` 발생 시 자연어로 이유 설명 | `autoOffRt.*`, `reason` | ⭐⭐ |
| **에러 진단** | `diag.heap`, `fs_used` 부족 시 원인/조치 안내 | `/diag` 응답 | ⭐⭐⭐ |

**응답 예시**: "실내 온도 28.3°C, 습도 65%로 무덥습니다. 40% 강도로 5분간 가동할까요? [실행/취소]"

---

## 3. 스케줄/프로필 AI 생성 (⭐ 가장 임팩트 큼)

사용자가 "매일 아침 7시에 부드럽게 깨우는 바람" 같은 문장을 넣으면, LLM이 실제 JSON 구조를 생성해 CRUD API 호출.

| 기능 | 설명 | 필요 API | 난이도 |
|---|---|---|---|
| **자연어 → 스케줄 JSON** | "평일 아침 7~9시 산들바람 30분 반복" | `POST /schedules` | ⭐⭐⭐ |
| **자연어 → 프로필 JSON** | "자기 전 30분 백색소음 같은 규칙적 바람" | `POST /user_profiles` | ⭐⭐⭐ |
| **기존 스케줄 수정** | "그 스케줄 강도만 20 올려줘" | `PUT /schedules/{id}` | ⭐⭐⭐ |
| **충돌 감지** | 새 스케줄이 기존 스케줄과 겹치면 경고 | `GET /schedules` + LLM 판단 | ⭐⭐ |
| **JSON 스키마 검증** | LLM이 만든 JSON을 백엔드 스키마로 검증 후 거부 | `patchSchedulesFromJson` | ⭐⭐ |

**핵심 설계**: Gemini의 `responseSchema` (Structured Output)를 활용해 정확한 JSON 형태를 강제. `temperature=0.2` 권장.

```cpp

v_req["generationConfig"]["responseMimeType"] = "application/json";
v_req["generationConfig"]["responseSchema"] = ... // 스케줄 스키마
```

5. Phase 1에 반드시 반영할 7가지 (재강조)

① 응답 포맷 확장 (필수)

```json
{
  "candidates": [ /* Gemini 원본 그대로 */ ],

  "_functionCall": {
    "name": "set_fixed_wind",
    "args": { "percent": 60, "seconds": 300 }
  },

  "_functionResult": {
    "result": "started",
    "percent": 60,
    "seconds": 300
  },

  "_speechHint": "60% 강도로 5분간 가동합니다",

  "_lang": "ko"
}
```

· _functionCall, _functionResult: 프론트가 UI/TTS 조립용으로 사용
· _speechHint: ESP32가 로컬 생성 (Gemini 2차 호출 생략)
· Gemini 원본 candidates도 함께 반환 → 채팅 UI는 그대로 사용 가능

② _speechHint 로컬 생성 규칙 (백엔드 상수 테이블)

```cpp
static String W10_buildSpeechHint(const char* fn, JsonVariant args, bool ok) {
    if (!ok) return "명령을 실행하지 못했습니다";

    if (strcmp(fn, "set_fixed_wind") == 0) {
        int pct = args["percent"] | 0;
        int sec = args["seconds"] | 0;
        bool forever = args["forever"] | false;
        if (forever) return String(pct) + "% 강도로 계속 가동합니다";
        return String(pct) + "% 강도로 " + String(sec) + "초간 가동합니다";
    }
    if (strcmp(fn, "set_preset_wind") == 0) {
        return String(args["presetCode"] | "") + " 프리셋으로 시작합니다";
    }
    if (strcmp(fn, "stop_wind") == 0)  return "바람을 멈췄습니다";
    if (strcmp(fn, "select_profile") == 0)
        return "프로필 " + String(args["profileNo"] | 0) + "번을 시작합니다";
    if (strcmp(fn, "set_mode") == 0) {
        const char* m = args["mode"] | "";
        return String(strcmp(m, "PROFILE") == 0 ? "프로필" : "스케줄") + " 모드로 전환했습니다";
    }
    if (strcmp(fn, "get_state") == 0) return "상태를 조회했습니다";
    return "완료했습니다";
}
```

6-3. JS (핵심 로직)

```javascript
const AI_VOICE = {
  state: 'idle',                  // idle | listening | processing | speaking | error
  recognition: null,
  synth: window.speechSynthesis,
  apiKey: '',
  lang: localStorage.getItem('voiceLang') || 'ko-KR',

  init(apiKey) {
    this.apiKey = apiKey;
    this.initRecognition();
    this.initLangSelect();
    document.getElementById('voice-btn')
      .addEventListener('click', () => this.startListening());
  },

  initRecognition() {
    const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SR) {
      document.getElementById('voice-btn').style.display = 'none';
      console.warn('[voice] STT 미지원 브라우저');
      return;
    }
    this.recognition = new SR();
    this.recognition.lang = this.lang;
    this.recognition.continuous = false;
    this.recognition.interimResults = true;
    this.recognition.maxAlternatives = 1;

    this.recognition.onresult = (e) => {
      const r = e.results[e.results.length - 1];
      const t = r[0].transcript;
      this.showInterim(t);
      if (r.isFinal) this.onFinalTranscript(t);
    };
    this.recognition.onerror = (e) => this.onSTTError(e.error);
    this.recognition.onend   = () => { if (this.state === 'listening') this.setState('idle'); };
  },

  startListening() {
    if (this.state === 'speaking') this.synth.cancel();
    if (this.state === 'listening') return;
    this.recognition.lang = this.lang;
    this.setState('listening');
    try { this.recognition.start(); } catch (_) { /* already started */ }
  },

  async onFinalTranscript(text) {
    if (!text.trim()) return this.setState('idle');
    this.setState('processing');
    this.showTranscript(text, 'user');

    try {
      const resp = await fetch('/api/v001/ai/gemini', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'X-API-Key': this.apiKey },
        body: JSON.stringify({
          contents: [{ role: 'user', parts: [{ text }] }],
          voiceMode: true,
          lang: this.lang.slice(0, 2)
        })
      });
      const data = await resp.json();
      this.handleResponse(data);
    } catch (err) {
      console.error(err);
      this.speak('네트워크 오류가 발생했습니다');
    }
  },

  handleResponse(data) {
    // 1) 백엔드가 만들어준 TTS 힌트 우선 사용
    let speech = data._speechHint;

    // 2) 없으면 Gemini 최종 텍스트 사용
    if (!speech) {
      speech = data.candidates?.[0]?.content?.parts?.[0]?.text || '';
    }

    // 3) 여전히 없으면 폴백
    if (!speech) speech = '이해하지 못했어요. 다시 말해주세요.';

    this.showTranscript(speech, 'ai');
    this.speak(speech);
  },

  speak(text) {
    if (!this.synth) { this.setState('idle'); return; }
    this.setState('speaking');
    this.synth.cancel();
    const u = new SpeechSynthesisUtterance(text);
    u.lang = this.lang;
    u.rate = 1.0;
    u.pitch = 1.0;
    u.onend = () => this.setState('idle');
    u.onerror = () => this.setState('idle');
    this.synth.speak(u);
  },

  onSTTError(err) {
    this.setState('error');
    const map = {
      'not-allowed':     '마이크 권한이 필요합니다',
      'no-speech':       '다시 말해주세요',
      'network':         '네트워크 오류',
      'service-not-allowed': '브라우저 설정 확인',
    };
    this.speak(map[err] || '음성 인식 오류');
  },

  initLangSelect() {
    const sel = document.getElementById('voice-lang');
    sel.value = this.lang;
    sel.addEventListener('change', e => {
      this.lang = e.target.value;
      localStorage.setItem('voiceLang', this.lang);
    });
  },

  setState(s) {
    this.state = s;
    document.getElementById('voice-btn').dataset.state = s;
    const labels = { idle:'대기 중', listening:'듣는 중…', processing:'처리 중…',
                     speaking:'말하는 중…', error:'오류' };
    document.getElementById('voice-state').textContent = labels[s] || s;
  },

  showInterim(t) {
    document.getElementById('voice-transcript').textContent = t;
  },

  showTranscript(text, who) {
    const box = document.getElementById('voice-history');
    const d = document.createElement('div');
    d.className = 'bubble ' + who;
    d.textContent = text;
    box.appendChild(d);
    box.scrollTop = box.scrollHeight;
    document.getElementById('voice-transcript').textContent = '';
  }
};

document.addEventListener('DOMContentLoaded', () => {
  AI_VOICE.init(localStorage.getItem('apiKey') || '');
});
```
