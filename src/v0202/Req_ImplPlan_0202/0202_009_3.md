```javascript
// [U-001/U-004] 정본은 프리셋.factors × 스타일.factors (S20 미러)
//  - target?.params → target?.factors (JSON 실제 구조)
//  - REFRESH/WAVE → 실제 5종(BALANCE/ACTIVE/FOCUS/RELAX/SLEEP) factor 곱
_resolvePresetParams(presetCode, styleCode) {
    const presets = (this._profileData?.windDict?.presets) || [];
    const styles  = (this._profileData?.windDict?.styles)  || [];

    const preset = presets.find(p => p.code === presetCode);
    const style  = styles.find(s => s.code === (styleCode || 'BALANCE'));

    const pf = preset?.factors || {};
    const sf = style?.factors  || {};

    const clamp = v => Math.max(0, Math.min(100, Math.round(v)));
    const mul   = (base, f) => (base ?? 50) * (f ?? 1);

    return {
        intensity:   clamp(mul(pf.windIntensity,   sf.intensityFactor)),
        variability: clamp(mul(pf.windVariability, sf.variabilityFactor)),
        gustFreq:    clamp(mul(pf.gustFrequency,   sf.gustFactor))
    };
}
```

📄 B-2. S10_Simul_IO_0202.cpp — patchFromJson 재설계

```cpp
// [U-004] Q5-1 스키마 + Q6-1 개별 override + Q7-3 preset/style 변경 시 초기화
bool CL_S10_Simulation::patchFromJson(const JsonDocument& p_doc) {
    CL_A40_MutexGuard_Semaphore v_MutxGuard(_recursiveMutex, G_A40_MUTEX_TIMEOUT_100, __func__);
    if (!v_MutxGuard.isAcquired()) return false;

    JsonObjectConst v_sim = p_doc["sim"].as<JsonObjectConst>();
    if (v_sim.isNull()) return false;

    bool v_changed = false;
    bool v_presetChanged = false;
    bool v_styleChanged  = false;

    // [1] preset/style 변경 감지
    if (!v_sim["presetCode"].isNull() && v_sim["presetCode"].is<const char*>()) {
        const char* v_new = v_sim["presetCode"].as<const char*>();
        if (v_new && v_new[0] && strcasecmp(v_new, presetCode) != 0) {
            strlcpy(presetCode, v_new, sizeof(presetCode));
            v_changed = true; v_presetChanged = true;
        }
    }
    if (!v_sim["styleCode"].isNull() && v_sim["styleCode"].is<const char*>()) {
        const char* v_new = v_sim["styleCode"].as<const char*>();
        if (v_new && v_new[0] && strcasecmp(v_new, styleCode) != 0) {
            strlcpy(styleCode, v_new, sizeof(styleCode));
            v_changed = true; v_styleChanged = true;
        }
    }

    // [2] override 파싱 (Q5-1)
    ST_A20_SegmentOverride_t v_newOvr = { false, -1.0f, -1.0f, -1.0f };
    bool v_overrideProvided = false;
    if (!v_sim["override"].isNull() && v_sim["override"].is<JsonObjectConst>()) {
        JsonObjectConst v_ovr = v_sim["override"].as<JsonObjectConst>();
        v_newOvr.enabled = v_ovr["enabled"] | false;
        if (!v_ovr["windIntensity"].isNull())   v_newOvr.windIntensity   = v_ovr["windIntensity"].as<float>();
        if (!v_ovr["windVariability"].isNull()) v_newOvr.windVariability = v_ovr["windVariability"].as<float>();
        if (!v_ovr["gustFrequency"].isNull())   v_newOvr.gustFrequency   = v_ovr["gustFrequency"].as<float>();
        v_overrideProvided = true;
    }

    // [3] Q7-3: preset/style 재선택 시 override 해제
    if (v_presetChanged || v_styleChanged) {
        v_newOvr = { false, -1.0f, -1.0f, -1.0f };
        v_overrideProvided = true;
    }

    // [4] S20 resolve → 최종 물리 반영
    if (g_A20_config_root.windDict) {
        ST_A20_ResolvedWind_t v_resolved;
        memset(&v_resolved, 0, sizeof(v_resolved));
        bool v_ok = S20_resolveWindParams(
            *g_A20_config_root.windDict, presetCode, styleCode,
            (v_newOvr.enabled ? &v_newOvr : nullptr), v_resolved);
        if (v_ok && v_resolved.valid) {
            userIntensity   = A40_ComFunc::clampVal<float>(v_resolved.windIntensity,   0.0f, 100.0f);
            userVariability = A40_ComFunc::clampVal<float>(v_resolved.windVariability, 0.0f, 100.0f);
            userGustFreq    = A40_ComFunc::clampVal<float>(v_resolved.gustFrequency,   0.0f, 100.0f);
            fanLimitPct     = A40_ComFunc::clampVal<float>(v_resolved.fanLimit,        0.0f, 100.0f);
            minFanPct       = A40_ComFunc::clampVal<float>(v_resolved.minFan,          0.0f, 100.0f);
            turbLenScale    = v_resolved.turbulenceLengthScale;
            turbSigma       = v_resolved.turbulenceIntensitySigma;
            thermalStrength = v_resolved.thermalBubbleStrength;
            thermalRadius   = v_resolved.thermalBubbleRadius;
            v_changed = true;
        }
    }

    if (v_overrideProvided) _overrideState = v_newOvr;

    // [5] fanPowerEnabled (기존 유지)
    if (!v_sim["fanPowerEnabled"].isNull()) {
        bool v_new = v_sim["fanPowerEnabled"].as<bool>();
        if (v_new != fanPowerEnabled) { fanPowerEnabled = v_new; v_changed = true; }
    }

    return v_changed;
}
```

```javascript
// =========================================================
// 135.util.preview.js — SNW.Preview (가상 미리보기 엔진 + 정성 라벨)
// Layer 1 (유틸) — 998.offline의 resolvePreset 재활용
// [SIM-2] 30초 파형 시뮬레이션 (프론트 간이 엔진, ±10% 오차 명시)
// [SIM-8] "느낌" 5점 정성 라벨 (시원함/부드러움/역동성)
// =========================================================
window.SNW = window.SNW || {};
(function (SNW) {
    'use strict';

    SNW.Preview = {
        // ---------------------------------------------
        // [SIM-2] 30초 파형 시뮬레이션
        //  - preset × style factor 계산 (백엔드 S20 미러)
        //  - 100ms 해상도로 300개 포인트 생성
        // ---------------------------------------------
        generate(presetCode, styleCode, override) {
            const presets = SNW.Store.get()._presetsCache || [];
            const styles  = SNW.Store.get()._stylesCache  || [];
            const p = presets.find(x => x.code === presetCode);
            const s = styles.find(x => x.code === (styleCode || 'BALANCE'));
            if (!p || !s) return null;

            const pf = p.factors || {};
            const sf = s.factors  || {};
            const ov = override || { enabled: false };

            const mul = (base, f, ovr) => {
                if (ov.enabled && ovr != null && ovr >= 0) return ovr;
                return (base ?? 50) * (f ?? 1);
            };

            const intensity   = Math.max(0, Math.min(100,
                mul(pf.windIntensity,   sf.intensityFactor,   ov.windIntensity)));
            const variability = Math.max(0, Math.min(100,
                mul(pf.windVariability, sf.variabilityFactor, ov.windVariability)));
            const gustFreq    = Math.max(0, Math.min(100,
                mul(pf.gustFrequency,   sf.gustFactor,        ov.gustFrequency)));

            const baseMin = pf.baseMinWind ?? 1.8;
            const baseMax = pf.baseMaxWind ?? 5.5;
            const avg     = baseMin + (baseMax - baseMin) * (intensity / 100);
            const turbAmp = (variability / 100) * (baseMax - baseMin) * 0.3;
            const gustProb = (gustFreq / 100) * 0.03;  // 100ms당 3% × gustFreq%
            const gustMul  = pf.gustStrengthMax ?? 2.0;

            const points = [];
            let gustActive = false;
            let gustRemain = 0;

            for (let i = 0; i < 300; i++) {  // 30초 × 10Hz
                const t = i * 100;
                // 난류 (sin 합성, 3개 주파수)
                const turb = Math.sin(t * 0.002) * turbAmp * 0.5
                           + Math.sin(t * 0.0071) * turbAmp * 0.3
                           + Math.sin(t * 0.0137) * turbAmp * 0.2;

                // 돌풍 발생
                if (!gustActive && Math.random() < gustProb) {
                    gustActive = true;
                    gustRemain = 5 + Math.floor(Math.random() * 20);  // 0.5~2.5초
                }

                let gustBoost = 0;
                if (gustActive) {
                    gustBoost = avg * (gustMul - 1) * 0.7;
                    if (--gustRemain <= 0) gustActive = false;
                }

                const wind = Math.max(0.2, Math.min(11.0, avg + turb + gustBoost));
                points.push({
                    t,
                    wind: Number(wind.toFixed(2)),
                    avg: Number(avg.toFixed(2)),
                    gust: gustBoost > 0,
                    intensity, variability, gustFreq
                });
            }
            return { points, avg, intensity, variability, gustFreq };
        },

        // ---------------------------------------------
        // [SIM-8] 정성 라벨 (시원함/부드러움/역동성 0~5)
        // ---------------------------------------------
        toFeelings(intensity, variability, gustFreq) {
            const clamp = v => Math.max(0, Math.min(5, Math.round(v)));
            return {
                coolness:     clamp(intensity / 20),                    // 시원함
                smoothness:   clamp(5 - variability / 20),             // 부드러움 (변동성 반비례)
                dynamism:     clamp((variability + gustFreq) / 40),    // 역동성
                calmness:     clamp(5 - (intensity + gustFreq) / 40)   // 차분함
            };
        },

        // ---------------------------------------------
        // 라벨 → i18n 문자열
        // ---------------------------------------------
        labelFor(key) {
            return SNW.I18n.t('preview.feel.' + key);
        }
    };
})(window.SNW);
```

```cpp
#include "CT10_Monitor_0202.h"
#include <esp_heap_caps.h>
#include <time.h>
#include "A40_Com_Utils_0202.h"
#include "D10_Logger_0202.h"

// Warm 용량 (Phase 1)
static constexpr uint16_t WIND_CAP   = 3600;   // 1시간 @ 1초
static constexpr uint8_t  STATE_CAP  = 200;
static constexpr uint8_t  CONFIG_CAP = 200;
static constexpr uint8_t  SENSOR_CAP = 120;    // 1시간 @ 30초
static constexpr uint8_t  SYSTEM_CAP = 200;

// 정적 정의
ST_MON_WindRaw_t*    CL_MON_Monitor::s_wind   = nullptr;
ST_MON_StateEvent_t* CL_MON_Monitor::s_state  = nullptr;
ST_MON_ConfigEvent_t* CL_MON_Monitor::s_config = nullptr;
ST_MON_SensorRaw_t*  CL_MON_Monitor::s_sensor = nullptr;
ST_MON_SystemEvent_t* CL_MON_Monitor::s_system = nullptr;
bool CL_MON_Monitor::s_enabled = false;

uint16_t CL_MON_Monitor::s_windHead = 0,   CL_MON_Monitor::s_windCount = 0;
uint8_t  CL_MON_Monitor::s_stateHead = 0,  CL_MON_Monitor::s_stateCount = 0;
uint8_t  CL_MON_Monitor::s_configHead = 0, CL_MON_Monitor::s_configCount = 0;
uint8_t  CL_MON_Monitor::s_sensorHead = 0, CL_MON_Monitor::s_sensorCount = 0;
uint8_t  CL_MON_Monitor::s_systemHead = 0, CL_MON_Monitor::s_systemCount = 0;

uint32_t CL_MON_Monitor::s_lastWindMs = 0, CL_MON_Monitor::s_lastSensorMs = 0;

uint8_t CL_MON_Monitor::s_prevState = 255, CL_MON_Monitor::s_prevReason = 255;
uint8_t CL_MON_Monitor::s_prevSchId = 255, CL_MON_Monitor::s_prevSegNo = 255, CL_MON_Monitor::s_prevProfileNo = 255;
float   CL_MON_Monitor::s_prevIntensity = -1, CL_MON_Monitor::s_prevVariability = -1, CL_MON_Monitor::s_prevGustFreq = -1;
uint8_t CL_MON_Monitor::s_prevPreset = 255, CL_MON_Monitor::s_prevStyle = 255;

// --------------------------------------------------
// PSRAM 할당 헬퍼
// --------------------------------------------------
template <typename T>
static T* _allocPsram(size_t p_n) {
    T* p = (T*)heap_caps_malloc(p_n * sizeof(T), MALLOC_CAP_SPIRAM);
    if (p) memset(p, 0, p_n * sizeof(T));
    return p;
}

bool CL_MON_Monitor::begin() {
    if (s_enabled) return true;

    s_wind   = _allocPsram<ST_MON_WindRaw_t>(WIND_CAP);
    s_state  = _allocPsram<ST_MON_StateEvent_t>(STATE_CAP);
    s_config = _allocPsram<ST_MON_ConfigEvent_t>(CONFIG_CAP);
    s_sensor = _allocPsram<ST_MON_SensorRaw_t>(SENSOR_CAP);
    s_system = _allocPsram<ST_MON_SystemEvent_t>(SYSTEM_CAP);

    if (!s_wind || !s_state || !s_config || !s_sensor || !s_system) {
        CL_D10_Logger::log(EN_L10_LOG_ERROR, "[MON] PSRAM alloc failed");
        return false;
    }

    s_windHead = s_stateHead = s_configHead = s_sensorHead = s_systemHead = 0;
    s_windCount = s_stateCount = s_configCount = s_sensorCount = s_systemCount = 0;
    s_lastWindMs = s_lastSensorMs = 0;
    s_prevState = 255;
    s_enabled = true;

    CL_D10_Logger::log(EN_L10_LOG_INFO,
        "[MON] begin: wind=%dB state=%dB sensor=%dB system=%dB (Warm PSRAM)",
        WIND_CAP * sizeof(ST_MON_WindRaw_t),
        STATE_CAP * sizeof(ST_MON_StateEvent_t),
        SENSOR_CAP * sizeof(ST_MON_SensorRaw_t),
        SYSTEM_CAP * sizeof(ST_MON_SystemEvent_t));
    return true;
}

// --------------------------------------------------
// Epoch (SNTP 미동기 시 boot epoch + millis)
// --------------------------------------------------
void CL_MON_Monitor::_nowEpoch(uint32_t& p_out) {
    time_t t = time(nullptr);
    if ((uint32_t)t > 1700000000UL) {
        p_out = (uint32_t)t;
    } else {
        p_out = (uint32_t)(millis() / 1000);
    }
}

// --------------------------------------------------
// tick (1초 throttle)
// --------------------------------------------------
void CL_MON_Monitor::tick(const ST_MON_Snapshot_t& p) {
    if (!s_enabled) return;
    uint32_t now = millis();

    // 1초 wind 캡처
    if (now - s_lastWindMs >= 1000) {
        s_lastWindMs = now;
        _captureWind(p);
    }

    // 30초 sensor 캡처
    if (now - s_lastSensorMs >= 30000) {
        s_lastSensorMs = now;
        _captureSensor(p);
    }

    // 이벤트 즉시 캡처
    _captureStateChange(p);
    _captureConfigChanges(p);
}

// --------------------------------------------------
// Wind 캡처
// --------------------------------------------------
void CL_MON_Monitor::_captureWind(const ST_MON_Snapshot_t& p) {
    ST_MON_WindRaw_t& e = s_wind[s_windHead];
    _nowEpoch(e.t);
    e.wind_x10   = (uint16_t)(p.windSpeed   * 10.0f);
    e.target_x10 = (uint16_t)(p.targetWind  * 10.0f);
    e.pwm_x10    = (uint16_t)(p.pwmDuty     * 10.0f);
    e.preset_idx = p.presetIdx;
    e.flags      = (p.gustActive ? 0x01 : 0) | (p.thermalActive ? 0x02 : 0);

    s_windHead = (s_windHead + 1) % WIND_CAP;
    if (s_windCount < WIND_CAP) s_windCount++;
}

// --------------------------------------------------
// Sensor 캡처
// --------------------------------------------------
void CL_MON_Monitor::_captureSensor(const ST_MON_Snapshot_t& p) {
    ST_MON_SensorRaw_t& e = s_sensor[s_sensorHead];
    _nowEpoch(e.t);
    e.temp_x10  = (int16_t)(p.tempC * 10.0f);
    e.humidity  = (uint8_t)(p.humidity);
    e.motion    = p.motionActive ? 1 : 0;
    s_sensorHead = (s_sensorHead + 1) % SENSOR_CAP;
    if (s_sensorCount < SENSOR_CAP) s_sensorCount++;
}

// --------------------------------------------------
// State 변경 감지
// --------------------------------------------------
void CL_MON_Monitor::_captureStateChange(const ST_MON_Snapshot_t& p) {
    if (p.state == s_prevState && p.reason == s_prevReason
        && p.activeSchId == s_prevSchId && p.activeSegNo == s_prevSegNo
        && p.activeProfileNo == s_prevProfileNo) return;

    ST_MON_StateEvent_t& e = s_state[s_stateHead];
    _nowEpoch(e.t);
    e.prevState = (s_prevState == 255) ? p.state : s_prevState;
    e.newState  = p.state;
    e.reason    = p.reason;
    e.activeSchId = p.activeSchId;
    e.activeSegNo = p.activeSegNo;
    e.activeProfileNo = p.activeProfileNo;
    e.preset_idx = p.presetIdx;
    e.style_idx  = p.styleIdx;
    e.intensity  = (uint8_t)p.intensity;
    e.variability = (uint8_t)p.variability;
    e.gustFreq    = (uint8_t)p.gustFreq;

    s_stateHead = (s_stateHead + 1) % STATE_CAP;
    if (s_stateCount < STATE_CAP) s_stateCount++;

    s_prevState = p.state; s_prevReason = p.reason;
    s_prevSchId = p.activeSchId; s_prevSegNo = p.activeSegNo;
    s_prevProfileNo = p.activeProfileNo;
}

// --------------------------------------------------
// Config 변경 감지 (intensity, variability, gustFreq, preset, style)
// --------------------------------------------------
void CL_MON_Monitor::_captureConfigChanges(const ST_MON_Snapshot_t& p) {
    auto rec = [](uint8_t type, uint8_t oldV, uint8_t newV, uint8_t src) {
        ST_MON_ConfigEvent_t& e = s_config[s_configHead];
        CL_MON_Monitor::_nowEpoch(e.t);
        e.type = type; e.old_val = oldV; e.new_val = newV; e.source = src;
        s_configHead = (s_configHead + 1) % CONFIG_CAP;
        if (s_configCount < CONFIG_CAP) s_configCount++;
    };

    if (s_prevIntensity >= 0 && fabsf(p.intensity - s_prevIntensity) >= 1.0f) {
        rec(2, (uint8_t)s_prevIntensity, (uint8_t)p.intensity, 0);
    }
    if (s_prevVariability >= 0 && fabsf(p.variability - s_prevVariability) >= 1.0f) {
        rec(3, (uint8_t)s_prevVariability, (uint8_t)p.variability, 0);
    }
    if (s_prevGustFreq >= 0 && fabsf(p.gustFreq - s_prevGustFreq) >= 1.0f) {
        rec(4, (uint8_t)s_prevGustFreq, (uint8_t)p.gustFreq, 0);
    }
    if (s_prevPreset != 255 && p.presetIdx != s_prevPreset) {
        rec(0, s_prevPreset, p.presetIdx, 0);
    }
    if (s_prevStyle != 255 && p.styleIdx != s_prevStyle) {
        rec(1, s_prevStyle, p.styleIdx, 0);
    }

    s_prevIntensity = p.intensity;
    s_prevVariability = p.variability;
    s_prevGustFreq = p.gustFreq;
    s_prevPreset = p.presetIdx;
    s_prevStyle  = p.styleIdx;
}

// --------------------------------------------------
// [MON-2] 조회 + 다운샘플
// --------------------------------------------------
// range: "5m"=300 / "30m"=360 / "1h"=120
static int _rangeToIntervalSec(const char* p_range) {
    if (!p_range) return 1;
    if (strcmp(p_range, "5m")  == 0) return 1;
    if (strcmp(p_range, "30m") == 0) return 5;
    if (strcmp(p_range, "1h")  == 0) return 30;
    return 1;
}

static uint32_t _rangeToSec(const char* p_range) {
    if (!p_range) return 300;
    if (strcmp(p_range, "5m")  == 0) return 300;
    if (strcmp(p_range, "30m") == 0) return 1800;
    if (strcmp(p_range, "1h")  == 0) return 3600;
    return 300;
}

void CL_MON_Monitor::queryWind(const char* p_range, JsonDocument& p_doc) {
    int itv = _rangeToIntervalSec(p_range);
    uint32_t rangeSec = _rangeToSec(p_range);
    uint32_t nowEpoch; _nowEpoch(nowEpoch);
    uint32_t fromEpoch = (nowEpoch > rangeSec) ? (nowEpoch - rangeSec) : 0;

    JsonObject r = p_doc.to<JsonObject>();
    r["type"]  = "wind";
    r["range"] = p_range ? p_range : "5m";
    r["source"] = "warm";
    r["interval"] = itv;
    r["unit"] = "0.1m/s";

    JsonArray arr = r["data"].to<JsonArray>();
    int pointsOut = 0;

    if (itv == 1) {
        // 원본 그대로
        for (uint16_t i = 0; i < s_windCount; i++) {
            uint16_t idx = (s_windHead + WIND_CAP - s_windCount + i) % WIND_CAP;
            const ST_MON_WindRaw_t& e = s_wind[idx];
            if (fromEpoch > 0 && e.t < fromEpoch) continue;
            JsonObject o = arr.add<JsonObject>();
            o["t"] = e.t; o["avg"] = e.wind_x10; o["min"] = e.wind_x10; o["max"] = e.wind_x10;
            o["pwm_avg"] = e.pwm_x10; o["preset"] = e.preset_idx;
            o["gust"] = (e.flags & 0x01) ? 1 : 0;
            o["thermal"] = (e.flags & 0x02) ? 1 : 0;
            pointsOut++;
        }
    } else {
        // 다운샘플: itv 초 평균
        int64_t bucketT = 0; uint32_t sum = 0; uint32_t sumPwm = 0;
        uint16_t minW = 65535, maxW = 0; uint16_t cnt = 0;
        uint16_t gustCnt = 0, thermalCnt = 0;
        uint8_t curPreset = 0;
        bool flushed = false;

        for (uint16_t i = 0; i < s_windCount; i++) {
            uint16_t idx = (s_windHead + WIND_CAP - s_windCount + i) % WIND_CAP;
            const ST_MON_WindRaw_t& e = s_wind[idx];
            if (fromEpoch > 0 && e.t < fromEpoch) continue;

            int64_t thisBucket = (e.t / itv) * itv;
            if (bucketT == 0) bucketT = thisBucket;
            if (thisBucket != bucketT) {
                // flush
                if (cnt > 0) {
                    JsonObject o = arr.add<JsonObject>();
                    o["t"] = (uint32_t)bucketT;
                    o["avg"] = (uint16_t)(sum / cnt);
                    o["min"] = minW; o["max"] = maxW;
                    o["pwm_avg"] = (uint16_t)(sumPwm / cnt);
                    o["preset"] = curPreset;
                    o["gust"] = gustCnt;
                    o["thermal"] = thermalCnt;
                    pointsOut++;
                }
                bucketT = thisBucket; sum = 0; sumPwm = 0;
                minW = 65535; maxW = 0; cnt = 0; gustCnt = 0; thermalCnt = 0;
            }
            sum += e.wind_x10; sumPwm += e.pwm_x10;
            if (e.wind_x10 < minW) minW = e.wind_x10;
            if (e.wind_x10 > maxW) maxW = e.wind_x10;
            if (e.flags & 0x01) gustCnt++;
            if (e.flags & 0x02) thermalCnt++;
            curPreset = e.preset_idx;
            cnt++;
        }
        if (cnt > 0) {
            JsonObject o = arr.add<JsonObject>();
            o["t"] = (uint32_t)bucketT;
            o["avg"] = (uint16_t)(sum / cnt);
            o["min"] = minW; o["max"] = maxW;
            o["pwm_avg"] = (uint16_t)(sumPwm / cnt);
            o["preset"] = curPreset;
            o["gust"] = gustCnt;
            o["thermal"] = thermalCnt;
            pointsOut++;
        }
    }
    r["count"] = pointsOut;
}

void CL_MON_Monitor::queryState(const char* p_range, JsonDocument& p_doc) {
    uint32_t rangeSec = _rangeToSec(p_range);
    uint32_t nowEpoch; _nowEpoch(nowEpoch);
    uint32_t fromEpoch = (nowEpoch > rangeSec) ? (nowEpoch - rangeSec) : 0;

    JsonObject r = p_doc.to<JsonObject>();
    r["type"] = "state"; r["range"] = p_range ? p_range : "5m"; r["source"] = "warm";

    JsonArray arr = r["data"].to<JsonArray>();
    int n = 0;
    for (uint8_t i = 0; i < s_stateCount; i++) {
        uint8_t idx = (s_stateHead + STATE_CAP - s_stateCount + i) % STATE_CAP;
        const ST_MON_StateEvent_t& e = s_state[idx];
        if (fromEpoch > 0 && e.t < fromEpoch) continue;
        JsonObject o = arr.add<JsonObject>();
        o["t"] = e.t; o["prev"] = e.prevState; o["new"] = e.newState;
        o["reason"] = e.reason; o["schId"] = e.activeSchId;
        o["segNo"] = e.activeSegNo; o["profileNo"] = e.activeProfileNo;
        o["preset"] = e.preset_idx; o["style"] = e.style_idx;
        o["intensity"] = e.intensity; o["variability"] = e.variability; o["gust"] = e.gustFreq;
        n++;
    }
    r["count"] = n;
}

void CL_MON_Monitor::queryConfig(const char* p_range, JsonDocument& p_doc) {
    uint32_t rangeSec = _rangeToSec(p_range);
    uint32_t nowEpoch; _nowEpoch(nowEpoch);
    uint32_t fromEpoch = (nowEpoch > rangeSec) ? (nowEpoch - rangeSec) : 0;

    JsonObject r = p_doc.to<JsonObject>();
    r["type"] = "config"; r["range"] = p_range ? p_range : "5m";

    JsonArray arr = r["data"].to<JsonArray>();
    int n = 0;
    for (uint8_t i = 0; i < s_configCount; i++) {
        uint8_t idx = (s_configHead + CONFIG_CAP - s_configCount + i) % CONFIG_CAP;
        const ST_MON_ConfigEvent_t& e = s_config[idx];
        if (fromEpoch > 0 && e.t < fromEpoch) continue;
        JsonObject o = arr.add<JsonObject>();
        o["t"] = e.t; o["type"] = e.type; o["old"] = e.old_val;
        o["new"] = e.new_val; o["source"] = e.source;
        n++;
    }
    r["count"] = n;
}

void CL_MON_Monitor::querySensor(const char* p_range, JsonDocument& p_doc) {
    uint32_t rangeSec = _rangeToSec(p_range);
    uint32_t nowEpoch; _nowEpoch(nowEpoch);
    uint32_t fromEpoch = (nowEpoch > rangeSec) ? (nowEpoch - rangeSec) : 0;

    JsonObject r = p_doc.to<JsonObject>();
    r["type"] = "sensor"; r["range"] = p_range ? p_range : "5m"; r["source"] = "warm";

    JsonArray arr = r["data"].to<JsonArray>();
    int n = 0;
    for (uint8_t i = 0; i < s_sensorCount; i++) {
        uint8_t idx = (s_sensorHead + SENSOR_CAP - s_sensorCount + i) % SENSOR_CAP;
        const ST_MON_SensorRaw_t& e = s_sensor[idx];
        if (fromEpoch > 0 && e.t < fromEpoch) continue;
        JsonObject o = arr.add<JsonObject>();
        o["t"] = e.t; o["temp_avg"] = e.temp_x10;
        o["temp_min"] = e.temp_x10; o["temp_max"] = e.temp_x10;
        o["hum_avg"] = e.humidity; o["motion_ratio"] = e.motion ? 255 : 0;
        n++;
    }
    r["count"] = n;
}

void CL_MON_Monitor::querySystem(const char* p_range, JsonDocument& p_doc) {

        uint8_t idx = (s_systemHead + SYSTEM_CAP - s_systemCount + i) % SYSTEM_CAP;
```

```cpp
void CL_W10_WebAPI::routeMonitor() {
    s_server->on(W10_Const::HTTP_API_MONITOR, HTTP_GET, [](AsyncWebServerRequest* p_request) {
        if (!checkApiKey(p_request)) {
            p_request->send(401, "application/json", "{\"error\":\"unauthorized\"}");
            return;
        }
        if (!CL_MON_Monitor::isEnabled()) {
            p_request->send(503, "application/json", "{\"error\":\"monitor not enabled\"}");
            return;
        }

        const char* type  = p_request->hasParam("type")  ? p_request->getParam("type")->value().c_str()  : "wind";
        const char* range = p_request->hasParam("range") ? p_request->getParam("range")->value().c_str() : "5m";

        JsonDocument v_doc;
        if      (strcmp(type, "wind")   == 0) CL_MON_Monitor::queryWind  (range, v_doc);
        else if (strcmp(type, "state")  == 0) CL_MON_Monitor::queryState (range, v_doc);
        else if (strcmp(type, "config") == 0) CL_MON_Monitor::queryConfig(range, v_doc);
        else if (strcmp(type, "sensor") == 0) CL_MON_Monitor::querySensor(range, v_doc);
        else if (strcmp(type, "system") == 0) CL_MON_Monitor::querySystem(range, v_doc);
        else { p_request->send(400, "application/json", "{\"error\":\"invalid type\"}"); return; }

        sendJson(p_request, v_doc);
    });
}
```

```javascript
_animatePreview(canvas, points) {
    const ctx = canvas.getContext('2d');
    const w = canvas.width, h = canvas.height;
    let min = Infinity, max = -Infinity;
    points.forEach(p => { if (p.wind < min) min = p.wind; if (p.wind > max) max = p.wind; });
    if (max <= min) max = min + 1;
    let drawn = 0;
    const drawNext = () => {
        if (drawn >= points.length || !canvas.isConnected) return;
        ctx.clearRect(0, 0, w, h);
        ctx.strokeStyle = 'rgba(139,148,158,0.1)';
        ctx.lineWidth = 1;
        for (let i = 0; i <= 4; i++) {
            const y = (h / 4) * i;
            ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(w, y); ctx.stroke();
        }
        ctx.strokeStyle = 'rgba(74,158,255,1)';
        ctx.lineWidth = 2;
        ctx.beginPath();
        for (let i = 0; i <= drawn; i++) {
            const p = points[i];
            const x = (i / (points.length - 1)) * w;
            const y = h - ((p.wind - min) / (max - min)) * (h - 20) - 10;
            if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
        }
        ctx.stroke();
        for (let i = 0; i <= drawn; i++) {
            if (!points[i].gust) continue;
            const x = (i / (points.length - 1)) * w;
            const y = h - ((points[i].wind - min) / (max - min)) * (h - 20) - 10;
            ctx.fillStyle = 'rgba(240,136,62,0.9)';
            ctx.beginPath(); ctx.arc(x, y, 3, 0, Math.PI * 2); ctx.fill();
        }
        drawn += 3;
        setTimeout(drawNext, 30);
    };
    drawNext();
}
```

```cpp
// --------------------------------------------------
// _writeHeader (파일 없으면 생성, 있으면 무시)
// --------------------------------------------------
void CL_MON_Monitor::_writeHeader(const char* p_path, uint16_t p_entrySize, uint32_t p_capacity) {
    if (LittleFS.exists(p_path)) {
        ST_MON_FileHeader_t h;
        if (_readHeader(p_path, h) && h.magic == 0x4D4E5731u && h.entry_size == p_entrySize) {
            return;  // 이미 유효
        }
    }
    File f = LittleFS.open(p_path, "w");
    if (!f) { CL_D10_Logger::log(EN_L10_LOG_ERROR, "[MON] header open fail: %s", p_path); return; }
    ST_MON_FileHeader_t h = {};
    h.magic = 0x4D4E5731u;  // "MNW1"
    h.version = 1;
    h.entry_size = p_entrySize;
    h.capacity = p_capacity;
    f.write((const uint8_t*)&h, sizeof(h));
    f.close();
    CL_D10_Logger::log(EN_L10_LOG_INFO, "[MON] Cold file created: %s (cap=%u)", p_path, p_capacity);
}

// --------------------------------------------------
// tick()에 5분 집계 추가
// --------------------------------------------------
void CL_MON_Monitor::tick(const ST_MON_Snapshot_t& p) {
    if (!s_enabled) return;
    uint32_t now = millis();

    if (now - s_lastWindMs >= 1000) {
        s_lastWindMs = now;
        _captureWind(p);
        // [MON-D] 방금 캡처한 wind를 5분 aggregator로
        _aggregateWind(s_wind[(s_windHead + WIND_CAP - 1) % WIND_CAP]);
    }

    if (now - s_lastSensorMs >= 30000) {
        s_lastSensorMs = now;
        _captureSensor(p);
        _aggregateSensor(s_sensor[(s_sensorHead + SENSOR_CAP - 1) % SENSOR_CAP]);
    }

    _captureStateChange(p);
    _captureConfigChanges(p);

    // [MON-D] 1시간 배치 flush
    if (now - s_lastFlushMs >= 3600000UL) {  // 1시간
        s_lastFlushMs = now;
        flushCold();
    }
}

// --------------------------------------------------
// Wind 5분 집계
// --------------------------------------------------
void CL_MON_Monitor::_aggregateWind(const ST_MON_WindRaw_t& p_e) {
    uint32_t bucket = (p_e.t / 300) * 300;  // 5분 정렬
    if (s_agg.bucketStart == 0) s_agg.bucketStart = bucket;
    if (bucket != s_agg.bucketStart) {
        // bucket 경계 → 다음 bucket으로 (flush는 1시간 주기이므로 지금은 계속 누적)
        // 단, 5분 경계마다 별도 엔트리로 저장하려면 여기서 s_pendingWind5min[]에 추가
        // **Simplicity First**: Phase 2는 1시간 flush 시 마지막 5분 평균만 저장
        s_agg.windSum = 0; s_agg.pwmSum = 0;
        s_agg.windMin = 65535; s_agg.windMax = 0; s_agg.windCnt = 0;
        s_agg.gustCnt = 0; s_agg.thermalCnt = 0;
        s_agg.bucketStart = bucket;
    }
    s_agg.windSum += p_e.wind_x10;
    s_agg.pwmSum  += p_e.pwm_x10;
    if (p_e.wind_x10 < s_agg.windMin) s_agg.windMin = p_e.wind_x10;
    if (p_e.wind_x10 > s_agg.windMax) s_agg.windMax = p_e.wind_x10;
    if (p_e.flags & 0x01) s_agg.gustCnt++;
    if (p_e.flags & 0x02) s_agg.thermalCnt++;
    s_agg.presetIdx = p_e.preset_idx;
    s_agg.windCnt++;
}

void CL_MON_Monitor::_aggregateSensor(const ST_MON_SensorRaw_t& p_e) {
    s_agg.sensorSum += (uint32_t)(p_e.temp_x10 + 1000);  // 음수 회피
    s_agg.sensorCnt++;
    if (p_e.temp_x10 < s_agg.tempMin) s_agg.tempMin = p_e.temp_x10;
    if (p_e.temp_x10 > s_agg.tempMax) s_agg.tempMax = p_e.temp_x10;
    s_agg.humSum += p_e.humidity;
    if (p_e.motion) s_agg.motionCnt++;
}

// --------------------------------------------------
// 1시간 배치 flush
// --------------------------------------------------
void CL_MON_Monitor::flushCold() {
    if (!s_enabled) return;
    if (s_agg.windCnt == 0 && s_agg.sensorCnt == 0) return;  // 빈 집계 스킵

    // [1] Wind 5분 평균 1건 저장 (이번 시간의 마지막 집계)
    if (s_agg.windCnt > 0) {
        ST_MON_Wind5Min_t w = {};
        w.t = s_agg.bucketStart;
        w.avg_x10 = s_agg.windSum / s_agg.windCnt;
        w.min_x10 = s_agg.windMin;
        w.max_x10 = s_agg.windMax;
        w.pwm_avg_x10 = s_agg.pwmSum / s_agg.windCnt;
        w.preset_idx = s_agg.presetIdx;
        w.gust_count = s_agg.gustCnt > 255 ? 255 : s_agg.gustCnt;
        w.thermal_count = s_agg.thermalCnt > 255 ? 255 : s_agg.thermalCnt;
        _appendFile(PATH_WIND, &w, sizeof(w), COLD_WIND_CAP);
    }

    // [2] Sensor 5분 평균 1건
    if (s_agg.sensorCnt > 0) {
        ST_MON_Sensor5Min_t s = {};
        s.t = s_agg.bucketStart;
        int32_t avg = (int32_t)(s_agg.sensorSum / s_agg.sensorCnt) - 1000;
        s.temp_avg_x10 = (int16_t)avg;
        s.temp_min_x10 = s_agg.tempMin;
        s.temp_max_x10 = s_agg.tempMax;
        s.hum_avg = (uint8_t)(s_agg.humSum / s_agg.sensorCnt);
        s.motion_ratio = (uint8_t)((s_agg.motionCnt * 255) / s_agg.sensorCnt);
        _appendFile(PATH_SENSOR, &s, sizeof(s), COLD_SENSOR_CAP);
    }

    // [3] 이벤트는 마지막 flush 이후 신규만 append (Warm에서 순회)
    // State
    for (uint8_t i = 0; i < s_stateCount; i++) {
        uint8_t idx = (s_stateHead + STATE_CAP - s_stateCount + i) % STATE_CAP;
        const ST_MON_StateEvent_t& e = s_state[idx];
        if (e.t <= s_agg.lastStateEpochFlushed) continue;
        _appendFile(PATH_STATE, &e, sizeof(e), COLD_STATE_CAP);
    }
    if (s_stateCount > 0) {
        s_agg.lastStateEpochFlushed =
            s_state[(s_stateHead + STATE_CAP - 1) % STATE_CAP].t;
    }
    // Config
    for (uint8_t i = 0; i < s_configCount; i++) {
        uint8_t idx = (s_configHead + CONFIG_CAP - s_configCount + i) % CONFIG_CAP;
        const ST_MON_ConfigEvent_t& e = s_config[idx];
        if (e.t <= s_agg.lastConfigEpochFlushed) continue;
        _appendFile(PATH_CONFIG, &e, sizeof(e), COLD_CONFIG_CAP);
    }
    if (s_configCount > 0) {
        s_agg.lastConfigEpochFlushed =
            s_config[(s_configHead + CONFIG_CAP - 1) % CONFIG_CAP].t;
    }
    // System
    for (uint8_t i = 0; i < s_systemCount; i++) {
        uint8_t idx = (s_systemHead + SYSTEM_CAP - s_systemCount + i) % SYSTEM_CAP;
        const ST_MON_SystemEvent_t& e = s_system[idx];
        if (e.t <= s_agg.lastSystemEpochFlushed) continue;
        _appendFile(PATH_SYSTEM, &e, sizeof(e), COLD_SYSTEM_CAP);
    }
    if (s_systemCount > 0) {
        s_agg.lastSystemEpochFlushed =
            s_system[(s_systemHead + SYSTEM_CAP - 1) % SYSTEM_CAP].t;
    }

    // 집계 초기화 (다음 1시간을 위해)
    memset(&s_agg, 0, sizeof(s_agg));
    s_agg.windMin = 65535;
    s_agg.tempMin = 32767; s_agg.tempMax = -32768;

    CL_D10_Logger::log(EN_L10_LOG_INFO,
        "[MON] Cold flush done (wind=%u state=%u config=%u sensor=%u system=%u)",
        s_windCount, s_stateCount, s_configCount, s_sensorCount, s_systemCount);
}

// --------------------------------------------------
// 파일 append (ring)
// --------------------------------------------------
bool CL_MON_Monitor::_appendFile(const char* p_path, const void* p_entry,
                                 uint16_t p_entrySize, uint32_t p_capacity) {
    ST_MON_FileHeader_t h;
    if (!_readHeader(p_path, h)) return false;
    if (h.entry_size != p_entrySize) return false;

    uint32_t writePos = h.head;
    size_t offset = sizeof(h) + writePos * p_entrySize;

    File f = LittleFS.open(p_path, "r+");
    if (!f) return false;
    f.seek(offset);
    size_t written = f.write((const uint8_t*)p_entry, p_entrySize);

    if (written != p_entrySize) { f.close(); return false; }

    // 헤더 갱신
    h.head = (h.head + 1) % h.capacity;
    if (h.count < h.capacity) h.count++;
    if (h.first_epoch == 0) h.first_epoch = ((const ST_MON_Wind5Min_t*)p_entry)->t;
    h.last_epoch = ((const ST_MON_Wind5Min_t*)p_entry)->t;
    f.seek(0);
    f.write((const uint8_t*)&h, sizeof(h));
    f.close();
    return true;
}
```

```cpp
    {
```

```javascript
_drawOverlay() {
    const chartRoot = this.el.querySelector('[data-slot="chart"]');
    if (!chartRoot) return;
    // 기존 오버레이 제거
    chartRoot.querySelectorAll('.snw-chart-overlay').forEach(c => c.remove());

    const sim = SNW.Store.get().sim || {};
    const pv = SNW.Preview.generate(sim.presetCode || 'OCEAN', sim.styleCode || 'BALANCE', null);
    if (!pv) return;

    const cv = document.createElement('canvas');
    cv.className = 'snw-chart-overlay';
    cv.style.cssText = 'position:absolute;inset:0;pointer-events:none;';
    chartRoot.style.position = 'relative';
    chartRoot.appendChild(cv);

    const rect = chartRoot.getBoundingClientRect();
    cv.width = rect.width; cv.height = rect.height;
    const ctx = cv.getContext('2d');
    const w = cv.width, h = cv.height;

    const vals = pv.points.map(p => p.wind);
    const min = Math.min(...vals), max = Math.max(...vals);
    const range = Math.max(1, max - min);

    ctx.strokeStyle = 'rgba(240,136,62,0.5)';
    ctx.setLineDash([4, 4]);
    ctx.lineWidth = 1.5;
    ctx.beginPath();
    pv.points.forEach((p, i) => {
        const x = (i / (pv.points.length - 1)) * w;
        const y = h - ((p.wind - min) / range) * (h - 20) - 10;
        if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
    });
    ctx.stroke();
}
```
