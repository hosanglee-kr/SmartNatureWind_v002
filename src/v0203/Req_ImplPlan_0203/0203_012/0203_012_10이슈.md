def check_ai_limits(repo: Path) -> list[str]:
    """SPEC §5.1.4.4 ↔ W10_Web_0203.h 대조"""
    spec = (repo / "src/v0203/contract_doc_0203/SPEC_0203.md").read_text(encoding="utf-8")
    hdr = (repo / "src/v0203/W10_Web_0203.h").read_text(encoding="utf-8")

    violations = []
    # G_MAX_REQUEST_BODY 값 추출
    m_h = re.search(r"G_MAX_REQUEST_BODY\s*=\s*(\d+)", hdr)
    m_s = re.search(r"`G_MAX_REQUEST_BODY\s*=\s*(\d+)`", spec)
    if m_h and m_s and m_h.group(1) != m_s.group(1):
        violations.append(f"G_MAX_REQUEST_BODY: SPEC={m_s.group(1)} vs 헤더={m_h.group(1)}")
    return violations

def check_fsm_names(repo: Path) -> list[str]:
    """C++ FSM 상태명 ↔ offline 어댑터 대조"""
    ctl = (repo / "src/v0203/CT10_Ctl_StMG_0203.cpp").read_text(encoding="utf-8")
    offline = (repo / "src/v0203/data_v0203_src/html_v0203/js/998.offline_0203.js").read_text(encoding="utf-8")

    valid_states = set(re.findall(r'return "([A-Z_]+)";', ctl))
    used_states = set(re.findall(r"state\s*[:=]\s*'([A-Z_]+)'", offline))

    violations = []
    for st in used_states:
        if st and st not in valid_states and st not in ("ACTIVE", "PAUSED", "COMPLETED", "NONE"):
            violations.append(f"offline state '{st}' not in C++ FSM")
    return violations

def main() -> int:
    repo = Path(__file__).resolve().parents[3]
    all_v = []
    all_v += check_api_paths(repo)
    all_v += check_ai_limits(repo)
    all_v += check_fsm_names(repo)

    if all_v:
        print(f"[FAIL] {len(all_v)} SSOT violations:")
        for v in all_v:
            print(f"  - {v}")
        return 1
    print("[OK] All SSOT contracts consistent.")
    return 0

if __name__ == "__main__":
    sys.exit(main())
